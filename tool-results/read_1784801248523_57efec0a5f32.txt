   351→/* ─── TicketCard Sub-component ────────────────────────── */
   352→
   353→function TicketCard({ tk, language, isBn, fontClass, navigate }: {
   354→  tk: Ticket;
   355→  language: 'en' | 'bn';
   356→  isBn: boolean;
   357→  fontClass: string;
   358→  navigate: (page: string, params?: Record<string, string>) => void;
   359→}) {
   360→  const Icon = transportIcons[tk.transportType] || Bus;
   361→  const color = transportColors[tk.transportType] || transportColors.bus;
   362→  const isOnlineCopy = tk.ticketType === 'online_copy';
   363→
   364→  const getSeatClassLabel = (id: string) => {
   365→    const cls = BUS_CLASSES.find((c) => c.id === id);
   366→    return cls ? (isBn ? cls.labelBn : cls.label) : id;
   367→  };
   368→
   369→  const getSaveInfo = (tk: Ticket) => {
   370→    if (!tk.originalPrice || tk.originalPrice <= tk.price) return { amount: 0, percent: 0 };
   371→    const saved = tk.originalPrice - tk.price;
   372→    const percent = Math.round((saved / tk.originalPrice) * 100);
   373→    return { amount: saved, percent };
   374→  };
   375→
   376→  const saveInfo = getSaveInfo(tk);
   377→  const hasDiscount = saveInfo.amount > 0;
   378→
   379→  return (
   380→    <Card
   381→      className="cursor-pointer hover:border-primary/30 hover:shadow-md transition-all group h-full flex flex-col"
   382→      onClick={() => navigate('ticket-details', { id: tk.id })}
   383→    >
   384→      <CardContent className="p-0 flex flex-col h-full">
   385→        {/* ── Row 1: Transport Type | Ticket Type | Save ── */}
   386→        <div className="flex items-center justify-between px-3 pt-3 pb-1.5">
   387→          <div className="flex items-center gap-1.5">
   388→            <Badge className={`${color} text-[10px] gap-0.5 font-semibold`}>
   389→              <Icon className="w-3 h-3" />
   390→              {isBn ? TRANSPORT_TYPES.find(x => x.id === tk.transportType)?.labelBn || tk.transportType : TRANSPORT_TYPES.find(x => x.id === tk.transportType)?.label || tk.transportType}
   391→            </Badge>
   392→            <Badge variant="outline" className={`text-[10px] font-medium ${
   393→              isOnlineCopy ? 'border-emerald-300 text-emerald-700 dark:border-emerald-700 dark:text-emerald-400'
   394→                : 'border-amber-300 text-amber-700 dark:border-amber-700 dark:text-amber-400'
   395→            }`}>
   396→              {isOnlineCopy
   397→                ? (<span className="flex items-center gap-0.5"><FileText className="w-2.5 h-2.5" />{t('onlineCopy', language)}</span>)
   398→                : (<span className="flex items-center gap-0.5"><ImageIcon className="w-2.5 h-2.5" />{t('counterCopy', language)}</span>)
   399→              }
   400→            </Badge>
   401→          </div>
   402→          {hasDiscount && (
   403→            <Badge className="bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400 text-[10px] gap-0.5 font-semibold">
   404→              <Tag className="w-3 h-3" />
   405→              {isBn ? `৳${saveInfo.amount.toLocaleString()} সাশ্রয় (${saveInfo.percent}%)` : `Save ৳${saveInfo.amount.toLocaleString()} (${saveInfo.percent}%)`}
   406→            </Badge>
   407→          )}
   408→        </div>
   409→
   410→        {/* ── Row 2: From → To ── */}
   411→        <div className="px-3 pb-1">
   412→          <div className="flex items-center gap-1.5">
   413→            <span className={`font-bold text-base ${fontClass}`}>{tk.from}</span>
   414→            <ArrowRight className="w-3.5 h-3.5 text-primary shrink-0" />
   415→            <span className={`font-bold text-base ${fontClass}`}>{tk.to}</span>
   416→          </div>
   417→        </div>
   418→
   419→        {/* ── Row 3: Transport Company ── */}
   420→        <div className="px-3 pb-1.5">
   421→          <p className={`text-xs text-muted-foreground truncate ${fontClass}`}>
   422→            {tk.transportCompany}
   423→          </p>
   424→        </div>
   425→
   426→        {/* ── Row 4: Date | Time ── */}
   427→        <div className="px-3 pb-1.5">
   428→          <div className="flex items-center gap-3 text-xs text-muted-foreground">
   429→            <span className="flex items-center gap-1">
   430→              <Calendar className="w-3 h-3 text-primary" />
   431→              <span className={`font-medium ${fontClass}`}>{formatDepartureDate(tk.departureDate, language)}</span>
   432→            </span>
   433→            <span className="flex items-center gap-1">
   434→              <Clock className="w-3 h-3 text-primary" />
   435→              <span className="font-medium">{formatDepartureTime(tk.departureTime, language)}</span>
   436→            </span>
   437→          </div>
   438→        </div>
   439→
   440→        {/* ── Row 5: Class | Seat ── */}
   441→        {(tk.seatClass || tk.seatNumber) && (
   442→          <div className="px-3 pb-1.5">
   443→            <div className="flex items-center gap-2 text-[10px]">
   444→              {tk.seatClass && (
   445→                <span className="inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded-md bg-muted/60 font-medium">
   446→                  <Armchair className="w-3 h-3 text-primary" />
   447→                  <span className={fontClass}>{getSeatClassLabel(tk.seatClass)}</span>
   448→                </span>
   449→              )}
   450→              {tk.seatNumber && (
   451→                <span className="inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded-md bg-muted/60 font-medium">
   452→                  {isBn ? 'আসন' : 'Seat'}: {tk.seatNumber}
   453→                </span>
   454→              )}
   455→            </div>
   456→          </div>
   457→        )}
   458→
   459→        {/* ── Spacer to push footer down ── */}
   460→        <div className="flex-1 min-h-[4px]" />
   461→
   462→        {/* ── Row 6: Price | View Ticket ── */}
   463→        <div className="flex items-center justify-between px-3 py-2.5 bg-muted/30 rounded-b-lg border-t border-border/50 mt-auto">
   464→          <div className="flex items-center gap-2">
   465→            {hasDiscount ? (
   466→              <>
   467→                <span className={`text-xs line-through text-muted-foreground ${fontClass}`}>
   468→                  ৳{tk.originalPrice.toLocaleString()}
   469→                </span>
   470→                <span className={`text-base font-bold text-primary ${fontClass}`}>
   471→                  ৳{tk.price.toLocaleString()}
   472→                </span>
   473→              </>
   474→            ) : (
   475→              <span className={`text-base font-bold text-primary ${fontClass}`}>
   476→                ৳{tk.price.toLocaleString()}
   477→              </span>
   478→            )}
   479→          </div>
   480→          <Button
   481→            size="sm"
   482→            className="min-h-[32px] bg-primary group-hover:shadow-md transition-shadow gap-1 text-xs"
   483→            onClick={(e) => { e.stopPropagation(); navigate('ticket-details', { id: tk.id }); }}
   484→          >
   485→            {isBn ? 'টিকেট দেখুন' : 'View Ticket'}
   486→            <ChevronRight className="w-3 h-3" />
   487→          </Button>
   488→        </div>
   489→      </CardContent>
   490→    </Card>
   491→  );
   492→}

   664→    selectedTicketTypes,
   665→    setSelectedTicketTypes,
   666→    minPrice,
   667→    setMinPrice,
   668→    maxPrice,
   669→    setMaxPrice,
   670→    selectedDepartureTime,
   671→    setSelectedDepartureTime,
   672→    selectedClasses,
   673→    setSelectedClasses,
   674→    selectedSeats,
   675→    setSelectedSeats,

   695→          </div>
   696→        </div>
   697→
   698→        {/* ── Search Bar (Route / Date selectors) ── */}
   699→        <Card className="mb-4 border-primary/10">
   700→          <CardContent className="p-3 sm:p-4">
   701→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
   702→              <Select value={from} onValueChange={setFrom}>
   703→                <SelectTrigger className="h-10 w-full">
   704→                  <MapPin className="w-4 h-4 mr-1 shrink-0 text-primary" />
   705→                  <SelectValue placeholder={t('from', language)} />
   706→                </SelectTrigger>
   707→                <SelectContent className="max-h-64">
   708→                  {ALL_BD_DISTRICTS.map((dist) => (

   879→                {/* ── Pagination ── */}
   880→                {totalPages > 1 && (

   907→                        <PaginationItem>

   910→                            className={currentPage >= totalPages ? 'pointer-events-none opacity-50' : 'cursor-pointer'}
   911→                          />
   912→                        </PaginationItem>
   913→                      </PaginationContent>
   914→                    </Pagination>

   917→              </>

   919→          </main>
   920→        </div>
   921→      </div>
   922→
   923→      {/* ── Mobile Filter Sheet (Drawer from bottom) ── */}
   924→      <Sheet open={mobileFilterOpen} onOpenChange={setMobileFilterOpen}>

   926→          <SheetHeader>
   927→            <SheetTitle className={fontClass}>{t('filterBy', language)}</SheetTitle>
   928→            <SheetDescription className={fontClass}>
   929→              {isBn ? 'আপনার পছন্দ অনুযায়ী টিকেট ফিল্টার করুন' : 'Filter tickets according to your preferences'}
   930→            </SheetDescription>
   931→          </SheetHeader>
   932→          <div className="flex-1 overflow-y-auto custom-scrollbar py-2">
   933→            <FilterSidebar {...filterSidebarProps} />
   934→          </div>
   935→          <SheetFooter className="flex-row gap-2 pt-2 border-t">
   936→            <Button
   937→              variant="outline"
   938→              className="flex-1"
   939→              disabled={activeFilterCount === 0}
   940→              onClick={() => { clearAllFilters(); setMobileFilterOpen(false); }}
   941→            >
   942→              <X className="w-3.5 h-3.5 mr-1.5" />
   943→              <span className={fontClass}>{t('clearFilters', language)}</span>
   944→            </Button>
   945→            <Button

   950→            </Button>
   951→          </SheetFooter>

   954→    </div>
   955→  );
   956→}
