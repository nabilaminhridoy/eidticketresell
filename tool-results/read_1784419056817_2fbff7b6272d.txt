     1→'use client';
     2→

     8→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     9→import { Label } from '@/components/ui/label';
    10→import { Checkbox } from '@/components/ui/checkbox';
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';

    13→import { Badge } from '@/components/ui/badge';

    15→import {
    16→  Ticket, Eye, EyeOff, Check, X, Loader2, ArrowLeft, ArrowRight,

    21→
    22→type Step = 1 | 2 | 3;
    23→
    24→export default function RegisterPage() {
    25→  const { navigate } = useAppStore();
    26→  const { login } = useAuthStore();
    27→  const { language } = useLanguageStore();
    28→  const isBn = language === 'bn';
    29→  const fontClass = isBn ? 'font-bangla' : '';
    30→
    31→  // Step management
    32→  const [currentStep, setCurrentStep] = useState<Step>(1);
    33→
    34→  // Personal Info fields
    35→  const [fullName, setFullName] = useState('');
    36→  const [username, setUsername] = useState('');
    37→  const [email, setEmail] = useState('');
    38→  const [phone, setPhone] = useState('');
    39→  const [gender, setGender] = useState('');
    40→  const [dateOfBirth, setDateOfBirth] = useState('');
    41→  const [password, setPassword] = useState('');
    42→  const [confirmPassword, setConfirmPassword] = useState('');
    43→  const [showPassword, setShowPassword] = useState(false);
    44→  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
    45→
    46→  // Username availability
    47→  const [usernameStatus, setUsernameStatus] = useState<'checking' | 'available' | 'taken' | null>(null);
    48→  const [usernameDebounce, setUsernameDebounce] = useState<NodeJS.Timeout | null>(null);
    49→
    50→  // Agreements
    51→  const [agreeAge, setAgreeAge] = useState(false);
    52→  const [agreeTerms, setAgreeTerms] = useState(false);
    53→  const [agreeNotifications, setAgreeNotifications] = useState(false);
    54→
    55→  // OTP verification
    56→  const [emailOtp, setEmailOtp] = useState('');
    57→  const [phoneOtp, setPhoneOtp] = useState('');
    58→  const [emailOtpSent, setEmailOtpSent] = useState(false);
    59→  const [phoneOtpSent, setPhoneOtpSent] = useState(false);
    60→  const [emailOtpVerified, setEmailOtpVerified] = useState(false);
    61→  const [phoneOtpVerified, setPhoneOtpVerified] = useState(false);
    62→  const [sendingOtp, setSendingOtp] = useState<'email' | 'phone' | null>(null);
    63→  const [verifyingOtp, setVerifyingOtp] = useState<'email' | 'phone' | null>(null);
    64→
    65→  // General
    66→  const [loading, setLoading] = useState(false);
    67→  const [error, setError] = useState('');
    68→  const [resendTimer, setResendTimer] = useState({ email: 0, phone: 0 });
    69→
    70→  // Resend timer countdown
    71→  useEffect(() => {
    72→    const interval = setInterval(() => {
    73→      setResendTimer(prev => ({
    74→        email: Math.max(0, prev.email - 1),
    75→        phone: Math.max(0, prev.phone - 1),
    76→      }));

    78→    return () => clearInterval(interval);
    79→  }, []);
    80→

    82→  const getPasswordStrength = useCallback(() => {
    83→    if (!password) return { score: 0, label: '', color: '' };
    84→    let score = 0;
    85→    if (password.length >= 8) score++;
    86→    if (/[A-Z]/.test(password)) score++;
    87→    if (/[a-z]/.test(password)) score++;
    88→    if (/\d/.test(password)) score++;
    89→    if (/[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password)) score++;
    90→
    91→    const levels = [
    92→      { score: 0, label: '', color: '' },
    93→      { score: 1, label: isBn ? 'দুর্বল' : 'Weak', color: 'bg-red-500' },
    94→      { score: 2, label: isBn ? 'মাঝারি' : 'Fair', color: 'bg-orange-500' },
    95→      { score: 3, label: isBn ? 'ভালো' : 'Good', color: 'bg-yellow-500' },
    96→      { score: 4, label: isBn ? 'শক্তিশালী' : 'Strong', color: 'bg-emerald-500' },
    97→      { score: 5, label: isBn ? 'শক্তিশালী' : 'Very Strong', color: 'bg-emerald-600' },
    98→    ];
    99→    return levels[score] || levels[0];
   100→  }, [password, isBn]);
   101→
   102→  const passwordStrength = getPasswordStrength();
   103→  const passwordChecks = {
   104→    length: password.length >= 8,
   105→    uppercase: /[A-Z]/.test(password),
   106→    lowercase: /[a-z]/.test(password),
   107→    number: /\d/.test(password),
   108→    special: /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password),
   109→  };
   110→
   111→  // Username availability check with debounce
   112→  useEffect(() => {
   113→    if (usernameDebounce) clearTimeout(usernameDebounce);
   114→
   115→    if (!username || username.length < 3) {
   116→      setUsernameStatus(null);
   117→      return;
   118→    }
   119→
   120→    setUsernameStatus('checking');

   122→      try {
   123→        const res = await fetch(`/api/auth/check-username?username=${encodeURIComponent(username)}`);
   124→        const data = await res.json();
   125→        setUsernameStatus(data.available ? 'available' : 'taken');
   126→      } catch {
   127→        setUsernameStatus(null);
   128→      }
   129→    }, 500);
   130→    setUsernameDebounce(timeout);
   131→  }, [username]);
   132→
   133→  // Phone number formatting
   134→  const formatPhone = (value: string): string => {
   135→    const digits = value.replace(/\D/g, '');
   136→    if (digits.startsWith('880')) {
   137→      return '+' + digits;
   138→    }
   139→    if (digits.startsWith('01') && digits.length <= 11) {
   140→      return '+88' + digits;
   141→    }
   142→    if (value.startsWith('+8801') || value.startsWith('+880')) {
   143→      return '+' + digits;
   144→    }
   145→    return value;
   146→  };

   148→  const handlePhoneChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   149→    const formatted = formatPhone(e.target.value);
   150→    setPhone(formatted);
   151→  };

   153→  // Send OTP
   154→  const handleSendOtp = async (type: 'email' | 'phone') => {
   155→    setSendingOtp(type);
   156→    try {
   157→      const body: Record<string, string> = { type: type === 'email' ? 'email_verification' : 'phone_verification' };
   158→      if (type === 'email') body.email = email;
   159→      else body.phone = phone;
   160→
   161→      const res = await fetch('/api/auth/otp/send', {
   162→        method: 'POST',
   163→        headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify(body),
   165→      });
   166→      const data = await res.json();
   167→      if (!res.ok) throw new Error(data.error || 'Failed to send OTP');
   168→
   169→      if (type === 'email') {
   170→        setEmailOtpSent(true);
   171→        setResendTimer(prev => ({ ...prev, email: 60 }));
   172→      } else {
   173→        setPhoneOtpSent(true);
   174→        setResendTimer(prev => ({ ...prev, phone: 60 }));
   175→      }
   176→
   177→      toast.success(type === 'email' ? t('otpSentToEmail', language) : t('otpSentToPhone', language));
   178→
   179→      // In development, show the OTP
   180→      if (data.otp) {
   181→        toast.info(`OTP: ${data.otp}`, { duration: 10000 });
   182→      }
   183→    } catch (err) {
   184→      toast.error(err instanceof Error ? err.message : 'Failed to send OTP');
   185→    } finally {
   186→      setSendingOtp(null);
   187→    }
   188→  };
   189→
   190→  // Verify OTP
   191→  const handleVerifyOtp = async (type: 'email' | 'phone') => {
   192→    setVerifyingOtp(type);
   193→    try {
   194→      const body: Record<string, string> = {
   195→        otp: type === 'email' ? emailOtp : phoneOtp,
   196→        type: type === 'email' ? 'email_verification' : 'phone_verification',
   197→      };
   198→      if (type === 'email') body.email = email;
   199→      else body.phone = phone;
   200→
   201→      const res = await fetch('/api/auth/otp/verify', {
   202→        method: 'POST',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify(body),
   205→      });
   206→      const data = await res.json();
   207→      if (!res.ok) throw new Error(data.error || 'OTP verification failed');
   208→
   209→      if (type === 'email') setEmailOtpVerified(true);
   210→      else setPhoneOtpVerified(true);
   211→
   212→      toast.success(isBn ? 'ওটিপি যাচাই সফল!' : 'OTP verified successfully!');
   213→    } catch (err) {
   214→      toast.error(err instanceof Error ? err.message : 'OTP verification failed');
   215→    } finally {
   216→      setVerifyingOtp(null);
   217→    }
   218→  };
   219→
   220→  // Step 1 validation
   221→  const isStep1Valid = () => {
   222→    return (
   223→      fullName.trim().length >= 2 &&
   224→      username.length >= 3 &&
   225→      usernameStatus === 'available' &&
   226→      email.includes('@') &&
   227→      /^\+8801[3-9]\d{8}$/.test(phone) &&
   228→      gender &&
   229→      dateOfBirth &&
   230→      password.length >= 8 &&
   231→      /[A-Z]/.test(password) &&
   232→      /[a-z]/.test(password) &&
   233→      /\d/.test(password) &&
   234→      /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password) &&
   235→      password === confirmPassword
   236→    );
   237→  };
   238→
   239→  // Step 2 validation
   240→  const isStep2Valid = () => {
   241→    return agreeAge && agreeTerms;
   242→  };
   243→
   244→  // Step 3 validation
   245→  const isStep3Valid = () => {
   246→    return emailOtpVerified && phoneOtpVerified;
   247→  };
   248→
   249→  // Handle registration
   250→  const handleRegister = async () => {
   251→    if (!isStep3Valid()) {
   252→      setError(isBn ? 'অনুগ্রহ করে ইমেইল ও ফোন যাচাই করুন' : 'Please verify both email and phone');
   253→      return;

   255→
   256→    setLoading(true);
   257→    setError('');
   258→
   259→    try {
   260→      const res = await fetch('/api/auth/register', {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({
   264→          name: fullName.trim(),
   265→          username,
   266→          email: email.toLowerCase(),
   267→          phone,
   268→          gender,
   269→          dateOfBirth,
   270→          password,
   271→          agreeAge,
   272→          agreeTerms,
   273→          agreeNotifications,
   274→        }),
   275→      });
   276→
   277→      const data = await res.json();
   278→      if (!res.ok) throw new Error(data.error || 'Registration failed');
   279→
   280→      login(data.user, data.token);
   281→      toast.success(t('registrationSuccess', language));
   282→      navigate('home');
   283→    } catch (err) {
   284→      setError(err instanceof Error ? err.message : 'Something went wrong');
   285→    } finally {
   286→      setLoading(false);
   287→    }
   288→  };
   289→
   290→  const steps = [
   291→    { number: 1, label: t('personalInfo', language), icon: User },
   292→    { number: 2, label: t('agreement', language), icon: Shield },
   293→    { number: 3, label: t('securityVerification', language), icon: Lock },
   294→  ];
   295→
   296→  return (
   297→    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-primary/5 via-background to-primary/10 p-4">
   298→      {/* Background decoration */}
   299→      <div className="fixed inset-0 overflow-hidden pointer-events-none">
   300→        <div className="absolute -top-40 -right-40 w-80 h-80 bg-primary/10 rounded-full blur-3xl" />
   301→        <div className="absolute -bottom-40 -left-40 w-80 h-80 bg-primary/5 rounded-full blur-3xl" />
   302→      </div>
   303→
   304→      <div className="w-full max-w-lg relative">
   305→        {/* Logo */}
   306→        <motion.div
   307→          className="text-center mb-6"
   308→          initial={{ opacity: 0, y: -20 }}
   309→          animate={{ opacity: 1, y: 0 }}
   310→          transition={{ duration: 0.4 }}
   311→        >
   312→          <div className="mx-auto w-14 h-14 rounded-2xl bg-gradient-to-br from-primary to-primary/80 flex items-center justify-center mb-3 shadow-lg">
   313→            <Ticket className="w-7 h-7 text-primary-foreground" />
   314→          </div>
   315→          <h1 className={`text-2xl font-bold ${fontClass}`}>{t('appName', language)}</h1>
   316→          <p className={`text-sm text-muted-foreground mt-1 ${fontClass}`}>{t('appSlogan', language)}</p>
   317→        </motion.div>
   318→
   319→        {/* Step indicator */}
   320→        <div className="flex items-center justify-center gap-2 mb-6">
   321→          {steps.map((step, index) => (
   322→            <div key={step.number} className="flex items-center">
   323→              <div className={`flex items-center gap-2 px-3 py-1.5 rounded-full text-xs font-medium transition-all ${
   324→                currentStep >= step.number
   325→                  ? 'bg-primary text-primary-foreground shadow-md'
   326→                  : 'bg-muted text-muted-foreground'
   327→              } ${fontClass}`}>
   328→                <step.icon className="w-3.5 h-3.5" />
   329→                <span className="hidden sm:inline">{step.label}</span>
   330→                <span className="sm:hidden">{step.number}</span>
   331→              </div>
   332→              {index < steps.length - 1 && (
   333→                <div className={`w-8 h-0.5 mx-1 rounded transition-all ${
   334→                  currentStep > step.number ? 'bg-primary' : 'bg-muted'
   335→                }`} />
   336→              )}
   337→            </div>
   338→          ))}
   339→        </div>
   340→
   341→        <Card className="shadow-xl border-primary/10">
   342→          <CardHeader className="pb-4">
   343→            <CardTitle className={`text-xl text-center ${fontClass}`}>
   344→              {currentStep === 1 && t('personalInfo', language)}
   345→              {currentStep === 2 && t('agreement', language)}
   346→              {currentStep === 3 && t('securityVerification', language)}
   347→            </CardTitle>
   348→            <CardDescription className={`text-center ${fontClass}`}>
   349→              {t('step', language)} {currentStep} {t('of', language)} 3
   350→            </CardDescription>
   351→          </CardHeader>
   352→          <CardContent>
   353→            <AnimatePresence mode="wait">
   354→              {/* STEP 1: Personal Information */}
   355→              {currentStep === 1 && (
   356→                <motion.div
   357→                  key="step1"
   358→                  initial={{ opacity: 0, x: 20 }}
   359→                  animate={{ opacity: 1, x: 0 }}
   360→                  exit={{ opacity: 0, x: -20 }}
   361→                  transition={{ duration: 0.2 }}
   362→                  className="space-y-4"
   363→                >
   364→                  {/* Full Name */}
   365→                  <div className="space-y-1.5">
   366→                    <Label className={`text-sm font-medium ${fontClass}`}>
   367→                      {t('fullName', language)} <span className="text-destructive">*</span>
   368→                    </Label>
   369→                    <div className="relative">
   370→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   371→                      <Input
   372→                        value={fullName}
   373→                        onChange={(e) => setFullName(e.target.value)}
   374→                        placeholder={isBn ? 'আপনার পূর্ণ নাম' : 'Your full name'}
   375→                        className="pl-10"
   376→                      />
   377→                    </div>
   378→                  </div>
   379→
   380→                  {/* Username */}
   381→                  <div className="space-y-1.5">
   382→                    <Label className={`text-sm font-medium ${fontClass}`}>
   383→                      {t('username', language)} <span className="text-destructive">*</span>
   384→                    </Label>
   385→                    <div className="relative">
   386→                      <AtSign className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   387→                      <Input
   388→                        value={username}
   389→                        onChange={(e) => setUsername(e.target.value.toLowerCase().replace(/[^a-z0-9_]/g, ''))}
   390→                        placeholder={isBn ? 'ইউজারনেম' : 'username'}
   391→                        className="pl-10"
   392→                      />
   393→                      {usernameStatus === 'checking' && (
   394→                        <Loader2 className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 animate-spin text-muted-foreground" />

   396→                      {usernameStatus === 'available' && (
   397→                        <Check className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-emerald-500" />

   399→                      {usernameStatus === 'taken' && (
   400→                        <X className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-red-500" />
   401→                      )}
   402→                    </div>
   403→                    {usernameStatus === 'available' && (
   404→                      <p className="text-xs text-emerald-600 flex items-center gap-1"><Check className="w-3 h-3" />{t('usernameAvailable', language)}</p>
   405→                    )}
   406→                    {usernameStatus === 'taken' && (
   407→                      <p className="text-xs text-red-600 flex items-center gap-1"><X className="w-3 h-3" />{t('usernameTaken', language)}</p>
   408→                    )}
   409→                    <p className={`text-xs text-muted-foreground ${fontClass}`}>
   410→                      {isBn ? 'ছোট হাতের ইংরেজি অক্ষর, সংখ্যা ও আন্ডারস্কোর। ৩-২০ অক্ষর।' : 'Lowercase letters, numbers, underscores. 3-20 characters. Can be changed later if available.'}
   411→                    </p>
   412→                  </div>
   413→
   414→                  {/* Email */}
   415→                  <div className="space-y-1.5">
   416→                    <Label className={`text-sm font-medium ${fontClass}`}>
   417→                      {t('email', language)} <span className="text-destructive">*</span>
   418→                    </Label>
   419→                    <div className="relative">
   420→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   421→                      <Input
   422→                        type="email"
   423→                        value={email}
   424→                        onChange={(e) => setEmail(e.target.value)}
   425→                        placeholder="you@gmail.com"
   426→                        className="pl-10"
   427→                      />
   428→                    </div>
   429→                    <p className={`text-xs text-muted-foreground ${fontClass}`}>
   430→                      {isBn ? '@gmail.com, @outlook.com, @yahoo.com ইত্যাদি' : 'Supported: @gmail.com, @outlook.com, @yahoo.com, etc.'}
   431→                    </p>
   432→                  </div>
   433→
   434→                  {/* Phone */}
   435→                  <div className="space-y-1.5">
   436→                    <Label className={`text-sm font-medium ${fontClass}`}>
   437→                      {t('phone', language)} <span className="text-destructive">*</span>
   438→                    </Label>
   439→                    <div className="relative">
   440→                      <Phone className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   441→                      <Input
   442→                        type="tel"
   443→                        value={phone}
   444→                        onChange={handlePhoneChange}
   445→                        placeholder="+8801712345678"
   446→                        className="pl-10"
   447→                      />
   448→                    </div>
   449→                    <p className={`text-xs text-muted-foreground ${fontClass}`}>
   450→                      {isBn ? '+88 দিয়ে শুরু, তারপর ১১ সংখ্যার বাংলাদেশ নম্বর' : '+88 country code followed by 11-digit BD number'}
   451→                    </p>
   452→                  </div>
   453→
   454→                  {/* Gender & DOB row */}
   455→                  <div className="grid grid-cols-2 gap-3">
   456→                    <div className="space-y-1.5">
   457→                      <Label className={`text-sm font-medium ${fontClass}`}>
   458→                        {t('gender', language)} <span className="text-destructive">*</span>
   459→                      </Label>
   460→                      <Select value={gender} onValueChange={setGender}>

   463→                        </SelectTrigger>
   464→                        <SelectContent>
   465→                          <SelectItem value="male">{t('male', language)}</SelectItem>
   466→                          <SelectItem value="female">{t('female', language)}</SelectItem>
   467→                          <SelectItem value="other">{t('other', language)}</SelectItem>
   468→                        </SelectContent>
   469→                      </Select>
   470→                    </div>
   471→                    <div className="space-y-1.5">
   472→                      <Label className={`text-sm font-medium ${fontClass}`}>
   473→                        {t('dateOfBirth', language)} <span className="text-destructive">*</span>
   474→                      </Label>
   475→                      <div className="relative">
   476→                        <Calendar className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   477→                        <Input
   478→                          type="date"
   479→                          value={dateOfBirth}
   480→                          onChange={(e) => setDateOfBirth(e.target.value)}
   481→                          className="pl-10"
   482→                          max={new Date(new Date().setFullYear(new Date().getFullYear() - 18)).toISOString().split('T')[0]}
   483→                        />
   484→                      </div>
   485→                    </div>
   486→                  </div>
   487→
   488→                  {/* Password */}
   489→                  <div className="space-y-1.5">
   490→                    <Label className={`text-sm font-medium ${fontClass}`}>
   491→                      {t('password', language)} <span className="text-destructive">*</span>
   492→                    </Label>
   493→                    <div className="relative">
   494→                      <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
