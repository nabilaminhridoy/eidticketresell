   201→   201→export default function SellTicketPage() {
   202→   202→  const { navigate } = useAppStore();
   203→   203→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   204→   204→  const { language } = useLanguageStore();
   205→   205→  const isBn = language === 'bn';
   206→   206→  const fontClass = isBn ? 'font-bangla' : '';
   207→   207→
   208→   208→  const [form, setForm] = useState<FormState>(initialForm);
   209→   209→  const [uploading, setUploading] = useState(false);
   210→   210→  const [uploadFileName, setUploadFileName] = useState('');
   211→   211→  const [submitting, setSubmitting] = useState(false);
   212→   212→  const [error, setError] = useState('');
   213→   213→  const [success, setSuccess] = useState(false);
   214→   214→  const [showPreview, setShowPreview] = useState(false);
   215→   215→  const [checkingAuth, setCheckingAuth] = useState(true);
   216→   216→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   217→   217→  const [dragOver, setDragOver] = useState(false);
   218→   218→  const fileInputRef = useRef<HTMLInputElement>(null);
   219→   219→
   220→   220→  // Fetch user profile & KYC status on mount
   221→   221→  useEffect(() => {
   222→   222→    if (!isAuthenticated || !token) {
   223→   223→      setCheckingAuth(false);
   224→   224→      return;
   225→   225→    }
   226→   226→    (async () => {
   227→   227→      try {
   228→   228→        const res = await fetch('/api/auth/me', {
   229→   229→          headers: { Authorization: `Bearer ${token}` },
   230→   230→        });
   231→   231→        if (res.ok) {
   232→   232→          const data = await res.json();
   233→   233→          if (data.user) updateUser(data.user);
   234→   234→          if (data.user?.isKycVerified) {
   235→   235→            setKycStatus('approved');
   236→   236→          } else {
   237→   237→            const kycRes = await fetch('/api/kyc', {
   238→   238→              headers: { Authorization: `Bearer ${token}` },
   239→   239→            });
   240→   240→            if (kycRes.ok) {
   241→   241→              const kycData = await kycRes.json();
   242→   242→              if (kycData.application) {
   243→   243→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   244→   244→              } else {
   245→   245→                setKycStatus('none');
   246→   246→              }
   247→   247→            } else {
   248→   248→              setKycStatus('none');
   249→   249→            }
   250→   250→          }
   251→   251→        }
   252→   252→      } catch {
   253→   253→        setKycStatus('none');
   254→   254→      } finally {
   255→   255→        setCheckingAuth(false);
   256→   256→      }
   257→   257→    })();
   258→   258→  }, [isAuthenticated, token, updateUser]);
   259→   259→
   260→   260→  /* ---- derived ---- */
   261→   261→  const platformFee = useMemo(() => {
   262→   262→    const sp = parseFloat(form.sellingPrice);
   263→   263→    if (!sp || sp <= 0) return 0;
   264→   264→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   265→   265→  }, [form.sellingPrice]);
   266→   266→
   267→   267→  const sellerReceives = useMemo(() => {
   268→   268→    const sp = parseFloat(form.sellingPrice);
   269→   269→    if (!sp || sp <= 0) return 0;
   270→   270→    return sp - platformFee;
   271→   271→  }, [form.sellingPrice, platformFee]);
   272→   272→
   273→   273→  const showSeatClass = form.transportType === 'bus';
   274→   274→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   275→   275→  const isCounterCopy = form.ticketType === 'counter_copy';
   276→   276→
   277→   277→  /* ---- setters ---- */
   278→   278→  const set = useCallback(
   279→   279→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   280→   280→      setForm((prev) => {
   281→   281→        const next = { ...prev, [key]: value };
   282→   282→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   283→   283→          next.deckType = '';
   284→   284→        }
   285→   285→        if (key === 'ticketType') {
   286→   286→          next.deliveryType = '';
   287→   287→          next.meetingPlace = '';
   288→   288→          next.courierName = '';
   289→   289→          next.deliverySpeed = '';
   290→   290→          next.deliveryChargePaidBy = 'seller';
   291→   291→          next.deliveryCharge = '';
   292→   292→          next.ticketDocument = '';
   293→   293→          setUploadFileName('');
   294→   294→        }
   295→   295→        if (key === 'transportType') {
   296→   296→          next.seatClass = '';
   297→   297→          next.deckType = '';
   298→   298→        }
   299→   299→        if (key === 'deliveryType') {
   300→   300→          next.meetingPlace = '';
   301→   301→          next.courierName = '';
   302→   302→          next.deliverySpeed = '';
   303→   303→          next.deliveryCharge = '';
   304→   304→          next.deliveryChargePaidBy = 'seller';
   305→   305→        }
   306→   306→        return next;
   307→   307→      });
   308→   308→    },
   309→   309→    [],
   310→   310→  );
   311→   311→
   312→   312→  /* ---- file upload ---- */
   313→   313→  const handleFileUpload = async (file: File) => {
   314→   314→    if (form.ticketType === 'online_copy') {
   315→   315→      if (file.type !== 'application/pdf') {
   316→   316→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   317→   317→        return;
   318→   318→      }
   319→   319→    } else {
   320→   320→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   321→   321→      if (!validImageTypes.includes(file.type)) {
   322→   322→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   323→   323→        return;
   324→   324→      }
   325→   325→    }
   326→   326→
   327→   327→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   328→   328→    if (file.size > maxSize) {
   329→   329→      toast.error(
   330→   330→        isBn
   331→   331→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   332→   332→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   333→   333→      );
   334→   334→      return;
   335→   335→    }
   336→   336→
   337→   337→    if (!token) {
   338→   338→      navigate('login');
   339→   339→      return;
   340→   340→    }
   341→   341→
   342→   342→    setUploading(true);
   343→   343→    try {
   344→   344→      const formData = new FormData();
   345→   345→      formData.append('file', file);
   346→   346→
   347→   347→      const res = await fetch('/api/upload', {
   348→   348→        method: 'POST',
   349→   349→        headers: { Authorization: `Bearer ${token}` },
   350→   350→        body: formData,
   351→   351→      });
   352→   352→
   353→   353→      if (!res.ok) {
   354→   354→        const data = await res.json();
   355→   355→        throw new Error(data.error || 'Upload failed');
   356→   356→      }
   357→   357→
   358→   358→      const data = await res.json();
   359→   359→      set('ticketDocument', data.url);
   360→   360→      setUploadFileName(file.name);
   361→   361→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   362→   362→    } catch (err: unknown) {
   363→   363→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   364→   364→    } finally {
   365→   365→      setUploading(false);
   366→   366→    }
   367→   367→  };
   368→   368→
   369→   369→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   370→   370→    const file = e.target.files?.[0];
   371→   371→    if (file) handleFileUpload(file);
   372→   372→  };
   373→   373→
   374→   374→  const handleDrop = (e: React.DragEvent) => {
   375→   375→    e.preventDefault();
   376→   376→    setDragOver(false);
   377→   377→    const file = e.dataTransfer.files?.[0];
   378→   378→    if (file) handleFileUpload(file);
   379→   379→  };
   380→   380→
   381→   381→  const handleDragOver = (e: React.DragEvent) => {
   382→   382→    e.preventDefault();
   383→   383→    setDragOver(true);
   384→   384→  };
   385→   385→
   386→   386→  const handleDragLeave = () => {
   387→   387→    setDragOver(false);
   388→   388→  };
   389→   389→
   390→   390→  /* ---- validation ---- */
   391→   391→  const validate = (): string | null => {
   392→   392→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';

   394→   394→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   395→   395→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   396→   396→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   397→   397→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   398→   398→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   399→   399→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   400→   400→
   401→   401→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   402→   402→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   403→   403→
   404→   404→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   405→   405→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   406→   406→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   407→   407→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   408→   408→
   409→   409→    if (isCounterCopy) {
   410→   410→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   411→   411→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   412→   412→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   413→   413→      if (form.deliveryType === 'courier') {
   414→   414→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   415→   415→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   416→   416→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   417→   417→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   418→   418→      }
   419→   419→    }
   420→   420→
   421→   421→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   422→   422→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   423→   423→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   424→   424→
   425→   425→    return null;
   426→   426→  };
   427→   427→
   428→   428→  /* ---- submit ---- */
   429→   429→  const handleSubmit = async (e: React.FormEvent) => {
   430→   430→    e.preventDefault();
   431→   431→    setError('');
   432→   432→
   433→   433→    if (!isAuthenticated || !token) {
   434→   434→      navigate('login');
   435→   435→      return;
   436→   436→    }
   437→   437→
   438→   438→    const validationError = validate();
   439→   439→    if (validationError) {
   440→   440→      setError(validationError);
   441→   441→      toast.error(validationError);
