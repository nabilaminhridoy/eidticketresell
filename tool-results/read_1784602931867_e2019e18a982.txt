     1→'use client';

     5→import { useNav } from '@/lib/use-nav';

    16→import {

    23→

    25→

    28→  kycDob: string;
    29→  kycGender: string;
    30→  nameChanged: boolean;
    31→  dobChanged: boolean;
    32→  genderChanged: boolean;
    33→  documentType: string;
    34→  documentNumber: string;
    35→  documentFront: string;
    36→  documentBack: string;
    37→  houseRoadVillage: string;
    38→  upazilaThana: string;
    39→  district: string;
    40→  division: string;
    41→  postalCode: string;
    42→  selfiePhoto: string;
    43→  selfieRight: string;
    44→  selfieLeft: string;
    45→  selfieSmile: string;
    46→  selfieBlink: string;
    47→  gpsLatitude: number | null;
    48→  gpsLongitude: number | null;
    49→  status: string;
    50→  reviewNote: string | null;
    51→}
    52→
    53→export default function KycPage() {
    54→  const { navigate } = useNav();
    55→  const { user, token, updateUser } = useAuthStore();
    56→  const { language } = useLanguageStore();
    57→  const isBn = language === 'bn';
    58→  const fontClass = isBn ? 'font-bangla' : '';
    59→
    60→  const [currentStep, setCurrentStep] = useState<Step>(1);
    61→  const [loading, setLoading] = useState(false);
    62→  const [error, setError] = useState('');
    63→  const [existingKyc, setExistingKyc] = useState<KycData | null>(null);
    64→  const [kycStatus, setKycStatus] = useState<string | null>(null);
    65→
    66→  // Step 1: Personal Info + Document + Address
    67→  const [kycName, setKycName] = useState('');
    68→  const [kycDob, setKycDob] = useState('');
    69→  const [kycGender, setKycGender] = useState('');
    70→  const [nameLocked, setNameLocked] = useState(false);
    71→  const [dobLocked, setDobLocked] = useState(false);
    72→  const [genderLocked, setGenderLocked] = useState(false);
    73→  const [documentType, setDocumentType] = useState('');
    74→  const [documentNumber, setDocumentNumber] = useState('');
    75→  const [documentFront, setDocumentFront] = useState('');
    76→  const [documentBack, setDocumentBack] = useState('');
    77→  const [houseRoadVillage, setHouseRoadVillage] = useState('');
    78→  const [upazilaThana, setUpazilaThana] = useState('');
    79→  const [district, setDistrict] = useState('');
    80→  const [division, setDivision] = useState('');
    81→  const [postalCode, setPostalCode] = useState('');
    82→
    83→  // Step 2: Selfie + GPS
    84→  const [selfiePhoto, setSelfiePhoto] = useState('');
    85→  const [selfieRight, setSelfieRight] = useState('');
    86→  const [selfieLeft, setSelfieLeft] = useState('');
    87→  const [selfieSmile, setSelfieSmile] = useState('');
    88→  const [selfieBlink, setSelfieBlink] = useState('');
    89→  const [gpsLatitude, setGpsLatitude] = useState<number | null>(null);
    90→  const [gpsLongitude, setGpsLongitude] = useState<number | null>(null);
    91→  const [gpsStatus, setGpsStatus] = useState<'pending' | 'granted' | 'denied'>('pending');
    92→
    93→  // Camera state
    94→  const videoRef = useRef<HTMLVideoElement>(null);
    95→  const canvasRef = useRef<HTMLCanvasElement>(null);
    96→  const [cameraActive, setCameraActive] = useState(false);
    97→  const [currentPose, setCurrentPose] = useState<'front' | 'right' | 'left' | 'smile' | 'blink'>('front');
    98→  const [capturing, setCapturing] = useState(false);
    99→
   100→  // Upload states
   101→  const [uploadingFront, setUploadingFront] = useState(false);
   102→  const [uploadingBack, setUploadingBack] = useState(false);
   103→
   104→  // Load existing KYC data
   105→  useEffect(() => {
   106→    if (!token) { navigate('login'); return; }
   107→    fetch('/api/kyc', { headers: { Authorization: `Bearer ${token}` } })
   108→      .then(r => r.json())
   109→      .then(data => {
   110→        if (data.kyc) {
   111→          setExistingKyc(data.kyc);
   112→          setKycStatus(data.kyc.status);
   113→          // Pre-fill form
   114→          setKycName(data.kyc.kycName || '');
   115→          setKycDob(data.kyc.kycDob || '');
   116→          setKycGender(data.kyc.kycGender || '');
   117→          setNameLocked(data.kyc.nameChanged);
   118→          setDobLocked(data.kyc.dobChanged);
   119→          setGenderLocked(data.kyc.genderChanged);
   120→          setDocumentType(data.kyc.documentType || '');
   121→          setDocumentNumber(data.kyc.documentNumber || '');
   122→          setDocumentFront(data.kyc.documentFront || '');
   123→          setDocumentBack(data.kyc.documentBack || '');
   124→          setHouseRoadVillage(data.kyc.houseRoadVillage || '');
   125→          setUpazilaThana(data.kyc.upazilaThana || '');
   126→          setDistrict(data.kyc.district || '');
   127→          setDivision(data.kyc.division || '');
   128→          setPostalCode(data.kyc.postalCode || '');
   129→          setSelfiePhoto(data.kyc.selfiePhoto || '');
   130→          setSelfieRight(data.kyc.selfieRight || '');
   131→          setSelfieLeft(data.kyc.selfieLeft || '');
   132→          setSelfieSmile(data.kyc.selfieSmile || '');
   133→          setSelfieBlink(data.kyc.selfieBlink || '');
   134→          setGpsLatitude(data.kyc.gpsLatitude);
   135→          setGpsLongitude(data.kyc.gpsLongitude);
   136→        } else {
   137→          // Pre-fill from profile
   138→          if (data.profileName) setKycName(data.profileName);
   139→          if (data.profileDob) setKycDob(data.profileDob);
   140→          if (data.profileGender) setKycGender(data.profileGender);
   141→        }
   142→        setKycStatus(data.isKycVerified ? 'approved' : data.kyc?.status || null);
   143→      })
   144→      .catch(() => {});
   145→  }, [token, navigate]);
   146→
   147→  // Request GPS
   148→  useEffect(() => {
   149→    if (navigator.geolocation) {
   150→      navigator.geolocation.getCurrentPosition(
   151→        (pos) => {
   152→          setGpsLatitude(pos.coords.latitude);
   153→          setGpsLongitude(pos.coords.longitude);
   154→          setGpsStatus('granted');
   155→        },
   156→        () => setGpsStatus('denied'),
   157→        { enableHighAccuracy: true }
   158→      );
   159→    }
   160→  }, []);
   161→
   162→  // Get districts for selected division
   163→  const districts = division ? (BD_DISTRICTS[division] || []) : [];
   164→
   165→  // File upload handler
   166→  const uploadFile = async (file: File): Promise<string> => {
   167→    const formData = new FormData();
   168→    formData.append('file', file);
   169→    const res = await fetch('/api/upload', {
   170→      method: 'POST',
   171→      headers: { Authorization: `Bearer ${token}` },
   172→      body: formData,
   173→    });
   174→    const data = await res.json();
   175→    if (!res.ok) throw new Error(data.error || 'Upload failed');
   176→    return data.url;
   177→  };
   178→
   179→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>, side: 'front' | 'back') => {
   180→    const file = e.target.files?.[0];
   181→    if (!file) return;
   182→
   183→    if (side === 'front') setUploadingFront(true);
   184→    else setUploadingBack(true);
   185→
   186→    try {
   187→      const url = await uploadFile(file);
   188→      if (side === 'front') setDocumentFront(url);
   189→      else setDocumentBack(url);
   190→      toast.success(isBn ? 'আপলোড সফল!' : 'Upload successful!');
   191→    } catch (err) {
   192→      toast.error(err instanceof Error ? err.message : 'Upload failed');
   193→    } finally {
   194→      if (side === 'front') setUploadingFront(false);
   195→      else setUploadingBack(false);
   196→    }
   197→  };
   198→
   199→  // Camera functions
   200→  const startCamera = useCallback(async () => {
   201→    try {
   202→      const stream = await navigator.mediaDevices.getUserMedia({
   203→        video: { facingMode: 'user', width: 640, height: 480 }
   204→      });
   205→      if (videoRef.current) {
   206→        videoRef.current.srcObject = stream;
   207→        setCameraActive(true);
   208→      }
   209→    } catch {
   210→      toast.error(isBn ? 'ক্যামেরা অ্যাক্সেস ব্যর্থ' : 'Camera access failed');
   211→    }
   212→  }, [isBn]);
   213→
   214→  const stopCamera = useCallback(() => {
   215→    if (videoRef.current?.srcObject) {
   216→      const stream = videoRef.current.srcObject as MediaStream;
   217→      stream.getTracks().forEach(track => track.stop());
   218→      videoRef.current.srcObject = null;
   219→    }
   220→    setCameraActive(false);
   221→  }, []);
   222→
   223→  const capturePhoto = useCallback(async (): Promise<string> => {
   224→    if (!videoRef.current || !canvasRef.current) return '';
   225→    const canvas = canvasRef.current;
   226→    const video = videoRef.current;
   227→    canvas.width = video.videoWidth;
   228→    canvas.height = video.videoHeight;
   229→    const ctx = canvas.getContext('2d');
   230→    if (!ctx) return '';
   231→    ctx.drawImage(video, 0, 0);
   232→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   233→
   234→    // Convert to file and upload
   235→    const blob = await (await fetch(dataUrl)).blob();
   236→    const file = new File([blob], `selfie-${currentPose}.jpg`, { type: 'image/jpeg' });
   237→    return await uploadFile(file);
   238→  }, [currentPose]);
   239→
   240→  const handleCapture = async () => {
   241→    setCapturing(true);
   242→    try {
   243→      const url = await capturePhoto();
   244→      switch (currentPose) {
   245→        case 'front': setSelfiePhoto(url); break;
   246→        case 'right': setSelfieRight(url); break;
   247→        case 'left': setSelfieLeft(url); break;
   248→        case 'smile': setSelfieSmile(url); break;
   249→        case 'blink': setSelfieBlink(url); break;
   250→      }
   251→    } catch (err) {
   252→      toast.error(isBn ? 'ক্যাপচার ব্যর্থ' : 'Capture failed');
   253→    } finally {
   254→      setCapturing(false);
   255→    }
   256→  };
   257→
   258→  // Document number label based on type
   259→  const getDocNumberLabel = () => {
   260→    switch (documentType) {
   261→      case 'nid': return t('nidNumber', language);
   262→      case 'driving_licence': return t('drivingLicenceNumber', language);
   263→      case 'passport': return t('passportNumber', language);
   264→      default: return t('documentNumber', language);
   265→    }
   266→  };
   267→
   268→  // Validate NID
   269→  const isValidNid = (num: string) => {
   270→    const digits = num.replace(/\D/g, '');
   271→    return [10, 13, 17].includes(digits.length);
   272→  };
   273→
   274→  // Step 1 validation
   275→  const isStep1Valid = () => {
   276→    return (
   277→      kycName.trim().length >= 2 &&
   278→      kycDob &&
   279→      kycGender &&
   280→      documentType &&
   281→      documentNumber &&
   282→      documentFront &&
   283→      (documentType === 'passport' || documentBack) &&
   284→      houseRoadVillage &&
   285→      upazilaThana &&
   286→      district &&
   287→      division
   288→    );
   289→  };
   290→
   291→  // Step 2 validation
   292→  const isStep2Valid = () => {
   293→    return selfiePhoto && gpsStatus === 'granted';
   294→  };
   295→
   296→  // Submit KYC
   297→  const handleSubmit = async () => {
   298→    if (!isStep2Valid()) {
   299→      setError(isBn ? 'সেলফি ক্যাপচার ও জিপিএস প্রয়োজন' : 'Selfie capture and GPS are required');
   300→      return;
   301→    }
   302→    setLoading(true);
   303→    setError('');
   304→    try {
   305→      const res = await fetch('/api/kyc', {
   306→        method: 'POST',
   307→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   308→        body: JSON.stringify({
   309→          kycName, kycDob, kycGender,
   310→          documentType, documentNumber, documentFront, documentBack,
   311→          houseRoadVillage, upazilaThana, district, division, postalCode,
   312→          selfiePhoto, selfieRight, selfieLeft, selfieSmile, selfieBlink,
   313→          gpsLatitude, gpsLongitude,
   314→        }),
   315→      });
   316→      const data = await res.json();
   317→      if (!res.ok) throw new Error(data.error || 'KYC submission failed');
   318→
   319→      toast.success(t('kycSubmitSuccess', language));
   320→      setKycStatus('pending');
   321→      setExistingKyc(data.kyc);
   322→      stopCamera();
   323→    } catch (err) {
   324→      setError(err instanceof Error ? err.message : 'Something went wrong');
   325→    } finally {
   326→      setLoading(false);
   327→    }
   328→  };
   329→
   330→  // Show status page if already submitted
   331→  if (kycStatus === 'pending' && existingKyc) {
   332→    return (
   333→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-primary/5 via-background to-primary/10 p-4">
   334→        <Card className="w-full max-w-md shadow-xl border-primary/10">
   335→          <CardContent className="p-8 text-center space-y-4">
   336→            <div className="mx-auto w-16 h-16 rounded-full bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center">
   337→              <Loader2 className="w-8 h-8 text-amber-600 animate-spin" />
   338→            </div>
   339→            <h2 className={`text-xl font-bold ${fontClass}`}>{t('kycPending', language)}</h2>
   340→            <p className={`text-sm text-muted-foreground ${fontClass}`}>{t('kycPendingMessage', language)}</p>
   341→            {existingKyc.reviewNote && (
   342→              <div className="p-3 rounded-lg bg-muted text-sm">{existingKyc.reviewNote}</div>
   343→            )}
   344→            <Button variant="outline" onClick={() => navigate('home')} className={fontClass}>
   345→              {t('back', language)}
   346→            </Button>
   347→          </CardContent>
   348→        </Card>
   349→      </div>
   350→    );
   351→  }
   352→
   353→  if (kycStatus === 'approved') {
   354→    return (
   355→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-primary/5 via-background to-primary/10 p-4">
   356→        <Card className="w-full max-w-md shadow-xl border-emerald-200 dark:border-emerald-900/30">
   357→          <CardContent className="p-8 text-center space-y-4">
   358→            <div className="mx-auto w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   359→              <ShieldCheck className="w-8 h-8 text-emerald-600" />
   360→            </div>
   361→            <h2 className={`text-xl font-bold ${fontClass}`}>{t('kycApproved', language)}</h2>
   362→            <p className={`text-sm text-muted-foreground ${fontClass}`}>{t('kycApprovedMessage', language)}</p>
   363→            <div className="grid grid-cols-3 gap-3 py-4">
   364→              {[
   365→                { icon: Check, label: t('verifiedBadge', language) },
   366→                { icon: FileText, label: t('canSellTickets', language) },
   367→                { icon: MapPin, label: t('canWithdraw', language) },
   368→              ].map(item => (
   369→                <div key={item.label} className="flex flex-col items-center gap-2 p-3 rounded-xl bg-emerald-50 dark:bg-emerald-900/20">
   370→                  <item.icon className="w-5 h-5 text-emerald-600" />
   371→                  <span className={`text-xs text-emerald-700 dark:text-emerald-400 text-center ${fontClass}`}>{item.label}</span>
   372→                </div>
   373→              ))}
   374→            </div>
   375→            <Button onClick={() => navigate('home')} className="bg-gradient-to-r from-emerald-600 to-emerald-500">
   376→              {isBn ? 'হোমে যান' : 'Go Home'}
   377→            </Button>
   378→          </CardContent>
   379→        </Card>
   380→      </div>
   381→    );
   382→  }
   383→
   384→  return (
   385→    <div className="min-h-screen bg-gradient-to-br from-primary/5 via-background to-primary/10 py-4 sm:py-6 px-3 sm:px-4 overflow-x-hidden">
   386→      <div className="w-full max-w-2xl mx-auto">
   387→        {/* Header */}
   388→        <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="text-center mb-4 sm:mb-6">
   389→          <div className="mx-auto w-12 h-12 sm:w-14 sm:h-14 rounded-2xl bg-gradient-to-br from-primary to-primary/80 flex items-center justify-center mb-2 sm:mb-3 shadow-lg">
   390→            <ShieldCheck className="w-6 h-6 sm:w-7 sm:h-7 text-primary-foreground" />
   391→          </div>
   392→          <h1 className={`text-xl sm:text-2xl font-bold ${fontClass}`}>{t('kycVerification', language)}</h1>
   393→          <p className={`text-xs sm:text-sm text-muted-foreground mt-1 ${fontClass}`}>{t('becomeVerifiedSeller', language)}</p>
   394→        </motion.div>
   395→
   396→        {/* Step indicator */}
   397→        <div className="flex items-center justify-center gap-1 sm:gap-2 mb-4 sm:mb-6">
   398→          {[
   399→            { num: 1, label: isBn ? 'নথি ও ঠিকানা' : 'Document & Address', shortLabel: isBn ? 'নথি' : 'Docs', icon: FileText },
   400→            { num: 2, label: isBn ? 'সেলফি ও জিপিএস' : 'Selfie & GPS', shortLabel: isBn ? 'সেলফি' : 'Selfie', icon: Camera },
   401→          ].map((step, idx) => (
   402→            <div key={step.num} className="flex items-center">
   403→              <div className={`flex items-center gap-1 sm:gap-2 px-2.5 sm:px-4 py-1.5 sm:py-2 rounded-full text-xs font-medium transition-all ${
   404→                currentStep >= step.num
   405→                  ? 'bg-primary text-primary-foreground shadow-md'
   406→                  : 'bg-muted text-muted-foreground'
   407→              } ${fontClass}`}>
   408→                <step.icon className="w-3.5 h-3.5 shrink-0" />
   409→                <span className="hidden sm:inline">{step.label}</span>
   410→                <span className="sm:hidden">{step.shortLabel}</span>
   411→              </div>
   412→              {idx < 1 && (
   413→                <div className={`w-4 sm:w-8 h-0.5 mx-0.5 sm:mx-1 rounded transition-all ${
   414→                  currentStep > step.num ? 'bg-primary' : 'bg-muted'
   415→                }`} />
   416→              )}
   417→            </div>
   418→          ))}
   419→        </div>
   420→
   421→        <Card className="shadow-xl border-primary/10">
   422→          <CardContent className="p-3 sm:p-6">
   423→            <AnimatePresence mode="wait">
   424→              {/* STEP 1: Personal Info + Document + Address */}
   425→              {currentStep === 1 && (
   426→                <motion.div key="step1" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4 sm:space-y-6">
   427→                  {/* Personal Information */}
   428→                  <div>
   429→                    <h3 className={`text-sm sm:text-base font-semibold mb-2 sm:mb-3 flex items-center gap-2 ${fontClass}`}>
   430→                      <User className="w-4 h-4 text-primary" />
   431→                      {t('kycPersonalInfo', language)}
   432→                    </h3>
   433→                    <div className="space-y-3">
   434→                      {/* Full Name */}
   435→                      <div className="space-y-1.5">
   436→                        <Label className={`text-sm ${fontClass}`}>
   437→                          {t('fullName', language)} <span className="text-destructive">*</span>
   438→                        </Label>
   439→                        <div className="relative">
   440→                          <Input value={kycName} onChange={(e) => !nameLocked && setKycName(e.target.value)} disabled={nameLocked} className={`h-11 ${nameLocked ? 'opacity-70' : ''}`} />
   441→                          {nameLocked && <Lock className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />}
   442→                        </div>
   443→                        {nameLocked ? (
   444→                          <p className="text-xs text-amber-600 flex items-center gap-1"><Lock className="w-3 h-3" />{t('kycChangedOnce', language)}</p>
   445→                        ) : (
   446→                          <p className="text-xs text-amber-600 flex items-center gap-1"><AlertCircle className="w-3 h-3" />{t('kycNameChangeWarning', language)}</p>
   447→                        )}
   448→                      </div>
   449→
   450→                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   451→                        {/* DOB */}
   452→                        <div className="space-y-1.5">
   453→                          <Label className={`text-sm ${fontClass}`}>
   454→                            {t('dateOfBirth', language)} <span className="text-destructive">*</span>
   455→                          </Label>
   456→                          <Input type="date" value={kycDob} onChange={(e) => !dobLocked && setKycDob(e.target.value)} disabled={dobLocked} className={`h-11 ${dobLocked ? 'opacity-70' : ''}`} />
   457→                          {dobLocked && <p className="text-xs text-amber-600 flex items-center gap-1"><Lock className="w-3 h-3" />{t('kycChangedOnce', language)}</p>}
   458→                          {!dobLocked && <p className="text-xs text-amber-600">{t('kycDobChangeWarning', language)}</p>}
   459→                        </div>
   460→
   461→                        {/* Gender */}
