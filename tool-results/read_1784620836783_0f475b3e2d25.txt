   502→                      {t('login', language)}
   503→                    </Button>
   504→                    <Button
   505→                      className="w-full min-h-[44px] bg-gradient-to-r from-primary to-primary/90"
   506→                      onClick={() => handleNavigate('register')}
   507→                    >
   508→                      {t('register', language)}
   509→                    </Button>
   510→                  </div>
   511→                )}
   512→
   513→                {/* Mobile Footer Controls */}
   514→                <div className="mt-auto border-t px-5 py-4 flex items-center justify-between gap-3">
   515→                  <Button
   516→                    variant="outline"
   517→                    size="sm"
   518→                    onClick={toggleLanguage}
   519→                    className="gap-1.5 min-h-[44px]"
   520→                  >
   521→                    <Globe className="w-4 h-4 text-primary" />
   522→                    {language === 'en' ? 'বাংলা' : 'English'}
   523→                  </Button>
   524→                  <div className="flex items-center gap-1">
   525→                    {(['light', 'dark', 'system'] as const).map((mode) => (
   526→                      <Button
   527→                        key={mode}
   528→                        variant={theme === mode ? 'default' : 'ghost'}
   529→                        size="icon"
   530→                        className="h-10 w-10 min-h-[44px] min-w-[44px]"
   531→                        onClick={() => setTheme(mode)}
   532→                      >
   533→                        {mode === 'light' && <Sun className="w-4 h-4" />}
   534→                        {mode === 'dark' && <Moon className="w-4 h-4" />}
   535→                        {mode === 'system' && <Monitor className="w-4 h-4" />}
   536→                      </Button>
   537→                    ))}
   538→                  </div>
   539→                </div>
   540→              </div>
   541→            </SheetContent>
   542→          </Sheet>
   543→        </div>
   544→      </div>
   545→    </motion.header>
   546→  );
   547→}
