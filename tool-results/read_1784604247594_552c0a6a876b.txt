   777→                        )}
   778→                      </div>

   788→                <AlertCircle className="w-4 h-4 shrink-0" />

   810→                  }}
   811→                  className="flex-1 min-h-[44px] bg-gradient-to-r from-primary to-primary/90"
   812→                  disabled={!isStep1Valid()}
   813→                >
   814→                  {t('next', language)}<ArrowRight className="w-4 h-4 ml-1 sm:ml-2" />
   815→                </Button>

   817→                <Button
   818→                  onClick={handleSubmit}

   821→                >
   822→                  {loading ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <ShieldCheck className="w-4 h-4 mr-2" />}
   823→                  {t('submit', language)}
   824→                </Button>

   826→            </div>

   828→        </Card>
