   236→export default function SellTicketPage() {
   237→  const { navigate } = useNav();
   238→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   239→  const { language } = useLanguageStore();
   240→  const isBn = language === 'bn';
   241→  const fontClass = isBn ? 'font-bangla' : '';
   242→
   243→  const [form, setForm] = useState<FormState>(initialForm);
   244→  const [uploading, setUploading] = useState(false);
   245→  const [uploadFileName, setUploadFileName] = useState('');
   246→  const [submitting, setSubmitting] = useState(false);
   247→  const [error, setError] = useState('');
   248→  const [success, setSuccess] = useState(false);
   249→  const [showPreview, setShowPreview] = useState(false);
   250→  const [checkingAuth, setCheckingAuth] = useState(true);
   251→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   252→  const [dragOver, setDragOver] = useState(false);
   253→  const [pnrChecking, setPnrChecking] = useState(false);
   254→  const [pnrAvailable, setPnrAvailable] = useState<boolean | null>(null);
   255→  const [pnrError, setPnrError] = useState('');
   256→  const pnrCheckTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   257→  const fileInputRef = useRef<HTMLInputElement>(null);
   258→
   259→  /* ---- debounced PNR uniqueness check ---- */
   260→  useEffect(() => {
   261→    if (pnrCheckTimeout.current) {
   262→      clearTimeout(pnrCheckTimeout.current);
   263→      pnrCheckTimeout.current = null;
   264→    }
   265→
   266→    const pnr = form.pnrNumber.trim();
   267→    if (!pnr || pnr.length < 3) {
   268→      setPnrAvailable(null);
   269→      setPnrChecking(false);
   270→      setPnrError('');
   271→      return;
   272→    }
   273→
   274→    setPnrChecking(true);
   275→    setPnrAvailable(null);
   276→    setPnrError('');
   277→
   278→    pnrCheckTimeout.current = setTimeout(async () => {
   279→      try {
   280→        const res = await fetch(`/api/tickets/check-pnr?pnr=${encodeURIComponent(pnr)}`);
   281→        if (res.ok) {
   282→          const data = await res.json();
   283→          if (data.available) {
   284→            setPnrAvailable(true);
   285→            setPnrError('');
   286→          } else {
   287→            setPnrAvailable(false);
   288→            setPnrError(isBn ? 'এই PNR নম্বর ইতিমধ্যে ব্যবহৃত হচ্ছে' : 'This PNR number is already in use');
   289→          }
   290→        } else {
   291→          setPnrAvailable(null);
   292→        }
   293→      } catch {
   294→        setPnrAvailable(null);
   295→      } finally {
   296→        setPnrChecking(false);
   297→      }
   298→    }, 500);
   299→
   300→    return () => {
   301→      if (pnrCheckTimeout.current) {
   302→        clearTimeout(pnrCheckTimeout.current);
   303→        pnrCheckTimeout.current = null;
   304→      }
   305→    };
   306→  }, [form.pnrNumber, isBn]);
   307→
   308→  // Fetch user profile & KYC status on mount
   309→  useEffect(() => {
   310→    if (!isAuthenticated || !token) {
   311→      setCheckingAuth(false);
   312→      return;
   313→    }
   314→    (async () => {
   315→      try {
   316→        const res = await fetch('/api/auth/me', {
   317→          headers: { Authorization: `Bearer ${token}` },
   318→        });
   319→        if (res.ok) {
   320→          const data = await res.json();
   321→          if (data.user) updateUser(data.user);
   322→          if (data.user?.isKycVerified) {
   323→            setKycStatus('approved');
   324→          } else {
   325→            const kycRes = await fetch('/api/kyc', {
   326→              headers: { Authorization: `Bearer ${token}` },
   327→            });
   328→            if (kycRes.ok) {
   329→              const kycData = await kycRes.json();
   330→              if (kycData.application) {
   331→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   332→              } else {
   333→                setKycStatus('none');
   334→              }
   335→            } else {
   336→              setKycStatus('none');
   337→            }
   338→          }
   339→        }
   340→      } catch {
   341→        setKycStatus('none');
   342→      } finally {
   343→        setCheckingAuth(false);
   344→      }
   345→    })();
   346→  }, [isAuthenticated, token, updateUser]);
   347→
   348→  /* ---- derived ---- */
   349→  const platformFee = useMemo(() => {
   350→    const sp = parseFloat(form.sellingPrice);
   351→    if (!sp || sp <= 0) return 0;
   352→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   353→  }, [form.sellingPrice]);
   354→
   355→  const isCounterCopy = form.ticketType === 'counter_copy';
   356→
   357→  const sellerReceives = useMemo(() => {
   358→    const sp = parseFloat(form.sellingPrice);
   359→    if (!sp || sp <= 0) return 0;
   360→    return sp - platformFee;
   361→  }, [form.sellingPrice, platformFee]);
   362→
   363→  // Buyer Pays depends on ticket type:
   364→  // Online Copy: Buyer Pays = Selling Price
   365→  // Counter Copy: Buyer Pays = Platform Fee
   366→  const buyerPays = useMemo(() => {
   367→    const sp = parseFloat(form.sellingPrice);
   368→    if (!sp || sp <= 0) return 0;
   369→    return isCounterCopy ? platformFee : sp;
   370→  }, [form.sellingPrice, platformFee, isCounterCopy]);
   371→
   372→  // Selling price exceeds original price check
   373→  const sellingPriceExceedsOriginal = useMemo(() => {
   374→    const sp = parseFloat(form.sellingPrice);
   375→    const op = parseFloat(form.originalPrice);
   376→    if (!sp || !op || sp <= 0 || op <= 0) return false;
   377→    return sp > op;
   378→  }, [form.sellingPrice, form.originalPrice]);
   379→
   380→  const showSeatClass = form.transportType === 'bus';
   381→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   382→
   383→  /* ---- setters ---- */
   384→  const set = useCallback(
   385→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   386→      setForm((prev) => {
   387→        const next = { ...prev, [key]: value };
   388→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   389→          next.deckType = '';
   390→        }
   391→        if (key === 'ticketType') {
   392→          next.deliveryType = '';
   393→          next.meetingPlace = '';
   394→          next.courierName = '';
   395→          next.deliverySpeed = '';
   396→          next.deliveryChargePaidBy = 'seller';
   397→          next.deliveryCharge = '';
   398→          next.ticketDocument = '';
   399→          setUploadFileName('');
   400→        }
   401→        if (key === 'transportType') {
   402→          next.seatClass = '';
   403→          next.deckType = '';
   404→        }
   405→        if (key === 'deliveryType') {
   406→          next.meetingPlace = '';
   407→          next.courierName = '';
   408→          next.deliverySpeed = '';
   409→          next.deliveryCharge = '';
   410→          next.deliveryChargePaidBy = 'seller';
   411→        }
   412→        return next;
   413→      });
   414→    },
   415→    [],
   416→  );
   417→
   418→  /* ---- file upload ---- */
   419→  const handleFileUpload = async (file: File) => {
   420→    if (form.ticketType === 'online_copy') {
   421→      if (file.type !== 'application/pdf') {
   422→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   423→        return;
   424→      }
   425→    } else {
   426→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   427→      if (!validImageTypes.includes(file.type)) {
   428→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   429→        return;
   430→      }
   431→    }
   432→
   433→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   434→    if (file.size > maxSize) {
   435→      toast.error(
   436→        isBn
   437→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   438→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   439→      );
   440→      return;
   441→    }
   442→
   443→    if (!token) {
   444→      navigate('login');
   445→      return;
   446→    }
   447→
   448→    setUploading(true);
   449→    try {
   450→      const formData = new FormData();
   451→      formData.append('file', file);
   452→
   453→      const res = await fetch('/api/upload', {
   454→        method: 'POST',
   455→        headers: { Authorization: `Bearer ${token}` },
   456→        body: formData,
   457→      });
   458→
   459→      if (!res.ok) {
   460→        const data = await res.json();
   461→        throw new Error(data.error || 'Upload failed');
   462→      }
   463→
   464→      const data = await res.json();
   465→      set('ticketDocument', data.url);
   466→      setUploadFileName(file.name);
   467→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   468→    } catch (err: unknown) {
   469→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   470→    } finally {
   471→      setUploading(false);
   472→    }
   473→  };
   474→
   475→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   476→    const file = e.target.files?.[0];
   477→    if (file) handleFileUpload(file);
   478→  };
   479→
   480→  const handleDrop = (e: React.DragEvent) => {
   481→    e.preventDefault();
   482→    setDragOver(false);
   483→    const file = e.dataTransfer.files?.[0];
   484→    if (file) handleFileUpload(file);
   485→  };
   486→
   487→  const handleDragOver = (e: React.DragEvent) => {
   488→    e.preventDefault();
   489→    setDragOver(true);
   490→  };
   491→
   492→  const handleDragLeave = () => {
   493→    setDragOver(false);
   494→  };
   495→
   496→  /* ---- validation ---- */
   497→  const validate = (): string | null => {
   498→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   499→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   500→    if (pnrAvailable === false) return isBn ? 'এই PNR নম্বর ইতিমধ্যে ব্যবহৃত হচ্ছে' : 'This PNR number is already in use';
   501→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   502→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   503→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   504→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   505→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   506→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   507→
   508→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   509→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   510→
   511→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   512→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   513→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   514→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   515→    if (parseFloat(form.sellingPrice) > parseFloat(form.originalPrice))
   516→      return isBn ? 'বিক্রয় মূল্য মূল টিকেট মূল্যের বেশি হতে পারে না' : 'Selling price cannot exceed original ticket price';
   517→
   518→    if (isCounterCopy) {
   519→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   520→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   521→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   522→      if (form.deliveryType === 'courier') {
   523→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   524→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   525→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   526→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   527→      }
