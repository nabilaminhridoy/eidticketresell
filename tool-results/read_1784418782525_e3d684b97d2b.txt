   205→   205→export default function SellTicketPage() {

   211→   211→  const [form, setForm] = useState<FormState>(initialForm);

   213→   213→  const [uploadFileName, setUploadFileName] = useState('');

   215→   215→  const [error, setError] = useState('');
   216→   216→  const [success, setSuccess] = useState(false);
   217→   217→  const [showPreview, setShowPreview] = useState(false);

   219→   219→  /* ---- derived ---- */

   221→   221→    const sp = parseFloat(form.sellingPrice);
   222→   222→    if (!sp || sp <= 0) return 0;
   223→   223→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));

   237→   237→          next.deckType = '';

   239→   239→        // Reset delivery fields when ticket type changes

   242→   242→          next.meetingPlace = '';
   243→   243→          next.courierName = '';
   244→   244→          next.deliverySpeed = '';
   245→   245→          next.deliveryChargePaidBy = 'seller';
   246→   246→          next.deliveryCharge = '';
   247→   247→          next.ticketDocument = '';
   248→   248→          setUploadFileName('');
   249→   249→        }
   250→   250→        // Reset transport-specific fields when transport type changes
   251→   251→        if (key === 'transportType') {
   252→   252→          next.seatClass = '';
   253→   253→          next.deckType = '';
   254→   254→        }
   255→   255→        // Reset courier fields when delivery type changes
   256→   256→        if (key === 'deliveryType') {
   257→   257→          next.meetingPlace = '';
   258→   258→          next.courierName = '';
   259→   259→          next.deliverySpeed = '';
   260→   260→          next.deliveryCharge = '';
   261→   261→          next.deliveryChargePaidBy = 'seller';
   262→   262→        }
   263→   263→        return next;
   264→   264→      });
   265→   265→    },
   266→   266→    [],
   267→   267→  );
   268→   268→
   269→   269→  /* ---- file upload ---- */
   270→   270→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   271→   271→    const file = e.target.files?.[0];
   272→   272→    if (!file) return;
   273→   273→
   274→   274→    // Validate file type based on ticket type
   275→   275→    if (form.ticketType === 'online_copy') {
   276→   276→      if (file.type !== 'application/pdf') {
   277→   277→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');

   279→   279→      }
   280→   280→    } else {
   281→   281→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   282→   282→      if (!validImageTypes.includes(file.type)) {
   283→   283→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   284→   284→        return;
   285→   285→      }
   286→   286→    }
   287→   287→
   288→   288→    // Validate file size (10MB for PDF, 5MB for images)

   291→   291→      toast.error(
   292→   292→        isBn
   293→   293→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   294→   294→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   295→   295→      );
   296→   296→      return;
   297→   297→    }
   298→   298→
   299→   299→    if (!token) {
   300→   300→      navigate('login');
   301→   301→      return;
   302→   302→    }
   303→   303→
   304→   304→    setUploading(true);
   305→   305→    try {
   306→   306→      const formData = new FormData();
   307→   307→      formData.append('file', file);
   308→   308→
   309→   309→      const res = await fetch('/api/upload', {
   310→   310→        method: 'POST',
   311→   311→        headers: { Authorization: `Bearer ${token}` },
   312→   312→        body: formData,
   313→   313→      });
   314→   314→
   315→   315→      if (!res.ok) {
   316→   316→        const data = await res.json();
   317→   317→        throw new Error(data.error || 'Upload failed');
   318→   318→      }
   319→   319→
   320→   320→      const data = await res.json();
   321→   321→      set('ticketDocument', data.url);
   322→   322→      setUploadFileName(file.name);
   323→   323→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   324→   324→    } catch (err: unknown) {
   325→   325→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   326→   326→    } finally {
   327→   327→      setUploading(false);
   328→   328→    }
   329→   329→  };
   330→   330→
   331→   331→  /* ---- validation ---- */
   332→   332→  const validate = (): string | null => {
   333→   333→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   334→   334→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   335→   335→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   336→   336→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   337→   337→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   338→   338→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   339→   339→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   340→   340→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   341→   341→
   342→   342→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';

   344→   344→
   345→   345→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   346→   346→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';

   355→   355→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   356→   356→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';

   358→   358→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   359→   359→      }
   360→   360→    }
   361→   361→
   362→   362→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   363→   363→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   364→   364→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   365→   365→
   366→   366→    return null;
   367→   367→  };
   368→   368→
   369→   369→  /* ---- submit ---- */
   370→   370→  const handleSubmit = async (e: React.FormEvent) => {
   371→   371→    e.preventDefault();
   372→   372→    setError('');
   373→   373→
   374→   374→    if (!isAuthenticated || !token) {
   375→   375→      navigate('login');
   376→   376→      return;
   377→   377→    }
   378→   378→
   379→   379→    const validationError = validate();
   380→   380→    if (validationError) {
   381→   381→      setError(validationError);
   382→   382→      toast.error(validationError);
   383→   383→      return;
   384→   384→    }
   385→   385→
   386→   386→    setSubmitting(true);
   387→   387→    try {
   388→   388→      const body = {
   389→   389→        transportType: form.transportType,
   390→   390→        transportCompany: form.transportCompany.trim(),
   391→   391→        ticketType: form.ticketType,
   392→   392→        pnrNumber: form.pnrNumber.trim(),
   393→   393→        ticketDocument: form.ticketDocument,
   394→   394→        routeFrom: form.from,
   395→   395→        routeTo: form.to,
   396→   396→        departureDate: form.departureDate,
   397→   397→        departureTime: form.departureTime,
   398→   398→        boardingPoint: form.boardingPoint.trim() || undefined,
   399→   399→        droppingPoint: form.droppingPoint.trim() || undefined,
   400→   400→        seatClass: form.seatClass || undefined,
   401→   401→        deckType: form.deckType || undefined,
   402→   402→        seatNumber: form.seatNumber.trim() || undefined,
   403→   403→        coachNumber: form.coachNumber.trim() || undefined,
   404→   404→        originalPrice: parseFloat(form.originalPrice),
   405→   405→        price: parseFloat(form.sellingPrice),
   406→   406→        description: form.description.trim() || undefined,
   407→   407→        sellerNotes: form.sellerNotes.trim() || undefined,
   408→   408→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   409→   409→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
   410→   410→        courierName: isCounterCopy && form.deliveryType === 'courier' ? form.courierName : undefined,
   411→   411→        deliverySpeed: isCounterCopy && form.deliveryType === 'courier' ? form.deliverySpeed : undefined,
   412→   412→        deliveryChargePaidBy: isCounterCopy && form.deliveryType === 'courier' ? form.deliveryChargePaidBy : undefined,
   413→   413→        deliveryCharge:
   414→   414→          isCounterCopy && form.deliveryType === 'courier' && form.deliveryChargePaidBy === 'buyer'
   415→   415→            ? parseFloat(form.deliveryCharge)
   416→   416→            : 0,
   417→   417→        isConfirmed: true,
   418→   418→      };
   419→   419→
   420→   420→      const res = await fetch('/api/tickets', {
   421→   421→        method: 'POST',
   422→   422→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   423→   423→        body: JSON.stringify(body),
   424→   424→      });
   425→   425→
   426→   426→      if (!res.ok) {
   427→   427→        const data = await res.json();
   428→   428→        throw new Error(data.error || isBn ? 'টিকেট তৈরি ব্যর্থ' : 'Failed to create ticket');
   429→   429→      }
   430→   430→
   431→   431→      setSuccess(true);
   432→   432→      toast.success(isBn ? 'টিকেট সফলভাবে তালিকাভুক্ত হয়েছে!' : 'Ticket listed successfully!');
   433→   433→    } catch (err: unknown) {
   434→   434→      const msg = err instanceof Error ? err.message : isBn ? 'একটি ত্রুটি ঘটেছে' : 'An error occurred';
   435→   435→      setError(msg);
