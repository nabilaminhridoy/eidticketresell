   205→// ─── Animated Counter Hook ──────────────────────────────────
   206→function useAnimatedCounter(target: number, inView: boolean, duration: number = 2000) {
   207→  const [count, setCount] = useState(0);
   208→
   209→  useEffect(() => {
   210→    if (!inView) return;
   211→    let start = 0;
   212→    const startTime = Date.now();
   213→
   214→    const step = () => {
   215→      const elapsed = Date.now() - startTime;
   216→      const progress = Math.min(elapsed / duration, 1);
   217→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   218→      const current = Math.floor(eased * target);
   219→      setCount(current);
   220→      if (progress < 1) {
   221→        requestAnimationFrame(step);
   222→      } else {
   223→        setCount(target);
   224→      }
   225→    };
   226→    requestAnimationFrame(step);
   227→  }, [inView, target, duration]);
   228→
   229→  return count;
   230→}
