   213→export default function SellTicketPage() {
   214→  const { navigate } = useNav();
   215→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   216→  const { language } = useLanguageStore();
   217→  const isBn = language === 'bn';
   218→  const fontClass = isBn ? 'font-bangla' : '';
   219→
   220→  const [form, setForm] = useState<FormState>(initialForm);
   221→  const [currentStep, setCurrentStep] = useState(0);
   222→  const [uploading, setUploading] = useState(false);
   223→  const [uploadFileName, setUploadFileName] = useState('');
   224→  const [submitting, setSubmitting] = useState(false);
   225→  const [error, setError] = useState('');
   226→  const [success, setSuccess] = useState(false);
   227→  const [checkingAuth, setCheckingAuth] = useState(true);
   228→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   229→  const [dragOver, setDragOver] = useState(false);
   230→  const [pnrChecking, setPnrChecking] = useState(false);
   231→  const [pnrAvailable, setPnrAvailable] = useState<boolean | null>(null);
   232→  const [pnrError, setPnrError] = useState('');
   233→  const pnrCheckTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   234→  const fileInputRef = useRef<HTMLInputElement>(null);
   235→
   236→  /* ---- debounced PNR uniqueness check ---- */
   237→  useEffect(() => {
   238→    if (pnrCheckTimeout.current) {
   239→      clearTimeout(pnrCheckTimeout.current);
   240→      pnrCheckTimeout.current = null;
   241→    }
   242→
   243→    const pnr = form.pnrNumber.trim();
   244→    if (!pnr || pnr.length < 3) {
   245→      setPnrAvailable(null);
   246→      setPnrChecking(false);
   247→      setPnrError('');
   248→      return;
   249→    }
   250→
   251→    setPnrChecking(true);
   252→    setPnrAvailable(null);
   253→    setPnrError('');
   254→
   255→    pnrCheckTimeout.current = setTimeout(async () => {
   256→      try {
   257→        const res = await fetch(`/api/tickets/check-pnr?pnr=${encodeURIComponent(pnr)}`);
   258→        if (res.ok) {
   259→          const data = await res.json();
   260→          if (data.available) {
   261→            setPnrAvailable(true);
   262→            setPnrError('');
   263→          } else {
   264→            setPnrAvailable(false);
   265→            setPnrError(isBn ? 'এই PNR নম্বর ইতিমধ্যে ব্যবহৃত হচ্ছে' : 'This PNR number is already in use');
   266→          }
   267→        } else {
   268→          setPnrAvailable(null);
   269→        }
   270→      } catch {
   271→        setPnrAvailable(null);
   272→      } finally {
   273→        setPnrChecking(false);
   274→      }
   275→    }, 500);
   276→
   277→    return () => {
   278→      if (pnrCheckTimeout.current) {
   279→        clearTimeout(pnrCheckTimeout.current);
   280→        pnrCheckTimeout.current = null;
   281→      }
   282→    };
   283→  }, [form.pnrNumber, isBn]);
   284→
   285→  // Fetch user profile & KYC status on mount
   286→  useEffect(() => {
   287→    if (!isAuthenticated || !token) {
   288→      setCheckingAuth(false);
   289→      return;
   290→    }
   291→    (async () => {
   292→      try {
   293→        const res = await fetch('/api/auth/me', {
   294→          headers: { Authorization: `Bearer ${token}` },
   295→        });
   296→        if (res.ok) {
   297→          const data = await res.json();
   298→          if (data.user) updateUser(data.user);
   299→          if (data.user?.isKycVerified) {
   300→            setKycStatus('approved');
   301→          } else {
   302→            const kycRes = await fetch('/api/kyc', {
   303→              headers: { Authorization: `Bearer ${token}` },
   304→            });
   305→            if (kycRes.ok) {
   306→              const kycData = await kycRes.json();
   307→              if (kycData.application) {
   308→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   309→              } else {
   310→                setKycStatus('none');
   311→              }
   312→            } else {
   313→              setKycStatus('none');
   314→            }
   315→          }
   316→        }
   317→      } catch {
   318→        setKycStatus('none');
   319→      } finally {
   320→        setCheckingAuth(false);
   321→      }
   322→    })();
   323→  }, [isAuthenticated, token, updateUser]);
   324→
   325→  /* ---- derived ---- */
   326→  const platformFee = useMemo(() => {
   327→    const sp = parseFloat(form.sellingPrice);
   328→    if (!sp || sp <= 0) return 0;
   329→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   330→  }, [form.sellingPrice]);
   331→
   332→  const isCounterCopy = form.ticketType === 'counter_copy';
   333→
   334→  const sellerReceives = useMemo(() => {
   335→    const sp = parseFloat(form.sellingPrice);
   336→    if (!sp || sp <= 0) return 0;
   337→    return sp - platformFee;
   338→  }, [form.sellingPrice, platformFee]);
   339→
   340→  const buyerPays = useMemo(() => {
   341→    const sp = parseFloat(form.sellingPrice);
   342→    if (!sp || sp <= 0) return 0;
   343→    return isCounterCopy ? platformFee : sp;
   344→  }, [form.sellingPrice, platformFee, isCounterCopy]);
   345→
   346→  const sellingPriceExceedsOriginal = useMemo(() => {
   347→    const sp = parseFloat(form.sellingPrice);
   348→    const op = parseFloat(form.originalPrice);
   349→    if (!sp || !op || sp <= 0 || op <= 0) return false;
   350→    return sp > op;
   351→  }, [form.sellingPrice, form.originalPrice]);
   352→
   353→  const showSeatClass = form.transportType === 'bus';
   354→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   355→
   356→  /* ---- setters ---- */
   357→  const set = useCallback(
   358→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   359→      setForm((prev) => {
   360→        const next = { ...prev, [key]: value };
   361→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   362→          next.deckType = '';
   363→        }
   364→        if (key === 'ticketType') {
   365→          next.deliveryType = '';
   366→          next.meetingPlace = '';
   367→          next.courierName = '';
   368→          next.deliverySpeed = '';
   369→          next.deliveryChargePaidBy = 'seller';
   370→          next.deliveryCharge = '';
   371→          next.ticketDocument = '';
   372→          setUploadFileName('');
   373→        }
   374→        if (key === 'transportType') {
   375→          next.seatClass = '';
   376→          next.deckType = '';
   377→        }
   378→        if (key === 'deliveryType') {
   379→          next.meetingPlace = '';
   380→          next.courierName = '';
   381→          next.deliverySpeed = '';
   382→          next.deliveryCharge = '';
   383→          next.deliveryChargePaidBy = 'seller';
   384→        }
   385→        return next;
   386→      });
   387→    },
   388→    [],
   389→  );
   390→
   391→  /* ---- file upload ---- */
   392→  const handleFileUpload = async (file: File) => {
   393→    if (form.ticketType === 'online_copy') {
   394→      if (file.type !== 'application/pdf') {
   395→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   396→        return;
   397→      }
   398→    } else {
   399→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   400→      if (!validImageTypes.includes(file.type)) {
   401→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   402→        return;
   403→      }
   404→    }
   405→
   406→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   407→    if (file.size > maxSize) {
   408→      toast.error(
   409→        isBn
   410→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   411→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   412→      );
   413→      return;
   414→    }
   415→
   416→    if (!token) {
   417→      navigate('login');
   418→      return;
   419→    }
   420→
   421→    setUploading(true);
   422→    try {
   423→      const formData = new FormData();
   424→      formData.append('file', file);
   425→
   426→      const res = await fetch('/api/upload', {
   427→        method: 'POST',
   428→        headers: { Authorization: `Bearer ${token}` },
   429→        body: formData,
   430→      });
   431→
   432→      if (!res.ok) {
   433→        const data = await res.json();
   434→        throw new Error(data.error || 'Upload failed');
   435→      }
   436→
   437→      const data = await res.json();
   438→      set('ticketDocument', data.url);
   439→      setUploadFileName(file.name);
   440→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   441→    } catch (err: unknown) {
   442→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   443→    } finally {
   444→      setUploading(false);
   445→    }
   446→  };
   447→
   448→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   449→    const file = e.target.files?.[0];
   450→    if (file) handleFileUpload(file);
   451→  };
   452→
   453→  const handleDrop = (e: React.DragEvent) => {
   454→    e.preventDefault();
   455→    setDragOver(false);
   456→    const file = e.dataTransfer.files?.[0];
   457→    if (file) handleFileUpload(file);
   458→  };
   459→
   460→  const handleDragOver = (e: React.DragEvent) => {
   461→    e.preventDefault();
   462→    setDragOver(true);
   463→  };
   464→
   465→  const handleDragLeave = () => {
   466→    setDragOver(false);
   467→  };
   468→
   469→  /* ---- step validation ---- */
   470→  const validateStep = (step: number): string | null => {
   471→    switch (step) {
   472→      case 0: // Transport & Ticket Type
   473→        if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   474→        if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   475→        if (pnrAvailable === false) return isBn ? 'এই PNR নম্বর ইতিমধ্যে ব্যবহৃত হচ্ছে' : 'This PNR number is already in use';
   476→        if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   477→        return null;
   478→
   479→      case 1: // Route
   480→        if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   481→        if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   482→        if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   483→        if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   484→        if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   485→        return null;
   486→
   487→      case 2: // Seat/Class
   488→        if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   489→        if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   490→        return null;
   491→
   492→      case 3: // Pricing
   493→        if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   494→          return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   495→        if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   496→          return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   497→        if (parseFloat(form.sellingPrice) > parseFloat(form.originalPrice))
   498→          return isBn ? 'বিক্রয় মূল্য মূল টিকেট মূল্যের বেশি হতে পারে না' : 'Selling price cannot exceed original ticket price';
   499→        return null;
   500→
   501→      case 4: // Delivery
   502→        if (isCounterCopy) {
   503→          if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   504→          if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   505→            return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   506→          if (form.deliveryType === 'courier') {
   507→            if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   508→            if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   509→            if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   510→              return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   511→          }
   512→        }
   513→        return null;
   514→
