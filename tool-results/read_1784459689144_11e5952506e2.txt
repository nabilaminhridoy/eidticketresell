   201→export default function SellTicketPage() {
   202→  const { navigate } = useAppStore();
   203→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   204→  const { language } = useLanguageStore();
   205→  const isBn = language === 'bn';
   206→  const fontClass = isBn ? 'font-bangla' : '';
   207→
   208→  const [form, setForm] = useState<FormState>(initialForm);
   209→  const [uploading, setUploading] = useState(false);
   210→  const [uploadFileName, setUploadFileName] = useState('');
   211→  const [submitting, setSubmitting] = useState(false);
   212→  const [error, setError] = useState('');
   213→  const [success, setSuccess] = useState(false);
   214→  const [showPreview, setShowPreview] = useState(false);
   215→  const [checkingAuth, setCheckingAuth] = useState(true);
   216→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   217→  const [dragOver, setDragOver] = useState(false);
   218→  const fileInputRef = useRef<HTMLInputElement>(null);
   219→
   220→  // Fetch user profile & KYC status on mount
   221→  useEffect(() => {
   222→    if (!isAuthenticated || !token) {
   223→      setCheckingAuth(false);
   224→      return;
   225→    }
   226→    (async () => {
   227→      try {
   228→        const res = await fetch('/api/auth/me', {
   229→          headers: { Authorization: `Bearer ${token}` },
   230→        });
   231→        if (res.ok) {
   232→          const data = await res.json();
   233→          if (data.user) updateUser(data.user);
   234→          if (data.user?.isKycVerified) {
   235→            setKycStatus('approved');
   236→          } else {
   237→            const kycRes = await fetch('/api/kyc', {
   238→              headers: { Authorization: `Bearer ${token}` },
   239→            });
   240→            if (kycRes.ok) {
   241→              const kycData = await kycRes.json();
   242→              if (kycData.application) {
   243→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   244→              } else {
   245→                setKycStatus('none');
   246→              }
   247→            } else {
   248→              setKycStatus('none');
   249→            }
   250→          }
   251→        }
   252→      } catch {
   253→        setKycStatus('none');
   254→      } finally {
   255→        setCheckingAuth(false);
   256→      }
   257→    })();
   258→  }, [isAuthenticated, token, updateUser]);
   259→
   260→  /* ---- derived ---- */
   261→  const platformFee = useMemo(() => {
   262→    const sp = parseFloat(form.sellingPrice);
   263→    if (!sp || sp <= 0) return 0;
   264→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   265→  }, [form.sellingPrice]);
   266→
   267→  const sellerReceives = useMemo(() => {
   268→    const sp = parseFloat(form.sellingPrice);
   269→    if (!sp || sp <= 0) return 0;
   270→    return sp - platformFee;
   271→  }, [form.sellingPrice, platformFee]);
   272→
   273→  // Buyer Pays depends on ticket type:
   274→  // Online Copy: Buyer Pays = Selling Price
   275→  // Counter Copy: Buyer Pays = Platform Fee
   276→  const buyerPays = useMemo(() => {
   277→    const sp = parseFloat(form.sellingPrice);
   278→    if (!sp || sp <= 0) return 0;
   279→    return isCounterCopy ? platformFee : sp;
   280→  }, [form.sellingPrice, platformFee, isCounterCopy]);
   281→
   282→  // Selling price exceeds original price check
   283→  const sellingPriceExceedsOriginal = useMemo(() => {
   284→    const sp = parseFloat(form.sellingPrice);
   285→    const op = parseFloat(form.originalPrice);
   286→    if (!sp || !op || sp <= 0 || op <= 0) return false;
   287→    return sp > op;
   288→  }, [form.sellingPrice, form.originalPrice]);
   289→
   290→  const showSeatClass = form.transportType === 'bus';
   291→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   292→  const isCounterCopy = form.ticketType === 'counter_copy';
   293→
   294→  /* ---- setters ---- */
   295→  const set = useCallback(
   296→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   297→      setForm((prev) => {
   298→        const next = { ...prev, [key]: value };
   299→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   300→          next.deckType = '';
   301→        }
   302→        if (key === 'ticketType') {
   303→          next.deliveryType = '';
   304→          next.meetingPlace = '';
   305→          next.courierName = '';
   306→          next.deliverySpeed = '';
   307→          next.deliveryChargePaidBy = 'seller';
   308→          next.deliveryCharge = '';
   309→          next.ticketDocument = '';
   310→          setUploadFileName('');
   311→        }
   312→        if (key === 'transportType') {
   313→          next.seatClass = '';
   314→          next.deckType = '';
   315→        }
   316→        if (key === 'deliveryType') {
   317→          next.meetingPlace = '';
   318→          next.courierName = '';
   319→          next.deliverySpeed = '';
   320→          next.deliveryCharge = '';
   321→          next.deliveryChargePaidBy = 'seller';
   322→        }
   323→        return next;
   324→      });
   325→    },
   326→    [],
   327→  );
   328→
   329→  /* ---- file upload ---- */
   330→  const handleFileUpload = async (file: File) => {
   331→    if (form.ticketType === 'online_copy') {
   332→      if (file.type !== 'application/pdf') {
   333→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   334→        return;
   335→      }
   336→    } else {
   337→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   338→      if (!validImageTypes.includes(file.type)) {
   339→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   340→        return;
   341→      }
   342→    }
   343→
   344→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   345→    if (file.size > maxSize) {
   346→      toast.error(
   347→        isBn
   348→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   349→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   350→      );
   351→      return;
   352→    }
   353→
   354→    if (!token) {
   355→      navigate('login');
   356→      return;
   357→    }
   358→
   359→    setUploading(true);
   360→    try {
   361→      const formData = new FormData();
   362→      formData.append('file', file);
   363→
   364→      const res = await fetch('/api/upload', {
   365→        method: 'POST',
   366→        headers: { Authorization: `Bearer ${token}` },
   367→        body: formData,
   368→      });
   369→
   370→      if (!res.ok) {
   371→        const data = await res.json();
   372→        throw new Error(data.error || 'Upload failed');
   373→      }
   374→
   375→      const data = await res.json();
   376→      set('ticketDocument', data.url);
   377→      setUploadFileName(file.name);
   378→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   379→    } catch (err: unknown) {
   380→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   381→    } finally {
   382→      setUploading(false);
   383→    }
   384→  };
   385→
   386→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   387→    const file = e.target.files?.[0];
   388→    if (file) handleFileUpload(file);
   389→  };
   390→
   391→  const handleDrop = (e: React.DragEvent) => {
   392→    e.preventDefault();
   393→    setDragOver(false);
   394→    const file = e.dataTransfer.files?.[0];
   395→    if (file) handleFileUpload(file);
   396→  };
   397→
   398→  const handleDragOver = (e: React.DragEvent) => {
   399→    e.preventDefault();
   400→    setDragOver(true);
   401→  };
   402→
   403→  const handleDragLeave = () => {
   404→    setDragOver(false);
   405→  };
   406→
   407→  /* ---- validation ---- */
   408→  const validate = (): string | null => {
   409→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   410→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   411→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   412→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   413→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   414→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   415→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   416→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   417→
   418→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   419→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   420→
   421→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   422→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   423→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   424→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   425→    if (parseFloat(form.sellingPrice) > parseFloat(form.originalPrice))
   426→      return isBn ? 'বিক্রয় মূল্য মূল টিকেট মূল্যের বেশি হতে পারে না' : 'Selling price cannot exceed original ticket price';
   427→
   428→    if (isCounterCopy) {
   429→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   430→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   431→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   432→      if (form.deliveryType === 'courier') {
   433→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   434→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   435→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   436→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   437→      }
   438→    }
   439→
   440→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   441→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   442→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   443→
   444→    return null;
   445→  };
   446→
   447→  /* ---- submit ---- */
   448→  const handleSubmit = async (e: React.FormEvent) => {
   449→    e.preventDefault();
   450→    setError('');
   451→
   452→    if (!isAuthenticated || !token) {
   453→      navigate('login');
   454→      return;
   455→    }
   456→
   457→    const validationError = validate();
   458→    if (validationError) {
   459→      setError(validationError);
   460→      toast.error(validationError);
   461→      return;
   462→    }
   463→
   464→    setSubmitting(true);
   465→    try {
   466→      const body = {
   467→        transportType: form.transportType,
   468→        transportCompany: form.transportCompany.trim(),
   469→        ticketType: form.ticketType,
   470→        pnrNumber: form.pnrNumber.trim(),
   471→        ticketDocument: form.ticketDocument,
   472→        routeFrom: form.from,
   473→        routeTo: form.to,
   474→        departureDate: form.departureDate,
   475→        departureTime: form.departureTime,
   476→        boardingPoint: form.boardingPoint.trim() || undefined,
   477→        droppingPoint: form.droppingPoint.trim() || undefined,
   478→        seatClass: form.seatClass || undefined,
   479→        deckType: form.deckType || undefined,
   480→        seatNumber: form.seatNumber.trim() || undefined,
   481→        coachNumber: form.coachNumber.trim() || undefined,
   482→        originalPrice: parseFloat(form.originalPrice),
   483→        price: parseFloat(form.sellingPrice),
   484→        description: form.description.trim() || undefined,
   485→        sellerNotes: form.sellerNotes.trim() || undefined,
   486→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   487→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
   488→        courierName: isCounterCopy && form.deliveryType === 'courier' ? form.courierName : undefined,
   489→        deliverySpeed: isCounterCopy && form.deliveryType === 'courier' ? form.deliverySpeed : undefined,
   490→        deliveryChargePaidBy: isCounterCopy && form.deliveryType === 'courier' ? form.deliveryChargePaidBy : undefined,
   491→        deliveryCharge:
   492→          isCounterCopy && form.deliveryType === 'courier' && form.deliveryChargePaidBy === 'buyer'
   493→            ? parseFloat(form.deliveryCharge)
   494→            : 0,
   495→        isConfirmed: true,
   496→      };
   497→
   498→      const res = await fetch('/api/tickets', {
   499→        method: 'POST',
   500→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   501→        body: JSON.stringify(body),
   502→      });
   503→
   504→      if (!res.ok) {
   505→        const data = await res.json();
   506→        throw new Error(data.error || (isBn ? 'টিকেট তৈরি ব্যর্থ' : 'Failed to create ticket'));
   507→      }
   508→
   509→      setSuccess(true);
   510→      toast.success(isBn ? 'টিকেট সফলভাবে তালিকাভুক্ত হয়েছে!' : 'Ticket listed successfully!');
