   204→
   205→export default function SellTicketPage() {
   206→  const { navigate } = useAppStore();
   207→  const { token, isAuthenticated } = useAuthStore();
   208→  const { language } = useLanguageStore();
   209→  const isBn = language === 'bn';
   210→
   211→  const [form, setForm] = useState<FormState>(initialForm);
   212→  const [uploading, setUploading] = useState(false);
   213→  const [uploadFileName, setUploadFileName] = useState('');
   214→  const [submitting, setSubmitting] = useState(false);
   215→  const [error, setError] = useState('');
   216→  const [success, setSuccess] = useState(false);
   217→  const [showPreview, setShowPreview] = useState(false);
   218→
   219→  /* ---- derived ---- */
   220→  const platformFee = useMemo(() => {
   221→    const sp = parseFloat(form.sellingPrice);
   222→    if (!sp || sp <= 0) return 0;
   223→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   224→  }, [form.sellingPrice]);
   225→
   226→  const showSeatClass = form.transportType === 'bus';
   227→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   228→  const isCounterCopy = form.ticketType === 'counter_copy';
   229→
   230→  /* ---- setters ---- */
   231→  const set = useCallback(
   232→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   233→      setForm((prev) => {
   234→        const next = { ...prev, [key]: value };
   235→        // Reset deckType when class changes away from deck-requiring classes
   236→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   237→          next.deckType = '';
   238→        }
   239→        // Reset delivery fields when ticket type changes
   240→        if (key === 'ticketType') {
   241→          next.deliveryType = '';
   242→          next.meetingPlace = '';
   243→          next.courierName = '';
   244→          next.deliverySpeed = '';
   245→          next.deliveryChargePaidBy = 'seller';
   246→          next.deliveryCharge = '';
   247→          next.ticketDocument = '';
   248→          setUploadFileName('');
   249→        }
   250→        // Reset transport-specific fields when transport type changes
   251→        if (key === 'transportType') {
   252→          next.seatClass = '';
   253→          next.deckType = '';
   254→        }
   255→        // Reset courier fields when delivery type changes
   256→        if (key === 'deliveryType') {
   257→          next.meetingPlace = '';
   258→          next.courierName = '';
   259→          next.deliverySpeed = '';
   260→          next.deliveryCharge = '';
   261→          next.deliveryChargePaidBy = 'seller';
   262→        }
   263→        return next;
   264→      });
   265→    },
   266→    [],
   267→  );
   268→
   269→  /* ---- file upload ---- */
   270→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   271→    const file = e.target.files?.[0];
   272→    if (!file) return;
   273→
   274→    // Validate file type based on ticket type
   275→    if (form.ticketType === 'online_copy') {
   276→      if (file.type !== 'application/pdf') {
   277→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   278→        return;
   279→      }
   280→    } else {
   281→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   282→      if (!validImageTypes.includes(file.type)) {
   283→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   284→        return;
   285→      }
   286→    }
   287→
   288→    // Validate file size (10MB for PDF, 5MB for images)
   289→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   290→    if (file.size > maxSize) {
   291→      toast.error(
   292→        isBn
   293→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   294→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   295→      );
   296→      return;
   297→    }
   298→
   299→    if (!token) {
   300→      navigate('login');
   301→      return;
   302→    }
   303→
   304→    setUploading(true);
   305→    try {
   306→      const formData = new FormData();
   307→      formData.append('file', file);
   308→
   309→      const res = await fetch('/api/upload', {
   310→        method: 'POST',
   311→        headers: { Authorization: `Bearer ${token}` },
   312→        body: formData,
   313→      });
   314→
   315→      if (!res.ok) {
   316→        const data = await res.json();
   317→        throw new Error(data.error || 'Upload failed');
   318→      }
   319→
   320→      const data = await res.json();
   321→      set('ticketDocument', data.url);
   322→      setUploadFileName(file.name);
   323→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   324→    } catch (err: unknown) {
   325→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   326→    } finally {
   327→      setUploading(false);
   328→    }
   329→  };
   330→
   331→  /* ---- validation ---- */
   332→  const validate = (): string | null => {
   333→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   334→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   335→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   336→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   337→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   338→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   339→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   340→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   341→
   342→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   343→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   344→
   345→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   346→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   347→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   348→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   349→
   350→    if (isCounterCopy) {
   351→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   352→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   353→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   354→      if (form.deliveryType === 'courier') {
   355→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   356→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   357→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   358→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   359→      }
   360→    }
   361→
   362→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   363→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   364→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   365→
   366→    return null;
   367→  };
   368→
   369→  /* ---- submit ---- */
   370→  const handleSubmit = async (e: React.FormEvent) => {
   371→    e.preventDefault();
   372→    setError('');
   373→
   374→    if (!isAuthenticated || !token) {
   375→      navigate('login');
   376→      return;
   377→    }
   378→
   379→    const validationError = validate();
   380→    if (validationError) {
   381→      setError(validationError);
   382→      toast.error(validationError);
   383→      return;
   384→    }
   385→
   386→    setSubmitting(true);
   387→    try {
   388→      const body = {
   389→        transportType: form.transportType,
   390→        transportCompany: form.transportCompany.trim(),
   391→        ticketType: form.ticketType,
   392→        pnrNumber: form.pnrNumber.trim(),
   393→        ticketDocument: form.ticketDocument,
   394→        routeFrom: form.from,
   395→        routeTo: form.to,
   396→        departureDate: form.departureDate,
   397→        departureTime: form.departureTime,
   398→        boardingPoint: form.boardingPoint.trim() || undefined,
   399→        droppingPoint: form.droppingPoint.trim() || undefined,
   400→        seatClass: form.seatClass || undefined,
   401→        deckType: form.deckType || undefined,
   402→        seatNumber: form.seatNumber.trim() || undefined,
   403→        coachNumber: form.coachNumber.trim() || undefined,
   404→        originalPrice: parseFloat(form.originalPrice),
   405→        price: parseFloat(form.sellingPrice),
   406→        description: form.description.trim() || undefined,
   407→        sellerNotes: form.sellerNotes.trim() || undefined,
   408→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   409→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
   410→        courierName: isCounterCopy && form.deliveryType === 'courier' ? form.courierName : undefined,
   411→        deliverySpeed: isCounterCopy && form.deliveryType === 'courier' ? form.deliverySpeed : undefined,
   412→        deliveryChargePaidBy: isCounterCopy && form.deliveryType === 'courier' ? form.deliveryChargePaidBy : undefined,
   413→        deliveryCharge:
   414→          isCounterCopy && form.deliveryType === 'courier' && form.deliveryChargePaidBy === 'buyer'
   415→            ? parseFloat(form.deliveryCharge)
   416→            : 0,
   417→        isConfirmed: true,
   418→      };
   419→
   420→      const res = await fetch('/api/tickets', {
   421→        method: 'POST',
   422→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   423→        body: JSON.stringify(body),
   424→      });
   425→
   426→      if (!res.ok) {
   427→        const data = await res.json();
   428→        throw new Error(data.error || isBn ? 'টিকেট তৈরি ব্যর্থ' : 'Failed to create ticket');
   429→      }
   430→
   431→      setSuccess(true);
   432→      toast.success(isBn ? 'টিকেট সফলভাবে তালিকাভুক্ত হয়েছে!' : 'Ticket listed successfully!');
   433→    } catch (err: unknown) {
   434→      const msg = err instanceof Error ? err.message : isBn ? 'একটি ত্রুটি ঘটেছে' : 'An error occurred';
   435→      setError(msg);
   436→      toast.error(msg);
   437→    } finally {
   438→      setSubmitting(false);
   439→    }
   440→  };
   441→
   442→  /* ---- helpers ---- */
   443→  const getDistrictLabel = (dist: (typeof ALL_BD_DISTRICTS)[number]) =>
   444→    isBn ? dist.labelBn : dist.label;
   445→
   446→  const getTransportLabel = (id: string) => {
   447→    const tt = TRANSPORT_TYPES.find((t) => t.id === id);
   448→    return tt ? (isBn ? tt.labelBn : tt.label) : id;
   449→  };
   450→
   451→  const getClassLabel = (id: string) => {
   452→    const cls = BUS_CLASSES.find((c) => c.id === id);
   453→    return cls ? (isBn ? cls.labelBn : cls.label) : id;
   454→  };
   455→
   456→  const getDeckLabel = (id: string) => {
   457→    const dk = DECK_TYPES.find((d) => d.id === id);
   458→    return dk ? (isBn ? dk.labelBn : dk.label) : id;
   459→  };
   460→
   461→  const getCourierLabel = (id: string) => {
   462→    const cr = COURIER_COMPANIES.find((c) => c.id === id);
   463→    return cr ? (isBn ? cr.labelBn : cr.label) : id;
   464→  };
   465→
   466→  const getDeliverySpeedLabel = (id: string) => {
   467→    const ds = DELIVERY_SPEEDS.find((d) => d.id === id);
   468→    return ds ? (isBn ? ds.labelBn : ds.label) : id;
   469→  };
   470→
   471→  /* ================================================================ */
   472→  /*  SUCCESS PAGE                                                     */
   473→  /* ================================================================ */
   474→
   475→  if (success) {
   476→    return (
   477→      <div className="container mx-auto px-4 py-16 max-w-md">
   478→        <motion.div
   479→          initial={{ opacity: 0, scale: 0.9 }}
   480→          animate={{ opacity: 1, scale: 1 }}
   481→          className="text-center"
   482→        >
   483→          <div className="w-20 h-20 mx-auto rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center mb-6">
   484→            <CheckCircle2 className="w-10 h-10 text-emerald-600 dark:text-emerald-400" />
   485→          </div>
   486→          <h2 className={`text-2xl font-bold mb-3 ${isBn ? 'font-bangla' : ''}`}>
   487→            {t('success', language)}! 🎉
   488→          </h2>
   489→          <p className={`text-muted-foreground mb-8 ${isBn ? 'font-bangla' : ''}`}>
   490→            {isBn
   491→              ? 'আপনার টিকেট বিক্রির জন্য তালিকাভুক্ত হয়েছে। ক্রেতারা এখন এটি দেখতে পাবেন।'
   492→              : 'Your ticket has been listed for sale. Buyers can now see it.'}
   493→          </p>
   494→          <div className="flex flex-col sm:flex-row gap-3 justify-center">
   495→            <Button onClick={() => navigate('my-tickets')} className="bg-gradient-to-r from-primary to-primary/90">
   496→              <Ticket className="w-4 h-4 mr-2" />
   497→              {t('myTickets', language)}
   498→            </Button>
   499→            <Button variant="outline" onClick={() => navigate('search')}>
   500→              {t('searchTickets', language)}
   501→            </Button>
   502→          </div>
   503→        </motion.div>
   504→      </div>
   505→    );
   506→  }
   507→
   508→  /* ================================================================ */
   509→  /*  PREVIEW CARD                                                     */
   510→  /* ================================================================ */
   511→
