   588→                      { mode: 'dark' as const, icon: Moon, color: 'text-blue hover:bg-blue/10' },
   589→                      { mode: 'system' as const, icon: Monitor, color: 'text-primary hover:bg-primary/10' }
   590→                    ]).map(({ mode, icon: Icon, color }) => (
   591→                      <Button
   592→                        key={mode}
   593→                        variant={theme === mode ? 'default' : 'ghost'}
   594→                        size="icon"
   595→                        className={`h-10 w-10 min-h-[44px] min-w-[44px] ${theme !== mode ? color : ''}`}
   596→                        onClick={() => setTheme(mode)}
   597→                      >
   598→                        <Icon className="w-4 h-4" />
   599→                      </Button>
   600→                    ))}
   601→                  </div>
   602→                </div>
   603→              </div>
   604→            </SheetContent>
   605→          </Sheet>
   606→        </div>
   607→      </div>
   608→
   609→      {/* Gradient divider line under header when scrolled */}
   610→      {scrolled && <div className="divider-gradient" />}
   611→    </motion.header>
   612→  );
   613→}
