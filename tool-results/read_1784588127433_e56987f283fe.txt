   493→                    <MobileNavItem
   494→                      icon={<Settings className="w-5 h-5" />}
   495→                      label={t('settings', language)}
   496→                      active={isActive('settings')}
   497→                      onClick={() => handleNavigate('settings')}
   498→                      lang={language}
   499→                    />
   500→                    <Separator className="my-1" />
   501→                    <MobileNavItem
   502→                      icon={<LogOut className="w-5 h-5" />}
   503→                      label={t('logout', language)}
   504→                      active={false}
   505→                      onClick={() => {
   506→                        logout();
   507→                        handleNavigate('home');
   508→                      }}
   509→                      lang={language}
   510→                      destructive
   511→                    />
   512→                  </nav>
   513→                ) : (
   514→                  <div className="px-5 py-4 flex flex-col gap-3">
   515→                    <Button
   516→                      variant="outline"
   517→                      className="w-full min-h-[44px]"
   518→                      onClick={() => handleNavigate('login')}
   519→                    >
   520→                      {t('login', language)}
   521→                    </Button>
   522→                    <Button
   523→                      className="w-full min-h-[44px] bg-gradient-to-r from-primary to-primary/90"
   524→                      onClick={() => handleNavigate('register')}
   525→                    >
   526→                      {t('register', language)}
   527→                    </Button>
   528→                  </div>
   529→                )}
   530→
   531→                {/* Mobile Footer Controls */}
   532→                <div className="mt-auto border-t px-5 py-4 flex items-center justify-between gap-3">
   533→                  <Button
   534→                    variant="outline"
   535→                    size="sm"
   536→                    onClick={switchLanguage}
   537→                    className="gap-1.5 min-h-[44px]"
   538→                  >
   539→                    <Globe className="w-4 h-4 text-primary" />
   540→                    {language === 'en' ? 'বাংলা' : 'English'}
   541→                  </Button>
   542→                  <div className="flex items-center gap-1">
   543→                    {(['light', 'dark', 'system'] as const).map((mode) => (
   544→                      <Button
   545→                        key={mode}
   546→                        variant={theme === mode ? 'default' : 'ghost'}
   547→                        size="icon"
   548→                        className="h-10 w-10 min-h-[44px] min-w-[44px]"
   549→                        onClick={() => setTheme(mode)}
   550→                      >
   551→                        {mode === 'light' && <Sun className="w-4 h-4" />}
   552→                        {mode === 'dark' && <Moon className="w-4 h-4" />}
   553→                        {mode === 'system' && <Monitor className="w-4 h-4" />}
   554→                      </Button>
   555→                    ))}
   556→                  </div>
   557→                </div>
   558→              </div>
   559→            </SheetContent>
   560→          </Sheet>
   561→        </div>
   562→      </div>
   563→    </motion.header>
   564→  );
   565→}

   567→// Desktop Nav Button
   568→function NavButton({
   569→  active,
   570→  onClick,
   571→  label,
   572→  lang,
   573→}: {
   574→  active: boolean;
   575→  onClick: () => void;
   576→  label: string;
   577→  lang: Language;
   578→}) {
   579→  return (
   580→    <motion.button
   581→      onClick={onClick}
   582→      className={`relative px-3 py-2 rounded-md text-sm font-medium transition-colors ${
   583→        active
   584→          ? 'text-primary bg-primary/10'
   585→          : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'
   586→      } ${lang === 'bn' ? 'font-bangla' : ''}`}
   587→      whileHover={{ scale: 1.02 }}
   588→      whileTap={{ scale: 0.98 }}
   589→    >
   590→      {label}
   591→      {active && (
   592→        <motion.div
   593→          layoutId="activeNav"
   594→          className="absolute bottom-0 left-1/2 -translate-x-1/2 w-4/5 h-0.5 bg-primary rounded-full"
   595→          transition={{ type: 'spring', stiffness: 380, damping: 30 }}
   596→        />
   597→      )}
   598→    </motion.button>
   599→  );
   600→}
