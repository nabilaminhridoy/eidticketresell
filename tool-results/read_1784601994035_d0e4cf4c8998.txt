   603→// Desktop Nav Button
   604→function NavButton({
   605→  active,
   606→  onClick,
   607→  label,
   608→  lang,
   609→}: {
   610→  active: boolean;
   611→  onClick: () => void;
   612→  label: string;
   613→  lang: Language;
   614→}) {

   616→    <button
   617→      onClick={onClick}
   618→      className={`relative px-3 py-2 rounded-md text-sm font-medium transition-all hover:scale-[1.02] active:scale-[0.98] ${
   619→        active
   620→          ? 'text-primary bg-primary/10'
   621→          : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'
   622→      } ${lang === 'bn' ? 'font-bangla' : ''}`}
   623→    >
   624→      {label}
   625→      {active && (
   626→        <span className="absolute bottom-0 left-1/2 -translate-x-1/2 w-4/5 h-0.5 bg-primary rounded-full" />
   627→      )}
   628→    </button>
   629→  );
