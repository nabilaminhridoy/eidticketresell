   455→                    lang={language}
   456→                  />
   457→                </nav>
   458→
   459→                {/* ─── Bottom Section: Auth or User Menu ──────────── */}
   460→                <Separator />
   461→                {isAuthenticated && user ? (
   462→                  <div className="px-4 py-3">
   463→                    {/* User Info */}
   464→                    <div className="flex items-center gap-3 mb-3">
   465→                      <Avatar className="h-10 w-10 ring-2 ring-primary/20">
   466→                        <AvatarImage src={user.avatar} alt={user.name} />
   467→                        <AvatarFallback className="bg-primary text-primary-foreground font-semibold">
   468→                          {getInitials(user.name)}
   469→                        </AvatarFallback>
   470→                      </Avatar>
   471→                      <div className="flex-1 min-w-0">
   472→                        <p className="text-sm font-medium truncate">{user.name}</p>
   473→                        <p className="text-xs text-muted-foreground truncate">{user.email}</p>
   474→                      </div>
   475→                      {user.isKycVerified && (
   476→                        <Badge variant="secondary" className="gap-1 shrink-0 bg-primary/10 text-primary">
   477→                          <Shield className="w-3 h-3" />
   478→                          {t('verified', language)}
   479→                        </Badge>
   480→                      )}
   481→                    </div>
   482→                    {/* User Menu Items */}
   483→                    <div className="space-y-1">
   484→                      {userMenuItems.map((menuItem) => (
   485→                        <MobileNavItem
   486→                          key={menuItem.id}
   487→                          icon={<menuItem.icon className="w-5 h-5" />}
   488→                          label={t(menuItem.labelKey, language)}
   489→                          active={isActive(menuItem.id)}
   490→                          onClick={() => handleNavigate(menuItem.id, userParams)}
   491→                          lang={language}
   492→                        />
   493→                      ))}
   494→                      <Separator className="my-2" />
   495→                      <MobileNavItem
   496→                        icon={<LogOut className="w-5 h-5" />}
   497→                        label={t('logout', language)}
   498→                        active={false}
   499→                        onClick={() => {
   500→                          logout();
   501→                          handleNavigate('home');
   502→                        }}
   503→                        lang={language}
   504→                        destructive
   505→                      />
   506→                    </div>
   507→                  </div>
   508→                ) : (
   509→                  <div className="px-5 py-4 flex gap-3">
   510→                    <Button
   511→                      variant="outline"
   512→                      className="w-full min-h-[44px] border-primary/30 hover:bg-primary/10 hover:text-primary hover:border-primary"
   513→                      onClick={() => handleNavigate('login')}
   514→                    >
   515→                      {t('login', language)}
   516→                    </Button>
   517→                    <Button
   518→                      className="w-full min-h-[44px] rounded-lg"
   519→                      onClick={() => handleNavigate('register')}
   520→                    >
   521→                      {t('register', language)}
   522→                    </Button>
   523→                  </div>
   524→                )}
   525→              </div>
   526→            </SheetContent>
   527→          </Sheet>
   528→        </div>
   529→      </div>
   530→
   531→      {/* Accent line under header when scrolled */}
   532→      {scrolled && <div className="h-0.5 bg-primary" />}
   533→    </motion.header>
   534→  );
   535→}

   537→// ── Desktop Nav Button ────────────────────────────────────────────────────
   538→function NavButton({
   539→  active,
   540→  onClick,
   541→  label,
   542→  lang,
   543→}: {
   544→  active: boolean;
   545→  onClick: () => void;
   546→  label: string;
   547→  lang: Language;
   548→}) {
   549→  return (
   550→    <motion.button
   551→      onClick={onClick}
   552→      className={`relative px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 ${
   553→        active
   554→          ? 'text-primary bg-primary/10'
   555→          : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'
   556→      } ${lang === 'bn' ? 'font-bangla' : ''}`}
   557→      whileHover={{ scale: 1.02 }}
   558→      whileTap={{ scale: 0.98 }}
   559→    >
   560→      {label}
   561→      {active && (
   562→        <motion.div
   563→          layoutId="activeNav"
   564→          className="absolute bottom-0 left-1/2 -translate-x-1/2 w-4/5 h-0.5 bg-primary rounded-full"
   565→          transition={{ type: 'spring', stiffness: 380, damping: 30 }}
   566→        />
   567→      )}
   568→    </motion.button>
   569→  );
   570→}
