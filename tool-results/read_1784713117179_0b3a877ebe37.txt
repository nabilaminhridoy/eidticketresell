   617→// Page Router
   618→function PageRouter() {
   619→  const { currentPage } = useAppStore();
   620→
   621→  // Home page renders inline (no dynamic import)
   622→  if (currentPage === 'home') {
   623→    return (
   624→      <AppShell>
   625→        <motion.div key="home" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   626→          <HomePage />
   627→        </motion.div>
   628→      </AppShell>
   629→    );
   630→  }
   631→
   632→  // Auth pages (no shell)
   633→  if (authPages.has(currentPage)) {
   634→    return (
   635→      <Suspense fallback={<PageLoader />}>
   636→        <motion.div key={currentPage} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   637→          <LazyPage pageName={currentPage} />
   638→        </motion.div>
   639→      </Suspense>
   640→    );
   641→  }
   642→
   643→  // All other pages (with shell)
   644→  if (pageComponents[currentPage]) {
   645→    return (
   646→      <AppShell>
   647→        <Suspense fallback={<PageLoader />}>
   648→          <motion.div key={currentPage} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   649→            <LazyPage pageName={currentPage} />
   650→          </motion.div>
   651→        </Suspense>
   652→      </AppShell>
   653→    );
   654→  }
   655→
   656→  // Fallback to home
   657→  return (
   658→    <AppShell>
   659→      <HomePage />
   660→    </AppShell>
   661→  );
   662→}
   663→
   664→export default function Home() {
   665→  return <PageRouter />;
   666→}
