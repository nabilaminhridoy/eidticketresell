   498→                ) : (
   499→                  <div className="px-5 py-4 flex flex-col gap-3">
   500→                    <Button
   501→                      variant="outline"
   502→                      className="w-full min-h-[44px]"
   503→                      onClick={() => handleNavigate('login')}
   504→                    >
   505→                      {t('login', language)}
   506→                    </Button>
   507→                    <Button
   508→                      className="w-full min-h-[44px] bg-gradient-to-r from-primary to-primary/90"
   509→                      onClick={() => handleNavigate('register')}
   510→                    >
   511→                      {t('register', language)}
   512→                    </Button>
   513→                  </div>
   514→                )}
   515→
   516→                {/* Mobile Footer Controls */}
   517→                <div className="mt-auto border-t px-5 py-4 flex items-center justify-between gap-3">
   518→                  <Button
   519→                    variant="outline"
   520→                    size="sm"
   521→                    onClick={switchLanguage}
   522→                    className="gap-1.5 min-h-[44px]"
   523→                  >
   524→                    <Globe className="w-4 h-4 text-primary" />
   525→                    {language === 'en' ? 'বাংলা' : 'English'}
   526→                  </Button>
   527→                  <div className="flex items-center gap-1">
   528→                    {(['light', 'dark', 'system'] as const).map((mode) => (
   529→                      <Button
   530→                        key={mode}
   531→                        variant={theme === mode ? 'default' : 'ghost'}
   532→                        size="icon"
   533→                        className="h-10 w-10 min-h-[44px] min-w-[44px]"
   534→                        onClick={() => setTheme(mode)}
   535→                      >
   536→                        {mode === 'light' && <Sun className="w-4 h-4" />}
   537→                        {mode === 'dark' && <Moon className="w-4 h-4" />}
   538→                        {mode === 'system' && <Monitor className="w-4 h-4" />}
   539→                      </Button>
   540→                    ))}
   541→                  </div>
   542→                </div>
   543→              </div>
   544→            </SheetContent>
   545→          </Sheet>
   546→        </div>
   547→      </div>
   548→    </header>
   549→  );
   550→}
