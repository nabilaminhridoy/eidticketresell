   236→   236→export default function SellTicketPage() {
   237→   237→  const { navigate } = useNav();
   238→   238→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   239→   239→  const { language } = useLanguageStore();
   240→   240→  const isBn = language === 'bn';
   241→   241→  const fontClass = isBn ? 'font-bangla' : '';
   242→   242→
   243→   243→  const [form, setForm] = useState<FormState>(initialForm);
   244→   244→  const [uploading, setUploading] = useState(false);
   245→   245→  const [uploadFileName, setUploadFileName] = useState('');
   246→   246→  const [submitting, setSubmitting] = useState(false);
   247→   247→  const [error, setError] = useState('');
   248→   248→  const [success, setSuccess] = useState(false);
   249→   249→  const [showPreview, setShowPreview] = useState(false);
   250→   250→  const [checkingAuth, setCheckingAuth] = useState(true);
   251→   251→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   252→   252→  const [dragOver, setDragOver] = useState(false);
   253→   253→  const [pnrChecking, setPnrChecking] = useState(false);
   254→   254→  const [pnrAvailable, setPnrAvailable] = useState<boolean | null>(null);
   255→   255→  const [pnrError, setPnrError] = useState('');
   256→   256→  const pnrCheckTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
   257→   257→  const fileInputRef = useRef<HTMLInputElement>(null);
   258→   258→
   259→   259→  /* ---- debounced PNR uniqueness check ---- */
   260→   260→  useEffect(() => {
   261→   261→    if (pnrCheckTimeout.current) {
   262→   262→      clearTimeout(pnrCheckTimeout.current);
   263→   263→      pnrCheckTimeout.current = null;
   264→   264→    }
   265→   265→
   266→   266→    const pnr = form.pnrNumber.trim();
   267→   267→    if (!pnr || pnr.length < 3) {
   268→   268→      setPnrAvailable(null);
   269→   269→      setPnrChecking(false);
   270→   270→      setPnrError('');
   271→   271→      return;
   272→   272→    }
   273→   273→
   274→   274→    setPnrChecking(true);
   275→   275→    setPnrAvailable(null);
   276→   276→    setPnrError('');
   277→   277→
   278→   278→    pnrCheckTimeout.current = setTimeout(async () => {
   279→   279→      try {
   280→   280→        const res = await fetch(`/api/tickets/check-pnr?pnr=${encodeURIComponent(pnr)}`);
   281→   281→        if (res.ok) {
   282→   282→          const data = await res.json();
   283→   283→          if (data.available) {
   284→   284→            setPnrAvailable(true);
   285→   285→            setPnrError('');
   286→   286→          } else {
   287→   287→            setPnrAvailable(false);
   288→   288→            setPnrError(isBn ? 'এই PNR নম্বর ইতিমধ্যে ব্যবহৃত হচ্ছে' : 'This PNR number is already in use');
   289→   289→          }
   290→   290→        } else {
   291→   291→          setPnrAvailable(null);
   292→   292→        }
   293→   293→      } catch {
   294→   294→        setPnrAvailable(null);
   295→   295→      } finally {
   296→   296→        setPnrChecking(false);
   297→   297→      }
   298→   298→    }, 500);
   299→   299→
   300→   300→    return () => {
   301→   301→      if (pnrCheckTimeout.current) {
   302→   302→        clearTimeout(pnrCheckTimeout.current);
   303→   303→        pnrCheckTimeout.current = null;
   304→   304→      }
   305→   305→    };
   306→   306→  }, [form.pnrNumber, isBn]);
   307→   307→
   308→   308→  // Fetch user profile & KYC status on mount
   309→   309→  useEffect(() => {
   310→   310→    if (!isAuthenticated || !token) {
   311→   311→      setCheckingAuth(false);
   312→   312→      return;
   313→   313→    }
   314→   314→    (async () => {
   315→   315→      try {
   316→   316→        const res = await fetch('/api/auth/me', {
   317→   317→          headers: { Authorization: `Bearer ${token}` },
   318→   318→        });
   319→   319→        if (res.ok) {
   320→   320→          const data = await res.json();
   321→   321→          if (data.user) updateUser(data.user);
   322→   322→          if (data.user?.isKycVerified) {
   323→   323→            setKycStatus('approved');
   324→   324→          } else {
   325→   325→            const kycRes = await fetch('/api/kyc', {
   326→   326→              headers: { Authorization: `Bearer ${token}` },
   327→   327→            });
   328→   328→            if (kycRes.ok) {
   329→   329→              const kycData = await kycRes.json();
   330→   330→              if (kycData.application) {
   331→   331→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   332→   332→              } else {
   333→   333→                setKycStatus('none');
   334→   334→              }
   335→   335→            } else {
   336→   336→              setKycStatus('none');
   337→   337→            }
   338→   338→          }
   339→   339→        }
   340→   340→      } catch {
   341→   341→        setKycStatus('none');
   342→   342→      } finally {
   343→   343→        setCheckingAuth(false);
   344→   344→      }
   345→   345→    })();
   346→   346→  }, [isAuthenticated, token, updateUser]);
   347→   347→
   348→   348→  /* ---- derived ---- */
   349→   349→  const platformFee = useMemo(() => {
   350→   350→    const sp = parseFloat(form.sellingPrice);
   351→   351→    if (!sp || sp <= 0) return 0;
   352→   352→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   353→   353→  }, [form.sellingPrice]);
   354→   354→
   355→   355→  const isCounterCopy = form.ticketType === 'counter_copy';
   356→   356→
   357→   357→  const sellerReceives = useMemo(() => {
   358→   358→    const sp = parseFloat(form.sellingPrice);
   359→   359→    if (!sp || sp <= 0) return 0;
   360→   360→    return sp - platformFee;
   361→   361→  }, [form.sellingPrice, platformFee]);
   362→   362→
   363→   363→  // Buyer Pays depends on ticket type:
   364→   364→  // Online Copy: Buyer Pays = Selling Price
   365→   365→  // Counter Copy: Buyer Pays = Platform Fee
   366→   366→  const buyerPays = useMemo(() => {
   367→   367→    const sp = parseFloat(form.sellingPrice);
   368→   368→    if (!sp || sp <= 0) return 0;
   369→   369→    return isCounterCopy ? platformFee : sp;
   370→   370→  }, [form.sellingPrice, platformFee, isCounterCopy]);
   371→   371→
   372→   372→  // Selling price exceeds original price check
   373→   373→  const sellingPriceExceedsOriginal = useMemo(() => {
   374→   374→    const sp = parseFloat(form.sellingPrice);
   375→   375→    const op = parseFloat(form.originalPrice);
   376→   376→    if (!sp || !op || sp <= 0 || op <= 0) return false;
   377→   377→    return sp > op;
   378→   378→  }, [form.sellingPrice, form.originalPrice]);
   379→   379→
   380→   380→  const showSeatClass = form.transportType === 'bus';
   381→   381→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   382→   382→
   383→   383→  /* ---- setters ---- */
   384→   384→  const set = useCallback(
   385→   385→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   386→   386→      setForm((prev) => {
   387→   387→        const next = { ...prev, [key]: value };
   388→   388→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   389→   389→          next.deckType = '';
   390→   390→        }
   391→   391→        if (key === 'ticketType') {
   392→   392→          next.deliveryType = '';
   393→   393→          next.meetingPlace = '';
   394→   394→          next.courierName = '';
   395→   395→          next.deliverySpeed = '';
   396→   396→          next.deliveryChargePaidBy = 'seller';
   397→   397→          next.deliveryCharge = '';
   398→   398→          next.ticketDocument = '';
   399→   399→          setUploadFileName('');
   400→   400→        }
   401→   401→        if (key === 'transportType') {
   402→   402→          next.seatClass = '';
   403→   403→          next.deckType = '';
   404→   404→        }
   405→   405→        if (key === 'deliveryType') {
   406→   406→          next.meetingPlace = '';
   407→   407→          next.courierName = '';
   408→   408→          next.deliverySpeed = '';
   409→   409→          next.deliveryCharge = '';
   410→   410→          next.deliveryChargePaidBy = 'seller';
   411→   411→        }
   412→   412→        return next;
   413→   413→      });
   414→   414→    },
   415→   415→    [],
   416→   416→  );
   417→   417→
   418→   418→  /* ---- file upload ---- */
   419→   419→  const handleFileUpload = async (file: File) => {
   420→   420→    if (form.ticketType === 'online_copy') {
   421→   421→      if (file.type !== 'application/pdf') {
   422→   422→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   423→   423→        return;
   424→   424→      }
   425→   425→    } else {
   426→   426→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   427→   427→      if (!validImageTypes.includes(file.type)) {
   428→   428→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   429→   429→        return;
   430→   430→      }
   431→   431→    }
   432→   432→
   433→   433→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   434→   434→    if (file.size > maxSize) {
   435→   435→      toast.error(
   436→   436→        isBn
   437→   437→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   438→   438→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   439→   439→      );
   440→   440→      return;
   441→   441→    }
   442→   442→
   443→   443→    if (!token) {
   444→   444→      navigate('login');
   445→   445→      return;
   446→   446→    }
   447→   447→
   448→   448→    setUploading(true);
   449→   449→    try {
   450→   450→      const formData = new FormData();
   451→   451→      formData.append('file', file);
   452→   452→
   453→   453→      const res = await fetch('/api/upload', {
   454→   454→        method: 'POST',
   455→   455→        headers: { Authorization: `Bearer ${token}` },
   456→   456→        body: formData,
   457→   457→      });
   458→   458→
   459→   459→      if (!res.ok) {
   460→   460→        const data = await res.json();
   461→   461→        throw new Error(data.error || 'Upload failed');
   462→   462→      }
   463→   463→
   464→   464→      const data = await res.json();
   465→   465→      set('ticketDocument', data.url);
   466→   466→      setUploadFileName(file.name);
   467→   467→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   468→   468→    } catch (err: unknown) {
   469→   469→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   470→   470→    } finally {
   471→   471→      setUploading(false);
   472→   472→    }
   473→   473→  };
   474→   474→
   475→   475→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   476→   476→    const file = e.target.files?.[0];
   477→   477→    if (file) handleFileUpload(file);
   478→   478→  };
   479→   479→
   480→   480→  const handleDrop = (e: React.DragEvent) => {
   481→   481→    e.preventDefault();
   482→   482→    setDragOver(false);
   483→   483→    const file = e.dataTransfer.files?.[0];
   484→   484→    if (file) handleFileUpload(file);
   485→   485→  };
   486→   486→
