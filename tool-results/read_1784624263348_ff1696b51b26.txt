   520→// Page Router
   521→function PageRouter() {
   522→  const { currentPage } = useAppStore();
   523→
   524→  // Home page renders inline (no dynamic import)
   525→  if (currentPage === 'home') {
   526→    return (
   527→      <AppShell>
   528→        <motion.div key="home" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   529→          <HomePage />
   530→        </motion.div>
   531→      </AppShell>
   532→    );
   533→  }
   534→
   535→  // Auth pages (no shell)
   536→  if (authPages.has(currentPage)) {
   537→    return (
   538→      <Suspense fallback={<PageLoader />}>
   539→        <motion.div key={currentPage} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   540→          <LazyPage pageName={currentPage} />
   541→        </motion.div>
   542→      </Suspense>
   543→    );
   544→  }
   545→
   546→  // All other pages (with shell)
   547→  if (pageComponents[currentPage]) {
   548→    return (
   549→      <AppShell>
   550→        <Suspense fallback={<PageLoader />}>
   551→          <motion.div key={currentPage} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   552→            <LazyPage pageName={currentPage} />
   553→          </motion.div>
   554→        </Suspense>
   555→      </AppShell>
   556→    );
   557→  }
   558→
   559→  // Fallback to home
   560→  return (
   561→    <AppShell>
   562→      <HomePage />
   563→    </AppShell>
   564→  );
   565→}
   566→
   567→export default function Home() {
   568→  return <PageRouter />;
   569→}
