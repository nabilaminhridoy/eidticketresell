   485→                      onClick={() => handleNavigate('profile')}
   486→                      lang={language}
   487→                    />
   488→                    <MobileNavItem

   490→                      label={t('wallet', language)}
   491→                      active={isActive('wallet')}
   492→                      onClick={() => handleNavigate('wallet')}
   493→                      lang={language}
   494→                      iconBg="bg-orange text-orange-foreground"
   495→                    />
   496→                    <MobileNavItem
   497→                      icon={<Ticket className="w-5 h-5" />}
   498→                      label={t('myTickets', language)}
   499→                      active={isActive('my-tickets')}
   500→                      onClick={() => handleNavigate('my-tickets')}
   501→                      lang={language}
   502→                      iconBg="bg-primary text-primary-foreground"
   503→                    />
   504→                    <MobileNavItem
   505→                      icon={<LayoutDashboard className="w-5 h-5" />}
   506→                      label={t('myOrders', language)}
   507→                      active={isActive('my-orders')}
   508→                      onClick={() => handleNavigate('my-orders')}
   509→                      lang={language}
   510→                      iconBg="bg-blue text-blue-foreground"
   511→                    />
   512→                    <MobileNavItem
   513→                      icon={<Bell className="w-5 h-5" />}
   514→                      label={t('notifications', language)}
   515→                      active={isActive('notifications')}
   516→                      onClick={() => handleNavigate('notifications')}
   517→                      lang={language}
   518→                      iconBg="bg-orange text-orange-foreground"
   519→                    />
   520→                    {!user.isKycVerified && (
   521→                      <MobileNavItem
   522→                        icon={<Shield className="w-5 h-5" />}
   523→                        label={t('kycVerification', language)}
   524→                        active={isActive('kyc')}
   525→                        onClick={() => handleNavigate('kyc')}
   526→                        lang={language}
   527→                      />
   528→                    )}
   529→                    <MobileNavItem
   530→                      icon={<Settings className="w-5 h-5" />}
   531→                      label={t('settings', language)}
   532→                      active={isActive('settings')}
   533→                      onClick={() => handleNavigate('settings')}
   534→                      lang={language}
   535→                    />
   536→                    <Separator className="my-1" />
   537→                    <MobileNavItem
   538→                      icon={<LogOut className="w-5 h-5" />}
   539→                      label={t('logout', language)}
   540→                      active={false}
   541→                      onClick={() => {
   542→                        logout();
   543→                        handleNavigate('home');
   544→                      }}
   545→                      lang={language}
   546→                      destructive
   547→                    />
   548→                  </nav>
   549→                ) : (
   550→                  <div className="px-5 py-4 flex flex-col gap-3">
   551→                    <Button
   552→                      variant="outline"
   553→                      className="w-full min-h-[44px] border-primary/30 hover:bg-primary/10 hover:text-primary hover:border-primary"
   554→                      onClick={() => handleNavigate('login')}
   555→                    >
   556→                      {t('login', language)}
   557→                    </Button>
   558→                    <Button
   559→                      className="w-full min-h-[44px] rounded-lg"
   560→                      onClick={() => handleNavigate('register')}
   561→                    >
   562→                      {t('register', language)}
   563→                    </Button>
   564→                  </div>
   565→                )}
   566→
   567→                {/* Mobile Footer Controls */}
   568→                <div className="mt-auto border-t px-5 py-4 flex items-center justify-between gap-3">
   569→                  <Button
   570→                    variant="outline"
   571→                    size="sm"
   572→                    onClick={toggleLanguage}
   573→                    className="gap-1.5 min-h-[44px] border-blue/30 hover:bg-blue/10 hover:text-blue hover:border-blue"
   574→                  >
   575→                    <Globe className="w-4 h-4 text-blue" />
   576→                    {language === 'en' ? 'বাংলা' : 'English'}
   577→                  </Button>
   578→                  <div className="flex items-center gap-1">
   579→                    {([
   580→                      { mode: 'light' as const, icon: Sun, color: 'text-orange hover:bg-orange/10' },
   581→                      { mode: 'dark' as const, icon: Moon, color: 'text-blue hover:bg-blue/10' },
   582→                      { mode: 'system' as const, icon: Monitor, color: 'text-primary hover:bg-primary/10' }
   583→                    ]).map(({ mode, icon: Icon, color }) => (
   584→                      <Button
   585→                        key={mode}
   586→                        variant={theme === mode ? 'default' : 'ghost'}
   587→                        size="icon"
   588→                        className={`h-10 w-10 min-h-[44px] min-w-[44px] ${theme !== mode ? color : ''}`}
   589→                        onClick={() => setTheme(mode)}
   590→                      >
   591→                        <Icon className="w-4 h-4" />
   592→                      </Button>
   593→                    ))}
   594→                  </div>
   595→                </div>
   596→              </div>
   597→            </SheetContent>
   598→          </Sheet>
   599→        </div>
   600→      </div>
   601→
   602→      {/* Solid divider line under header when scrolled */}
   603→      {scrolled && <div className="h-0.5 bg-primary" />}
   604→    </motion.header>
   605→  );
   606→}

   608→// Desktop Nav Button
   609→function NavButton({
   610→  active,
   611→  onClick,
   612→  label,
   613→  lang,
   614→}: {
   615→  active: boolean;
   616→  onClick: () => void;
   617→  label: string;
   618→  lang: Language;
   619→}) {
   620→  return (
   621→    <motion.button
   622→      onClick={onClick}
   623→      className={`relative px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 ${
   624→        active
   625→          ? 'text-primary bg-primary/10'
   626→          : 'text-muted-foreground hover:bg-accent hover:text-accent-foreground'
   627→      } ${lang === 'bn' ? 'font-bangla' : ''}`}
   628→      whileHover={{ scale: 1.02 }}
   629→      whileTap={{ scale: 0.98 }}
   630→    >
   631→      {label}
   632→      {active && (
   633→        <motion.div
   634→          layoutId="activeNav"
   635→          className="absolute bottom-0 left-1/2 -translate-x-1/2 w-4/5 h-0.5 bg-primary rounded-full"
   636→          transition={{ type: 'spring', stiffness: 380, damping: 30 }}
   637→        />
   638→      )}
   639→    </motion.button>
   640→  );
   641→}
