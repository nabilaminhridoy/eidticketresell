   456→                      </Avatar>
   457→                      <div className="flex-1 min-w-0">
   458→                        <p className="text-sm font-medium truncate">{user.name}</p>
   459→                        <p className="text-xs text-muted-foreground truncate">{user.email}</p>
   460→                      </div>
   461→                      {user.isKycVerified && (
   462→                        <Badge variant="secondary" className="gap-1 shrink-0 bg-primary/10 text-primary">
   463→                          <Shield className="w-3 h-3" />
   464→                          {t('verified', language)}
   465→                        </Badge>
   466→                      )}
   467→                    </div>
   468→                    {/* User Menu Items */}
   469→                    <div className="space-y-1">
   470→                      {userMenuItems.map((menuItem) => (
   471→                        <MobileNavItem
   472→                          key={menuItem.id}
   473→                          icon={<menuItem.icon className="w-5 h-5" />}
   474→                          label={t(menuItem.labelKey, language)}
   475→                          active={isActive(menuItem.id)}
   476→                          onClick={() => handleNavigate(menuItem.id, userParams)}
   477→                          lang={language}
   478→                        />
   479→                      ))}
   480→                      <Separator className="my-2" />
   481→                      <MobileNavItem
   482→                        icon={<LogOut className="w-5 h-5" />}
   483→                        label={t('logout', language)}
   484→                        active={false}
   485→                        onClick={() => {
   486→                          logout();
   487→                          handleNavigate('home');
   488→                        }}
   489→                        lang={language}
   490→                        destructive
   491→                      />
   492→                    </div>
   493→                  </div>
   494→                ) : (
   495→                  <div className="px-5 py-4 flex gap-3">
   496→                    <Button
   497→                      variant="outline"
   498→                      className="w-full min-h-[44px] border-primary/30 hover:bg-primary/10 hover:text-primary hover:border-primary"
   499→                      onClick={() => handleNavigate('login')}
   500→                    >
   501→                      {t('login', language)}
   502→                    </Button>
   503→                    <Button
   504→                      className="w-full min-h-[44px] rounded-lg"
   505→                      onClick={() => handleNavigate('register')}
   506→                    >
   507→                      {t('register', language)}
   508→                    </Button>
   509→                  </div>
   510→                )}
   511→              </div>
   512→            </SheetContent>
   513→          </Sheet>
   514→        </div>
   515→      </div>
   516→
   517→      {/* Accent line under header when scrolled */}
   518→      {scrolled && <div className="h-0.5 bg-primary" />}
   519→    </header>
   520→  );
   521→}
