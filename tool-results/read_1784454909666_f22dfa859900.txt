     1→'use client';
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react';
     4→import { useAppStore, useAuthStore, useLanguageStore } from '@/lib/store';
     5→import { t } from '@/lib/i18n';
     6→import { BD_DIVISIONS, BD_DISTRICTS } from '@/lib/constants';
     7→import { Button } from '@/components/ui/button';
     8→import { Input } from '@/components/ui/input';
     9→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    10→import { Label } from '@/components/ui/label';
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    12→import { Badge } from '@/components/ui/badge';
    13→import { Separator } from '@/components/ui/separator';
    14→import { Progress } from '@/components/ui/progress';
    15→import {
    16→  ShieldCheck, User, Upload, Camera, MapPin, ArrowLeft, ArrowRight,
    17→  Loader2, Check, AlertCircle, FileText, ImageIcon, Eye, EyeOff,
    18→  RotateCcw, Lock, ChevronRight, Sun
    19→} from 'lucide-react';
    20→import { motion, AnimatePresence } from 'framer-motion';
    21→import { toast } from 'sonner';
    22→
    23→type Step = 1 | 2;
    24→
    25→interface KycData {
    26→  kycName: string;
    27→  kycDob: string;
    28→  kycGender: string;
    29→  nameChanged: boolean;
    30→  dobChanged: boolean;
    31→  genderChanged: boolean;
    32→  documentType: string;
    33→  documentNumber: string;
    34→  documentFront: string;
    35→  documentBack: string;
    36→  houseRoadVillage: string;
    37→  upazilaThana: string;
    38→  district: string;
    39→  division: string;
    40→  postalCode: string;
    41→  selfiePhoto: string;
    42→  selfieRight: string;
    43→  selfieLeft: string;
    44→  selfieSmile: string;
    45→  selfieBlink: string;
    46→  gpsLatitude: number | null;
    47→  gpsLongitude: number | null;
    48→  status: string;
    49→  reviewNote: string | null;
    50→}
    51→
    52→export default function KycPage() {
    53→  const { navigate } = useAppStore();
    54→  const { user, token, updateUser } = useAuthStore();
    55→  const { language } = useLanguageStore();
    56→  const isBn = language === 'bn';
    57→  const fontClass = isBn ? 'font-bangla' : '';
    58→
    59→  const [currentStep, setCurrentStep] = useState<Step>(1);
    60→  const [loading, setLoading] = useState(false);
    61→  const [error, setError] = useState('');
    62→  const [existingKyc, setExistingKyc] = useState<KycData | null>(null);
    63→  const [kycStatus, setKycStatus] = useState<string | null>(null);
    64→
    65→  // Step 1: Personal Info + Document + Address
    66→  const [kycName, setKycName] = useState('');
    67→  const [kycDob, setKycDob] = useState('');
    68→  const [kycGender, setKycGender] = useState('');
    69→  const [nameLocked, setNameLocked] = useState(false);
    70→  const [dobLocked, setDobLocked] = useState(false);
    71→  const [genderLocked, setGenderLocked] = useState(false);
    72→  const [documentType, setDocumentType] = useState('');
    73→  const [documentNumber, setDocumentNumber] = useState('');
    74→  const [documentFront, setDocumentFront] = useState('');
    75→  const [documentBack, setDocumentBack] = useState('');
    76→  const [houseRoadVillage, setHouseRoadVillage] = useState('');
    77→  const [upazilaThana, setUpazilaThana] = useState('');
    78→  const [district, setDistrict] = useState('');
    79→  const [division, setDivision] = useState('');
    80→  const [postalCode, setPostalCode] = useState('');
    81→
    82→  // Step 2: Selfie + GPS
    83→  const [selfiePhoto, setSelfiePhoto] = useState('');
    84→  const [selfieRight, setSelfieRight] = useState('');
    85→  const [selfieLeft, setSelfieLeft] = useState('');
    86→  const [selfieSmile, setSelfieSmile] = useState('');
    87→  const [selfieBlink, setSelfieBlink] = useState('');
    88→  const [gpsLatitude, setGpsLatitude] = useState<number | null>(null);
    89→  const [gpsLongitude, setGpsLongitude] = useState<number | null>(null);
    90→  const [gpsStatus, setGpsStatus] = useState<'pending' | 'granted' | 'denied'>('pending');
    91→
    92→  // Camera state
    93→  const videoRef = useRef<HTMLVideoElement>(null);
    94→  const canvasRef = useRef<HTMLCanvasElement>(null);
    95→  const [cameraActive, setCameraActive] = useState(false);
    96→  const [currentPose, setCurrentPose] = useState<'front' | 'right' | 'left' | 'smile' | 'blink'>('front');
    97→  const [capturing, setCapturing] = useState(false);
    98→
    99→  // Upload states
   100→  const [uploadingFront, setUploadingFront] = useState(false);
   101→  const [uploadingBack, setUploadingBack] = useState(false);
   102→
   103→  // Load existing KYC data
   104→  useEffect(() => {
   105→    if (!token) { navigate('login'); return; }
   106→    fetch('/api/kyc', { headers: { Authorization: `Bearer ${token}` } })
   107→      .then(r => r.json())
   108→      .then(data => {
   109→        if (data.kyc) {
   110→          setExistingKyc(data.kyc);
   111→          setKycStatus(data.kyc.status);
   112→          // Pre-fill form
   113→          setKycName(data.kyc.kycName || '');
   114→          setKycDob(data.kyc.kycDob || '');
   115→          setKycGender(data.kyc.kycGender || '');
   116→          setNameLocked(data.kyc.nameChanged);
   117→          setDobLocked(data.kyc.dobChanged);
   118→          setGenderLocked(data.kyc.genderChanged);
   119→          setDocumentType(data.kyc.documentType || '');
   120→          setDocumentNumber(data.kyc.documentNumber || '');
   121→          setDocumentFront(data.kyc.documentFront || '');
   122→          setDocumentBack(data.kyc.documentBack || '');
   123→          setHouseRoadVillage(data.kyc.houseRoadVillage || '');
   124→          setUpazilaThana(data.kyc.upazilaThana || '');
   125→          setDistrict(data.kyc.district || '');
   126→          setDivision(data.kyc.division || '');
   127→          setPostalCode(data.kyc.postalCode || '');
   128→          setSelfiePhoto(data.kyc.selfiePhoto || '');
   129→          setSelfieRight(data.kyc.selfieRight || '');
   130→          setSelfieLeft(data.kyc.selfieLeft || '');
   131→          setSelfieSmile(data.kyc.selfieSmile || '');
   132→          setSelfieBlink(data.kyc.selfieBlink || '');
   133→          setGpsLatitude(data.kyc.gpsLatitude);
   134→          setGpsLongitude(data.kyc.gpsLongitude);
   135→        } else {
   136→          // Pre-fill from profile
   137→          if (data.profileName) setKycName(data.profileName);
   138→          if (data.profileDob) setKycDob(data.profileDob);
   139→          if (data.profileGender) setKycGender(data.profileGender);
   140→        }
   141→        setKycStatus(data.isKycVerified ? 'approved' : data.kyc?.status || null);
   142→      })
   143→      .catch(() => {});
   144→  }, [token, navigate]);
   145→
   146→  // Request GPS
   147→  useEffect(() => {
   148→    if (navigator.geolocation) {
   149→      navigator.geolocation.getCurrentPosition(
   150→        (pos) => {
   151→          setGpsLatitude(pos.coords.latitude);
   152→          setGpsLongitude(pos.coords.longitude);
   153→          setGpsStatus('granted');
   154→        },
   155→        () => setGpsStatus('denied'),
   156→        { enableHighAccuracy: true }
   157→      );
   158→    }
   159→  }, []);
   160→
   161→  // Get districts for selected division
   162→  const districts = division ? (BD_DISTRICTS[division] || []) : [];
   163→
   164→  // File upload handler
   165→  const uploadFile = async (file: File): Promise<string> => {
   166→    const formData = new FormData();
   167→    formData.append('file', file);
   168→    const res = await fetch('/api/upload', {
   169→      method: 'POST',
   170→      headers: { Authorization: `Bearer ${token}` },
   171→      body: formData,
   172→    });
   173→    const data = await res.json();
   174→    if (!res.ok) throw new Error(data.error || 'Upload failed');
   175→    return data.url;
   176→  };
   177→
   178→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>, side: 'front' | 'back') => {
   179→    const file = e.target.files?.[0];
   180→    if (!file) return;
   181→
   182→    if (side === 'front') setUploadingFront(true);
   183→    else setUploadingBack(true);
   184→
   185→    try {
   186→      const url = await uploadFile(file);
   187→      if (side === 'front') setDocumentFront(url);
   188→      else setDocumentBack(url);
   189→      toast.success(isBn ? 'আপলোড সফল!' : 'Upload successful!');
   190→    } catch (err) {
   191→      toast.error(err instanceof Error ? err.message : 'Upload failed');
   192→    } finally {
   193→      if (side === 'front') setUploadingFront(false);
   194→      else setUploadingBack(false);
   195→    }
   196→  };
   197→
   198→  // Camera functions
   199→  const startCamera = useCallback(async () => {
   200→    try {
   201→      const stream = await navigator.mediaDevices.getUserMedia({
   202→        video: { facingMode: 'user', width: 640, height: 480 }
   203→      });
   204→      if (videoRef.current) {
   205→        videoRef.current.srcObject = stream;
   206→        setCameraActive(true);
   207→      }
   208→    } catch {
   209→      toast.error(isBn ? 'ক্যামেরা অ্যাক্সেস ব্যর্থ' : 'Camera access failed');
   210→    }
   211→  }, [isBn]);
   212→
   213→  const stopCamera = useCallback(() => {
   214→    if (videoRef.current?.srcObject) {
   215→      const stream = videoRef.current.srcObject as MediaStream;
   216→      stream.getTracks().forEach(track => track.stop());
   217→      videoRef.current.srcObject = null;
   218→    }
   219→    setCameraActive(false);
   220→  }, []);
   221→
   222→  const capturePhoto = useCallback(async (): Promise<string> => {
   223→    if (!videoRef.current || !canvasRef.current) return '';
   224→    const canvas = canvasRef.current;
   225→    const video = videoRef.current;
   226→    canvas.width = video.videoWidth;
   227→    canvas.height = video.videoHeight;
   228→    const ctx = canvas.getContext('2d');
   229→    if (!ctx) return '';
   230→    ctx.drawImage(video, 0, 0);
   231→    const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
   232→
   233→    // Convert to file and upload
   234→    const blob = await (await fetch(dataUrl)).blob();
   235→    const file = new File([blob], `selfie-${currentPose}.jpg`, { type: 'image/jpeg' });
   236→    return await uploadFile(file);
   237→  }, [currentPose]);
   238→
   239→  const handleCapture = async () => {
   240→    setCapturing(true);
   241→    try {
   242→      const url = await capturePhoto();
   243→      switch (currentPose) {
   244→        case 'front': setSelfiePhoto(url); break;
   245→        case 'right': setSelfieRight(url); break;
   246→        case 'left': setSelfieLeft(url); break;
   247→        case 'smile': setSelfieSmile(url); break;
   248→        case 'blink': setSelfieBlink(url); break;
   249→      }
   250→    } catch (err) {
   251→      toast.error(isBn ? 'ক্যাপচার ব্যর্থ' : 'Capture failed');
   252→    } finally {
   253→      setCapturing(false);
   254→    }
   255→  };
   256→
   257→  // Document number label based on type
   258→  const getDocNumberLabel = () => {
   259→    switch (documentType) {
   260→      case 'nid': return t('nidNumber', language);
   261→      case 'driving_licence': return t('drivingLicenceNumber', language);
   262→      case 'passport': return t('passportNumber', language);
   263→      default: return t('documentNumber', language);
   264→    }
   265→  };
   266→
   267→  // Validate NID
   268→  const isValidNid = (num: string) => {
   269→    const digits = num.replace(/\D/g, '');
   270→    return [10, 13, 17].includes(digits.length);
   271→  };
   272→
   273→  // Step 1 validation
   274→  const isStep1Valid = () => {
   275→    return (
   276→      kycName.trim().length >= 2 &&
   277→      kycDob &&
   278→      kycGender &&
   279→      documentType &&
   280→      documentNumber &&
   281→      documentFront &&
   282→      (documentType === 'passport' || documentBack) &&
   283→      houseRoadVillage &&
   284→      upazilaThana &&
   285→      district &&
   286→      division
   287→    );
   288→  };
   289→
   290→  // Step 2 validation
   291→  const isStep2Valid = () => {
   292→    return selfiePhoto && gpsStatus === 'granted';
   293→  };
   294→
   295→  // Submit KYC
   296→  const handleSubmit = async () => {
   297→    if (!isStep2Valid()) {
   298→      setError(isBn ? 'সেলফি ক্যাপচার ও জিপিএস প্রয়োজন' : 'Selfie capture and GPS are required');
   299→      return;
   300→    }
   301→    setLoading(true);
   302→    setError('');
   303→    try {
   304→      const res = await fetch('/api/kyc', {
   305→        method: 'POST',
   306→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   307→        body: JSON.stringify({
   308→          kycName, kycDob, kycGender,
   309→          documentType, documentNumber, documentFront, documentBack,
   310→          houseRoadVillage, upazilaThana, district, division, postalCode,
   311→          selfiePhoto, selfieRight, selfieLeft, selfieSmile, selfieBlink,
   312→          gpsLatitude, gpsLongitude,
   313→        }),
   314→      });
   315→      const data = await res.json();
   316→      if (!res.ok) throw new Error(data.error || 'KYC submission failed');
   317→
   318→      toast.success(t('kycSubmitSuccess', language));
   319→      setKycStatus('pending');
   320→      setExistingKyc(data.kyc);
   321→      stopCamera();
   322→    } catch (err) {
   323→      setError(err instanceof Error ? err.message : 'Something went wrong');
   324→    } finally {
   325→      setLoading(false);
   326→    }
   327→  };
   328→
   329→  // Show status page if already submitted
   330→  if (kycStatus === 'pending' && existingKyc) {
   331→    return (
   332→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-primary/5 via-background to-primary/10 p-4">
   333→        <Card className="w-full max-w-md shadow-xl border-primary/10">
   334→          <CardContent className="p-8 text-center space-y-4">
   335→            <div className="mx-auto w-16 h-16 rounded-full bg-amber-100 dark:bg-amber-900/30 flex items-center justify-center">
   336→              <Loader2 className="w-8 h-8 text-amber-600 animate-spin" />
   337→            </div>
   338→            <h2 className={`text-xl font-bold ${fontClass}`}>{t('kycPending', language)}</h2>
   339→            <p className={`text-sm text-muted-foreground ${fontClass}`}>{t('kycPendingMessage', language)}</p>
   340→            {existingKyc.reviewNote && (
   341→              <div className="p-3 rounded-lg bg-muted text-sm">{existingKyc.reviewNote}</div>
   342→            )}
   343→            <Button variant="outline" onClick={() => navigate('home')} className={fontClass}>
   344→              {t('back', language)}
   345→            </Button>
   346→          </CardContent>
   347→        </Card>
   348→      </div>
   349→    );
   350→  }
   351→
   352→  if (kycStatus === 'approved') {
   353→    return (
   354→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-primary/5 via-background to-primary/10 p-4">
   355→        <Card className="w-full max-w-md shadow-xl border-emerald-200 dark:border-emerald-900/30">
   356→          <CardContent className="p-8 text-center space-y-4">
   357→            <div className="mx-auto w-16 h-16 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">
   358→              <ShieldCheck className="w-8 h-8 text-emerald-600" />
   359→            </div>
   360→            <h2 className={`text-xl font-bold ${fontClass}`}>{t('kycApproved', language)}</h2>
   361→            <p className={`text-sm text-muted-foreground ${fontClass}`}>{t('kycApprovedMessage', language)}</p>
   362→            <div className="grid grid-cols-3 gap-3 py-4">
   363→              {[
   364→                { icon: Check, label: t('verifiedBadge', language) },
   365→                { icon: FileText, label: t('canSellTickets', language) },
   366→                { icon: MapPin, label: t('canWithdraw', language) },
   367→              ].map(item => (
   368→                <div key={item.label} className="flex flex-col items-center gap-2 p-3 rounded-xl bg-emerald-50 dark:bg-emerald-900/20">
   369→                  <item.icon className="w-5 h-5 text-emerald-600" />
   370→                  <span className={`text-xs text-emerald-700 dark:text-emerald-400 text-center ${fontClass}`}>{item.label}</span>
   371→                </div>
   372→              ))}
   373→            </div>
   374→            <Button onClick={() => navigate('home')} className="bg-gradient-to-r from-emerald-600 to-emerald-500">
   375→              {isBn ? 'হোমে যান' : 'Go Home'}
   376→            </Button>
   377→          </CardContent>
   378→        </Card>
   379→      </div>
   380→    );
   381→  }
   382→
   383→  return (
   384→    <div className="min-h-screen bg-gradient-to-br from-primary/5 via-background to-primary/10 py-4 sm:py-6 px-3 sm:px-4 overflow-x-hidden">
   385→      <div className="w-full max-w-2xl mx-auto">
   386→        {/* Header */}
   387→        <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="text-center mb-4 sm:mb-6">
   388→          <div className="mx-auto w-12 h-12 sm:w-14 sm:h-14 rounded-2xl bg-gradient-to-br from-primary to-primary/80 flex items-center justify-center mb-2 sm:mb-3 shadow-lg">
   389→            <ShieldCheck className="w-6 h-6 sm:w-7 sm:h-7 text-primary-foreground" />
   390→          </div>
   391→          <h1 className={`text-xl sm:text-2xl font-bold ${fontClass}`}>{t('kycVerification', language)}</h1>
   392→          <p className={`text-xs sm:text-sm text-muted-foreground mt-1 ${fontClass}`}>{t('becomeVerifiedSeller', language)}</p>
   393→        </motion.div>
   394→
   395→        {/* Step indicator */}
   396→        <div className="flex items-center justify-center gap-1 sm:gap-2 mb-4 sm:mb-6">
   397→          {[
   398→            { num: 1, label: isBn ? 'নথি ও ঠিকানা' : 'Document & Address', shortLabel: isBn ? 'নথি' : 'Docs', icon: FileText },
   399→            { num: 2, label: isBn ? 'সেলফি ও জিপিএস' : 'Selfie & GPS', shortLabel: isBn ? 'সেলফি' : 'Selfie', icon: Camera },
   400→          ].map((step, idx) => (
   401→            <div key={step.num} className="flex items-center">
   402→              <div className={`flex items-center gap-1 sm:gap-2 px-2.5 sm:px-4 py-1.5 sm:py-2 rounded-full text-xs font-medium transition-all ${
   403→                currentStep >= step.num
   404→                  ? 'bg-primary text-primary-foreground shadow-md'
   405→                  : 'bg-muted text-muted-foreground'
   406→              } ${fontClass}`}>
   407→                <step.icon className="w-3.5 h-3.5 shrink-0" />
   408→                <span className="hidden sm:inline">{step.label}</span>
   409→                <span className="sm:hidden">{step.shortLabel}</span>
   410→              </div>
   411→              {idx < 1 && (
   412→                <div className={`w-4 sm:w-8 h-0.5 mx-0.5 sm:mx-1 rounded transition-all ${
   413→                  currentStep > step.num ? 'bg-primary' : 'bg-muted'
   414→                }`} />
   415→              )}
   416→            </div>
   417→          ))}
   418→        </div>
   419→
   420→        <Card className="shadow-xl border-primary/10">
   421→          <CardContent className="p-3 sm:p-6">
   422→            <AnimatePresence mode="wait">
   423→              {/* STEP 1: Personal Info + Document + Address */}
   424→              {currentStep === 1 && (
   425→                <motion.div key="step1" initial={{ opacity: 0, x: 20 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -20 }} className="space-y-4 sm:space-y-6">
   426→                  {/* Personal Information */}
   427→                  <div>
   428→                    <h3 className={`text-sm sm:text-base font-semibold mb-2 sm:mb-3 flex items-center gap-2 ${fontClass}`}>
   429→                      <User className="w-4 h-4 text-primary" />
   430→                      {t('kycPersonalInfo', language)}
   431→                    </h3>
   432→                    <div className="space-y-3">
   433→                      {/* Full Name */}
   434→                      <div className="space-y-1.5">
   435→                        <Label className={`text-sm ${fontClass}`}>
   436→                          {t('fullName', language)} <span className="text-destructive">*</span>
   437→                        </Label>
   438→                        <div className="relative">
   439→                          <Input value={kycName} onChange={(e) => !nameLocked && setKycName(e.target.value)} disabled={nameLocked} className={`h-11 ${nameLocked ? 'opacity-70' : ''}`} />
   440→                          {nameLocked && <Lock className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />}
   441→                        </div>
   442→                        {nameLocked ? (
   443→                          <p className="text-xs text-amber-600 flex items-center gap-1"><Lock className="w-3 h-3" />{t('kycChangedOnce', language)}</p>
   444→                        ) : (
   445→                          <p className="text-xs text-amber-600 flex items-center gap-1"><AlertCircle className="w-3 h-3" />{t('kycNameChangeWarning', language)}</p>
   446→                        )}
   447→                      </div>
   448→
   449→                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   450→                        {/* DOB */}
   451→                        <div className="space-y-1.5">
   452→                          <Label className={`text-sm ${fontClass}`}>
   453→                            {t('dateOfBirth', language)} <span className="text-destructive">*</span>
   454→                          </Label>
   455→                          <Input type="date" value={kycDob} onChange={(e) => !dobLocked && setKycDob(e.target.value)} disabled={dobLocked} className={`h-11 ${dobLocked ? 'opacity-70' : ''}`} />
   456→                          {dobLocked && <p className="text-xs text-amber-600 flex items-center gap-1"><Lock className="w-3 h-3" />{t('kycChangedOnce', language)}</p>}
   457→                          {!dobLocked && <p className="text-xs text-amber-600">{t('kycDobChangeWarning', language)}</p>}
   458→                        </div>
   459→
   460→                        {/* Gender */}
   461→                        <div className="space-y-1.5">
