     4→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     5→import { Button } from '@/components/ui/button';

    10→import {

    12→  Palette, CreditCard, FolderOpen, Eye, RefreshCw, AlertCircle
    13→} from 'lucide-react';
    14→

    16→  const token = localStorage.getItem('etr_admin_token');
    17→  return { 'Authorization': `Bearer ${token}` };
    18→}

    22→  value: string;
    23→  group: string;
    24→}
    25→
    26→interface MediaFile {
    27→  key: string;
    28→  value: string;
    29→  group: string;
    30→}

    37→];

    40→const PAYMENT_IMAGES = [
    41→  { key: 'payment_gateway_logo', label: 'Payment Gateway Logo', description: 'General payment section logo', category: 'payment', subDir: 'payment' },
    42→  { key: 'bkash_logo', label: 'bKash Logo', description: 'bKash mobile payment logo', category: 'payment', subDir: 'payment' },
    43→  { key: 'nagad_logo', label: 'Nagad Logo', description: 'Nagad mobile payment logo', category: 'payment', subDir: 'payment' },

    45→];
    46→
    47→export default function AdminMediaPage({ section }: { section?: string }) {
    48→  const [imageSettings, setImageSettings] = useState<Record<string, string>>({});
    49→  const [mediaFiles, setMediaFiles] = useState<MediaFile[]>([]);
    50→  const [loading, setLoading] = useState(true);
    51→  const [uploadingKey, setUploadingKey] = useState<string | null>(null);
    52→  const [uploadProgress, setUploadProgress] = useState<string>('');
    53→  const [deletingKey, setDeletingKey] = useState<string | null>(null);
    54→  const [successMessage, setSuccessMessage] = useState<string>('');
    55→  const [dragOverKey, setDragOverKey] = useState<string | null>(null);
    56→  const [searchQuery, setSearchQuery] = useState('');
    57→  const fileInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    58→  const currentSection = section || null;
    59→
    60→  const fetchData = useCallback(async () => {
    61→    setLoading(true);
    62→    try {
    63→      const res = await fetch('/api/admin/media', { headers: getAuthHeaders() });
    64→      if (res.ok) {
    65→        const data = await res.json();
    66→        const map: Record<string, string> = {};
    67→        if (data.images) {
    68→          data.images.forEach((s: ImageSetting) => {
    69→            map[s.key] = s.value;
    70→          });
    71→        }
    72→        setImageSettings(map);
    73→        setMediaFiles(data.media || []);
    74→      }
    75→    } catch {
    76→      // silently handle
    77→    }
    78→    setLoading(false);
    79→  }, []);
    80→
    81→  useEffect(() => {
    82→    fetchData();
    83→  }, [fetchData]);
    84→
    85→  const showMessage = (msg: string) => {
    86→    setSuccessMessage(msg);
    87→    setTimeout(() => setSuccessMessage(''), 3000);
    88→  };
    89→
    90→  const handleUpload = async (file: File, settingKey: string, category: string, subDir: string) => {
    91→    setUploadingKey(settingKey);
    92→    setUploadProgress(`Uploading ${file.name}...`);
    93→
    94→    const formData = new FormData();
    95→    formData.append('file', file);
    96→    formData.append('settingKey', settingKey);
    97→    formData.append('category', category);
    98→
    99→    try {
   100→      const res = await fetch('/api/admin/media', {
   101→        method: 'POST',
   102→        headers: getAuthHeaders(),
   103→        body: formData,
   104→      });
   105→
   106→      if (res.ok) {
   107→        const data = await res.json();
   108→        setImageSettings(prev => ({ ...prev, [settingKey]: data.path }));
   109→        showMessage(`${settingKey} uploaded successfully!`);
   110→        await fetchData(); // Refresh data
   111→      } else {
   112→        const err = await res.json();
   113→        setUploadProgress(`Upload failed: ${err.error || 'Unknown error'}`);
   114→        setTimeout(() => setUploadProgress(''), 3000);
   115→      }
   116→    } catch {
   117→      setUploadProgress('Upload failed: Network error');
   118→      setTimeout(() => setUploadProgress(''), 3000);
   119→    }
   120→
   121→    setUploadingKey(null);
   122→    setUploadProgress('');
   123→  };
   124→
   125→  const handleDelete = async (settingKey: string) => {
   126→    setDeletingKey(settingKey);
   127→    try {
   128→      const res = await fetch(`/api/admin/media?key=${settingKey}`, {
   129→        method: 'DELETE',
   130→        headers: getAuthHeaders(),
   131→      });
   132→      if (res.ok) {
   133→        setImageSettings(prev => ({ ...prev, [settingKey]: '' }));
   134→        showMessage(`${settingKey} removed successfully!`);
   135→        await fetchData();
   136→      }
   137→    } catch {
   138→      // silently handle
   139→    }
   140→    setDeletingKey(null);
   141→  };
   142→
   143→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>, item: { key: string; category: string; subDir: string }) => {
   144→    const file = e.target.files?.[0];
   145→    if (file) {
   146→      // Validate file size (max 5MB)
   147→      if (file.size > 5 * 1024 * 1024) {
   148→        setUploadProgress('File too large. Maximum size is 5MB.');
   149→        setTimeout(() => setUploadProgress(''), 3000);
   150→        return;
   151→      }
   152→      // Validate file type
   153→      const allowedTypes = ['image/png', 'image/jpeg', 'image/gif', 'image/svg+xml', 'image/webp', 'image/x-icon', 'image/vnd.microsoft.icon'];
   154→      if (!allowedTypes.includes(file.type)) {
   155→        setUploadProgress('Invalid file type. Allowed: PNG, JPG, GIF, SVG, WEBP, ICO');
   156→        setTimeout(() => setUploadProgress(''), 3000);
   157→        return;
   158→      }
   159→      handleUpload(file, item.key, item.category, item.subDir);
   160→    }
   161→    // Reset the input so the same file can be re-selected
   162→    e.target.value = '';
   163→  };
   164→
   165→  const handleDrop = (e: React.DragEvent, item: { key: string; category: string; subDir: string }) => {
   166→    e.preventDefault();
   167→    e.stopPropagation();
   168→    setDragOverKey(null);
   169→    const file = e.dataTransfer.files?.[0];
   170→    if (file) {
   171→      handleUpload(file, item.key, item.category, item.subDir);
   172→    }
   173→  };
   174→
   175→  const handleDragOver = (e: React.DragEvent, key: string) => {
   176→    e.preventDefault();
   177→    e.stopPropagation();
   178→    setDragOverKey(key);
   179→  };
   180→
   181→  const handleDragLeave = (e: React.DragEvent) => {
   182→    e.preventDefault();
   183→    e.stopPropagation();
   184→    setDragOverKey(null);
   185→  };
   186→
   187→  // Image upload card component
   188→  const ImageUploadCard = ({ item }: { item: { key: string; label: string; description: string; category: string; subDir: string } }) => {
   189→    const currentValue = imageSettings[item.key] || '';
   190→    const isUploading = uploadingKey === item.key;
   191→    const isDeleting = deletingKey === item.key;
   192→    const isDragOver = dragOverKey === item.key;
   193→
   194→    return (
   195→      <Card className="overflow-hidden">
   196→        <CardContent className="p-4 space-y-3">
   197→          <div className="flex items-center justify-between">
   198→            <div>
   199→              <h4 className="font-medium text-sm">{item.label}</h4>
   200→              <p className="text-xs text-muted-foreground">{item.description}</p>
   201→            </div>
   202→            {currentValue && (
   203→              <Button
   204→                variant="ghost"
   205→                size="sm"
   206→                className="text-red-600 hover:text-red-700 hover:bg-red-50 h-8"
   207→                disabled={isDeleting}
   208→                onClick={() => handleDelete(item.key)}
   209→              >
   210→                {isDeleting ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Trash2 className="w-3.5 h-3.5" />}
   211→              </Button>
   212→            )}
   213→          </div>
   214→
   215→          {/* Image preview / upload drop zone */}
   216→          <div
   217→            className={`relative rounded-lg border-2 transition-all ${
   218→              isDragOver ? 'border-primary bg-primary/10 scale-[1.02]' :
   219→              currentValue ? 'border-border' : 'border-dashed border-muted-foreground/30 hover:border-primary/50 bg-muted/10'
   220→            }`}
   221→            onDrop={(e) => handleDrop(e, item)}
   222→            onDragOver={(e) => handleDragOver(e, item.key)}
   223→            onDragLeave={handleDragLeave}
   224→            onClick={() => {
   225→              if (!isUploading && fileInputRefs.current[item.key]) {
   226→                fileInputRefs.current[item.key]?.click();
   227→              }
   228→            }}
   229→            role="button"
   230→            tabIndex={0}
   231→            aria-label={`Upload ${item.label}`}
   232→          >
   233→            <input
   234→              ref={(el) => { fileInputRefs.current[item.key] = el; }}
   235→              type="file"
   236→              accept="image/png,image/jpeg,image/gif,image/svg+xml,image/webp,image/x-icon,image/vnd.microsoft.icon"
   237→              className="hidden"
   238→              onChange={(e) => handleFileSelect(e, item)}
   239→              aria-hidden="true"
   240→            />
   241→
   242→            {isUploading ? (
   243→              <div className="aspect-[4/3] flex items-center justify-center">
   244→                <div className="flex flex-col items-center gap-2">
   245→                  <Loader2 className="w-8 h-8 animate-spin text-primary" />
   246→                  <p className="text-xs text-muted-foreground">{uploadProgress}</p>
   247→                </div>
   248→              </div>
   249→            ) : currentValue ? (
   250→              <div className="aspect-[4/3] flex items-center justify-center overflow-hidden bg-muted/20 relative group">
   251→                {/* eslint-disable-next-line @next/next/no-img-element */}
   252→                <img
   253→                  src={currentValue}
   254→                  alt={item.label}
   255→                  className="max-w-full max-h-full object-contain p-2 transition-transform group-hover:scale-105"
   256→                />
   257→                <div className="absolute inset-0 bg-black/0 group-hover:bg-black/40 transition-all flex items-center justify-center opacity-0 group-hover:opacity-100">
   258→                  <div className="flex gap-2">
   259→                    <Eye className="w-5 h-5 text-white" />
   260→                    <Upload className="w-5 h-5 text-white" />
   261→                  </div>
   262→                </div>
   263→              </div>
   264→            ) : (
   265→              <div className="aspect-[4/3] flex items-center justify-center cursor-pointer">
   266→                <div className="flex flex-col items-center gap-2 text-muted-foreground">
   267→                  <Upload className="w-8 h-8" />
   268→                  <p className="text-sm font-medium">Drop image or click to upload</p>
   269→                  <p className="text-xs">PNG, JPG, SVG, GIF, WEBP, ICO (max 5MB)</p>
   270→                </div>
   271→              </div>
   272→            )}
   273→          </div>
   274→
   275→          {currentValue && (
   276→            <div className="flex items-center gap-2">
   277→              <Badge variant="outline" className="text-xs">
   278→                <CheckCircle2 className="w-3 h-3 mr-1 text-emerald-500" />
   279→                Uploaded
   280→              </Badge>
   281→              <span className="text-xs text-muted-foreground truncate max-w-[200px]">{currentValue}</span>
   282→            </div>
   283→          )}
   284→        </CardContent>
   285→      </Card>
   286→    );
   287→  };
   288→
   289→  // Upload view
   290→  if (currentSection === 'upload') {
   291→    return (
   292→      <div className="space-y-6">
   293→        <h1 className="text-2xl font-bold flex items-center gap-2"><Upload className="w-6 h-6" />Upload Media</h1>
   294→        <Card>
   295→          <CardContent className="p-6 space-y-4">
   296→            <div className="p-8 border-2 border-dashed rounded-lg text-center bg-muted/10 hover:bg-muted/20 transition-colors cursor-pointer">
   297→              <Upload className="w-12 h-12 text-muted-foreground mx-auto mb-3" />
   298→              <p className="font-medium">Drag and drop files here</p>
   299→              <p className="text-sm text-muted-foreground">or click to browse your files</p>
   300→              <p className="text-xs text-muted-foreground mt-2">Supported: JPG, PNG, GIF, SVG, WEBP, ICO (Max 5MB)</p>
   301→            </div>
   302→          </CardContent>
   303→        </Card>
   304→      </div>
   305→    );
   306→  }
   307→
   308→  if (loading) {
   309→    return (
   310→      <div className="flex items-center justify-center py-12">
   311→        <Loader2 className="w-6 h-6 animate-spin text-primary" />
   312→      </div>
   313→    );
   314→  }
   315→
   316→  // Success message toast
   317→  const SuccessToast = successMessage ? (
   318→    <div className="fixed top-4 right-4 z-50 animate-in fade-in slide-in-from-top-2 duration-300">
   319→      <div className="flex items-center gap-2 p-3 rounded-lg bg-emerald-50 border border-emerald-200 text-emerald-700 shadow-lg">
   320→        <CheckCircle2 className="w-4 h-4" />
   321→        <span className="text-sm font-medium">{successMessage}</span>
   322→        <button onClick={() => setSuccessMessage('')} className="ml-2 hover:bg-emerald-100 rounded p-0.5">
   323→          <X className="w-3 h-3" />
   324→        </button>
   325→      </div>
   326→    </div>
   327→  ) : null;
   328→
   329→  // Upload progress toast
   330→  const ProgressToast = uploadProgress ? (
   331→    <div className="fixed top-4 right-4 z-50">
   332→      <div className="flex items-center gap-2 p-3 rounded-lg bg-yellow-50 border border-yellow-200 text-yellow-700 shadow-lg">
   333→        <AlertCircle className="w-4 h-4" />
   334→        <span className="text-sm">{uploadProgress}</span>
   335→      </div>
   336→    </div>
   337→  ) : null;
   338→
   339→  return (
   340→    <div className="space-y-6">
   341→      {SuccessToast}
   342→      {ProgressToast}
   343→
   344→      {/* Header */}
   345→      <div className="flex items-center justify-between">
   346→        <div>
   347→          <h1 className="text-2xl font-bold flex items-center gap-2"><ImageLucide className="w-6 h-6" />Media Library</h1>
   348→          <p className="text-sm text-muted-foreground">Manage logos, favicon, payment images, and uploaded media</p>
   349→        </div>
   350→        <Button variant="outline" size="sm" className="gap-1" onClick={fetchData} disabled={loading}>
   351→          <RefreshCw className={`w-3.5 h-3.5 ${loading ? 'animate-spin' : ''}`} />Refresh
   352→        </Button>
   353→      </div>
   354→
   355→      <Tabs defaultValue="brand">
   356→        <TabsList>
   357→          <TabsTrigger value="brand" className="gap-1">
   358→            <Palette className="w-3.5 h-3.5" />Brand Assets
   359→          </TabsTrigger>
   360→          <TabsTrigger value="payment" className="gap-1">
   361→            <CreditCard className="w-3.5 h-3.5" />Payment Images
   362→          </TabsTrigger>
   363→          <TabsTrigger value="library" className="gap-1">
   364→            <FolderOpen className="w-3.5 h-3.5" />Media Library
   365→          </TabsTrigger>
   366→        </TabsList>
   367→
   368→        {/* Brand Assets Tab */}
   369→        <TabsContent value="brand" className="space-y-6">
   370→          <Card>
   371→            <CardHeader>
   372→              <CardTitle className="flex items-center gap-2">
   373→                <Palette className="w-5 h-5 text-primary" />
   374→                Logo & Favicon
   375→              </CardTitle>
   376→              <CardDescription>
   377→                Upload and manage your site&apos;s brand identity assets. These images appear in the header, browser tab, and throughout the site.
   378→              </CardDescription>
   379→            </CardHeader>
   380→            <CardContent>
   381→              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   382→                {BRAND_IMAGES.map((item) => (
   383→                  <ImageUploadCard key={item.key} item={item} />
   384→                ))}
   385→              </div>
   386→            </CardContent>
   387→          </Card>
   388→
   389→          {/* Preview section */}
   390→          {(imageSettings['site_logo'] || imageSettings['dark_mode_logo'] || imageSettings['site_favicon']) && (
   391→            <Card>
   392→              <CardHeader>
   393→                <CardTitle className="text-lg">Brand Preview</CardTitle>
   394→                <CardDescription>How your brand assets appear on the site</CardDescription>
   395→              </CardHeader>
   396→              <CardContent>
   397→                <div className="flex flex-wrap items-center gap-6 p-4 rounded-lg bg-muted/30">
   398→                  {/* Light mode preview */}
   399→                  <div className="flex items-center gap-3 p-3 rounded-lg bg-white border">
   400→                    <span className="text-xs text-gray-500 mr-1">Light:</span>
   401→                    {imageSettings['site_logo'] ? (
   402→                      // eslint-disable-next-line @next/next/no-img-element
   403→                      <img src={imageSettings['site_logo']} alt="Site Logo" className="h-8 max-w-[120px] object-contain" />
   404→                    ) : (
   405→                      <span className="text-xs text-gray-400">No logo</span>
   406→                    )}
   407→                  </div>
   408→                  {/* Dark mode preview */}
   409→                  <div className="flex items-center gap-3 p-3 rounded-lg bg-gray-900 border border-gray-700">
   410→                    <span className="text-xs text-gray-400 mr-1">Dark:</span>
   411→                    {imageSettings['dark_mode_logo'] ? (
   412→                      // eslint-disable-next-line @next/next/no-img-element
   413→                      <img src={imageSettings['dark_mode_logo']} alt="Dark Mode Logo" className="h-8 max-w-[120px] object-contain" />
   414→                    ) : (
   415→                      <span className="text-xs text-gray-500">No logo</span>
   416→                    )}
   417→                  </div>
   418→                  {/* Favicon preview */}
   419→                  <div className="flex items-center gap-3 p-3 rounded-lg bg-muted/30 border">
   420→                    <span className="text-xs text-muted-foreground mr-1">Favicon:</span>
   421→                    {imageSettings['site_favicon'] ? (
   422→                      // eslint-disable-next-line @next/next/no-img-element
   423→                      <img src={imageSettings['site_favicon']} alt="Favicon" className="h-6 w-6 object-contain" />
   424→                    ) : (
   425→                      <span className="text-xs text-muted-foreground">No favicon</span>
   426→                    )}
   427→                  </div>
   428→                </div>
   429→              </CardContent>
   430→            </Card>
   431→          )}
   432→        </TabsContent>
   433→
   434→        {/* Payment Gateway Images Tab */}
   435→        <TabsContent value="payment" className="space-y-6">
   436→          <Card>
   437→            <CardHeader>
   438→              <CardTitle className="flex items-center gap-2">
   439→                <CreditCard className="w-5 h-5 text-primary" />
   440→                Payment Gateway Images
   441→              </CardTitle>
   442→              <CardDescription>
   443→                Upload logos for payment methods displayed on the checkout and wallet pages.
   444→              </CardDescription>
   445→            </CardHeader>
   446→            <CardContent>
   447→              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   448→                {PAYMENT_IMAGES.map((item) => (
   449→                  <ImageUploadCard key={item.key} item={item} />
   450→                ))}
   451→              </div>
   452→            </CardContent>
   453→          </Card>
   454→
   455→          {/* Payment preview section */}
   456→          {(imageSettings['bkash_logo'] || imageSettings['nagad_logo'] || imageSettings['rocket_logo'] || imageSettings['payment_gateway_logo']) && (
   457→            <Card>
