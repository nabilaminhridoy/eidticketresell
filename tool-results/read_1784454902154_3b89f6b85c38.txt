   133→  switch (type) {
   134→    case 'bus':
   135→      return <Bus className={className} />;
   136→    case 'train':
   137→      return <Train className={className} />;
   138→    case 'flight':
   139→      return <Plane className={className} />;
   140→    case 'launch':
   141→      return <Ship className={className} />;
   142→    default:
   143→      return <Bus className={className} />;
   144→  }
   145→}

   208→export default function SellTicketPage() {
   209→  const { navigate } = useAppStore();
   210→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   211→  const { language } = useLanguageStore();
   212→  const isBn = language === 'bn';
   213→
   214→  const [form, setForm] = useState<FormState>(initialForm);
   215→  const [uploading, setUploading] = useState(false);
   216→  const [uploadFileName, setUploadFileName] = useState('');
   217→  const [submitting, setSubmitting] = useState(false);
   218→  const [error, setError] = useState('');
   219→  const [success, setSuccess] = useState(false);
   220→  const [showPreview, setShowPreview] = useState(false);
   221→  const [checkingAuth, setCheckingAuth] = useState(true);
   222→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   223→
   224→  // Fetch user profile & KYC status on mount
   225→  useEffect(() => {
   226→    if (!isAuthenticated || !token) {
   227→      setCheckingAuth(false);
   228→      return;
   229→    }
   230→    (async () => {
   231→      try {
   232→        const res = await fetch('/api/auth/me', {
   233→          headers: { Authorization: `Bearer ${token}` },
   234→        });
   235→        if (res.ok) {
   236→          const data = await res.json();
   237→          if (data.user) updateUser(data.user);
   238→          if (data.user?.isKycVerified) {
   239→            setKycStatus('approved');
   240→          } else {
   241→            // Check KYC application status
   242→            const kycRes = await fetch('/api/kyc', {
   243→              headers: { Authorization: `Bearer ${token}` },
   244→            });
   245→            if (kycRes.ok) {
   246→              const kycData = await kycRes.json();
   247→              if (kycData.application) {
   248→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   249→              } else {
   250→                setKycStatus('none');
   251→              }
   252→            } else {
   253→              setKycStatus('none');
   254→            }
   255→          }
   256→        }
   257→      } catch {
   258→        setKycStatus('none');
   259→      } finally {
   260→        setCheckingAuth(false);
   261→      }
   262→    })();
   263→  }, [isAuthenticated, token, updateUser]);
   264→
   265→  /* ---- derived ---- */
   266→  const platformFee = useMemo(() => {
   267→    const sp = parseFloat(form.sellingPrice);
   268→    if (!sp || sp <= 0) return 0;
   269→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   270→  }, [form.sellingPrice]);
   271→
   272→  const showSeatClass = form.transportType === 'bus';
   273→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   274→  const isCounterCopy = form.ticketType === 'counter_copy';
   275→
   276→  /* ---- setters ---- */
   277→  const set = useCallback(
   278→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   279→      setForm((prev) => {
   280→        const next = { ...prev, [key]: value };
   281→        // Reset deckType when class changes away from deck-requiring classes
   282→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   283→          next.deckType = '';
   284→        }
   285→        // Reset delivery fields when ticket type changes
   286→        if (key === 'ticketType') {
   287→          next.deliveryType = '';
   288→          next.meetingPlace = '';
   289→          next.courierName = '';
   290→          next.deliverySpeed = '';
   291→          next.deliveryChargePaidBy = 'seller';
   292→          next.deliveryCharge = '';
   293→          next.ticketDocument = '';
   294→          setUploadFileName('');
   295→        }
   296→        // Reset transport-specific fields when transport type changes
   297→        if (key === 'transportType') {
   298→          next.seatClass = '';
   299→          next.deckType = '';
   300→        }
   301→        // Reset courier fields when delivery type changes
   302→        if (key === 'deliveryType') {
   303→          next.meetingPlace = '';
   304→          next.courierName = '';
   305→          next.deliverySpeed = '';
   306→          next.deliveryCharge = '';
   307→          next.deliveryChargePaidBy = 'seller';
   308→        }
   309→        return next;
   310→      });
   311→    },
   312→    [],
   313→  );
   314→
   315→  /* ---- file upload ---- */
   316→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   317→    const file = e.target.files?.[0];
   318→    if (!file) return;
   319→
   320→    // Validate file type based on ticket type
   321→    if (form.ticketType === 'online_copy') {
   322→      if (file.type !== 'application/pdf') {
   323→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   324→        return;
   325→      }
   326→    } else {
   327→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   328→      if (!validImageTypes.includes(file.type)) {
   329→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   330→        return;
   331→      }
   332→    }
   333→
   334→    // Validate file size (10MB for PDF, 5MB for images)
   335→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   336→    if (file.size > maxSize) {
   337→      toast.error(
   338→        isBn
   339→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   340→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   341→      );
   342→      return;
   343→    }
   344→
   345→    if (!token) {
   346→      navigate('login');
   347→      return;
   348→    }
   349→
   350→    setUploading(true);
   351→    try {
   352→      const formData = new FormData();
   353→      formData.append('file', file);
   354→
   355→      const res = await fetch('/api/upload', {
   356→        method: 'POST',
   357→        headers: { Authorization: `Bearer ${token}` },
   358→        body: formData,
   359→      });
   360→
   361→      if (!res.ok) {
   362→        const data = await res.json();
   363→        throw new Error(data.error || 'Upload failed');
   364→      }
   365→
   366→      const data = await res.json();
   367→      set('ticketDocument', data.url);
   368→      setUploadFileName(file.name);
   369→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   370→    } catch (err: unknown) {
   371→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   372→    } finally {
   373→      setUploading(false);
   374→    }
   375→  };
   376→
   377→  /* ---- validation ---- */
   378→  const validate = (): string | null => {
   379→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   380→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   381→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   382→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   383→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   384→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   385→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   386→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   387→
   388→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   389→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   390→
   391→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   392→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   393→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   394→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   395→
   396→    if (isCounterCopy) {
   397→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   398→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   399→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   400→      if (form.deliveryType === 'courier') {
   401→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   402→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   403→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   404→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   405→      }
   406→    }
   407→
   408→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   409→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   410→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   411→
   412→    return null;
   413→  };
   414→
   415→  /* ---- submit ---- */
   416→  const handleSubmit = async (e: React.FormEvent) => {
   417→    e.preventDefault();
   418→    setError('');
   419→
   420→    if (!isAuthenticated || !token) {
   421→      navigate('login');
   422→      return;
   423→    }
   424→
   425→    const validationError = validate();
   426→    if (validationError) {
   427→      setError(validationError);
   428→      toast.error(validationError);
   429→      return;
   430→    }
   431→
   432→    setSubmitting(true);
   433→    try {
   434→      const body = {
   435→        transportType: form.transportType,
   436→        transportCompany: form.transportCompany.trim(),
   437→        ticketType: form.ticketType,
   438→        pnrNumber: form.pnrNumber.trim(),
   439→        ticketDocument: form.ticketDocument,
   440→        routeFrom: form.from,
   441→        routeTo: form.to,
   442→        departureDate: form.departureDate,
   443→        departureTime: form.departureTime,
   444→        boardingPoint: form.boardingPoint.trim() || undefined,
   445→        droppingPoint: form.droppingPoint.trim() || undefined,
   446→        seatClass: form.seatClass || undefined,
   447→        deckType: form.deckType || undefined,
   448→        seatNumber: form.seatNumber.trim() || undefined,
   449→        coachNumber: form.coachNumber.trim() || undefined,
   450→        originalPrice: parseFloat(form.originalPrice),
   451→        price: parseFloat(form.sellingPrice),
   452→        description: form.description.trim() || undefined,
   453→        sellerNotes: form.sellerNotes.trim() || undefined,
   454→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   455→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
   456→        courierName: isCounterCopy && form.deliveryType === 'courier' ? form.courierName : undefined,
   457→        deliverySpeed: isCounterCopy && form.deliveryType === 'courier' ? form.deliverySpeed : undefined,
   458→        deliveryChargePaidBy: isCounterCopy && form.deliveryType === 'courier' ? form.deliveryChargePaidBy : undefined,
   459→        deliveryCharge:
   460→          isCounterCopy && form.deliveryType === 'courier' && form.deliveryChargePaidBy === 'buyer'
   461→            ? parseFloat(form.deliveryCharge)
   462→            : 0,
   463→        isConfirmed: true,
   464→      };
   465→
   466→      const res = await fetch('/api/tickets', {
   467→        method: 'POST',
   468→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   469→        body: JSON.stringify(body),
   470→      });
   471→
   472→      if (!res.ok) {
   473→        const data = await res.json();
   474→        throw new Error(data.error || isBn ? 'টিকেট তৈরি ব্যর্থ' : 'Failed to create ticket');
   475→      }
   476→
   477→      setSuccess(true);
   478→      toast.success(isBn ? 'টিকেট সফলভাবে তালিকাভুক্ত হয়েছে!' : 'Ticket listed successfully!');
   479→    } catch (err: unknown) {
   480→      const msg = err instanceof Error ? err.message : isBn ? 'একটি ত্রুটি ঘটেছে' : 'An error occurred';
   481→      setError(msg);
   482→      toast.error(msg);
   483→    } finally {
   484→      setSubmitting(false);
   485→    }
   486→  };
   487→
   488→  /* ---- helpers ---- */
   489→  const getDistrictLabel = (dist: (typeof ALL_BD_DISTRICTS)[number]) =>
   490→    isBn ? dist.labelBn : dist.label;
   491→
   492→  const getTransportLabel = (id: string) => {
   493→    const tt = TRANSPORT_TYPES.find((t) => t.id === id);
   494→    return tt ? (isBn ? tt.labelBn : tt.label) : id;
   495→  };
   496→
   497→  const getClassLabel = (id: string) => {
   498→    const cls = BUS_CLASSES.find((c) => c.id === id);
   499→    return cls ? (isBn ? cls.labelBn : cls.label) : id;
   500→  };
   501→
   502→  const getDeckLabel = (id: string) => {
   503→    const dk = DECK_TYPES.find((d) => d.id === id);
   504→    return dk ? (isBn ? dk.labelBn : dk.label) : id;
   505→  };
   506→
   507→  const getCourierLabel = (id: string) => {
   508→    const cr = COURIER_COMPANIES.find((c) => c.id === id);
   509→    return cr ? (isBn ? cr.labelBn : cr.label) : id;
   510→  };
   511→
   512→  const getDeliverySpeedLabel = (id: string) => {
   513→    const ds = DELIVERY_SPEEDS.find((d) => d.id === id);
   514→    return ds ? (isBn ? ds.labelBn : ds.label) : id;
   515→  };
   516→
   517→  /* ================================================================ */
   518→  /*  AUTH GATE                                                        */
   519→  /* ================================================================ */
   520→
   521→  if (checkingAuth) {
   522→    return (
