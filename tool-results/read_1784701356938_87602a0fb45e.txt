   641→   641→function PageRouter() {
   642→   642→  const { currentPage } = useAppStore();
   643→   643→
   644→   644→  // Home page renders inline (no dynamic import)
   645→   645→  if (currentPage === 'home') {
   646→   646→    return (
   647→   647→      <AppShell>
   648→   648→        <motion.div key="home" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   649→   649→          <HomePage />
   650→   650→        </motion.div>
   651→   651→      </AppShell>
   652→   652→    );
   653→   653→  }
   654→   654→
   655→   655→  // Auth pages (no shell)
   656→   656→  if (authPages.has(currentPage)) {
   657→   657→    return (
   658→   658→      <Suspense fallback={<PageLoader />}>
   659→   659→        <motion.div key={currentPage} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   660→   660→          <LazyPage pageName={currentPage} />
   661→   661→        </motion.div>
   662→   662→      </Suspense>
   663→   663→    );
   664→   664→  }
   665→   665→
   666→   666→  // All other pages (with shell)
   667→   667→  if (pageComponents[currentPage]) {
   668→   668→    return (
   669→   669→      <AppShell>
   670→   670→        <Suspense fallback={<PageLoader />}>
   671→   671→          <motion.div key={currentPage} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   672→   672→            <LazyPage pageName={currentPage} />
   673→   673→          </motion.div>
   674→   674→        </Suspense>
   675→   675→      </AppShell>
   676→   676→    );
   677→   677→  }
   678→   678→
   679→   679→  // Fallback to home
   680→   680→  return (
   681→   681→    <AppShell>
   682→   682→      <HomePage />
   683→   683→    </AppShell>
   684→   684→  );
   685→   685→}
