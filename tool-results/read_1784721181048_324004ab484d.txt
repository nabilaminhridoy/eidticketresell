   234→export default function SellTicketPage() {
   235→  const { navigate } = useAppStore();
   236→  const { user, token, isAuthenticated, updateUser } = useAuthStore();
   237→  const { language } = useLanguageStore();
   238→  const isBn = language === 'bn';
   239→  const fontClass = isBn ? 'font-bangla' : '';
   240→
   241→  const [form, setForm] = useState<FormState>(initialForm);
   242→  const [uploading, setUploading] = useState(false);
   243→  const [uploadFileName, setUploadFileName] = useState('');
   244→  const [submitting, setSubmitting] = useState(false);
   245→  const [error, setError] = useState('');
   246→  const [success, setSuccess] = useState(false);
   247→  const [showPreview, setShowPreview] = useState(false);
   248→  const [checkingAuth, setCheckingAuth] = useState(true);
   249→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   250→  const [dragOver, setDragOver] = useState(false);
   251→  const fileInputRef = useRef<HTMLInputElement>(null);
   252→
   253→  // Fetch user profile & KYC status on mount
   254→  useEffect(() => {
   255→    if (!isAuthenticated || !token) {
   256→      setCheckingAuth(false);
   257→      return;
   258→    }
   259→    (async () => {
   260→      try {
   261→        const res = await fetch('/api/auth/me', {
   262→          headers: { Authorization: `Bearer ${token}` },
   263→        });
   264→        if (res.ok) {
   265→          const data = await res.json();
   266→          if (data.user) updateUser(data.user);
   267→          if (data.user?.isKycVerified) {
   268→            setKycStatus('approved');
   269→          } else {
   270→            const kycRes = await fetch('/api/kyc', {
   271→              headers: { Authorization: `Bearer ${token}` },
   272→            });
   273→            if (kycRes.ok) {
   274→              const kycData = await kycRes.json();
   275→              if (kycData.application) {
   276→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   277→              } else {
   278→                setKycStatus('none');
   279→              }
   280→            } else {
   281→              setKycStatus('none');
   282→            }
   283→          }
   284→        }
   285→      } catch {
   286→        setKycStatus('none');
   287→      } finally {
   288→        setCheckingAuth(false);
   289→      }
   290→    })();
   291→  }, [isAuthenticated, token, updateUser]);
   292→
   293→  /* ---- derived ---- */
   294→  const platformFee = useMemo(() => {
   295→    const sp = parseFloat(form.sellingPrice);
   296→    if (!sp || sp <= 0) return 0;
   297→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   298→  }, [form.sellingPrice]);
   299→
   300→  const isCounterCopy = form.ticketType === 'counter_copy';
   301→
   302→  const sellerReceives = useMemo(() => {
   303→    const sp = parseFloat(form.sellingPrice);
   304→    if (!sp || sp <= 0) return 0;
   305→    return sp - platformFee;
   306→  }, [form.sellingPrice, platformFee]);
   307→
   308→  // Buyer Pays depends on ticket type:
   309→  // Online Copy: Buyer Pays = Selling Price
   310→  // Counter Copy: Buyer Pays = Platform Fee
   311→  const buyerPays = useMemo(() => {
   312→    const sp = parseFloat(form.sellingPrice);
   313→    if (!sp || sp <= 0) return 0;
   314→    return isCounterCopy ? platformFee : sp;
   315→  }, [form.sellingPrice, platformFee, isCounterCopy]);
   316→
   317→  // Selling price exceeds original price check
   318→  const sellingPriceExceedsOriginal = useMemo(() => {
   319→    const sp = parseFloat(form.sellingPrice);
   320→    const op = parseFloat(form.originalPrice);
   321→    if (!sp || !op || sp <= 0 || op <= 0) return false;
   322→    return sp > op;
   323→  }, [form.sellingPrice, form.originalPrice]);
   324→
   325→  const showSeatClass = form.transportType === 'bus';
   326→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   327→
   328→  /* ---- setters ---- */
   329→  const set = useCallback(
   330→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   331→      setForm((prev) => {
   332→        const next = { ...prev, [key]: value };
   333→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   334→          next.deckType = '';
   335→        }
   336→        if (key === 'ticketType') {
   337→          next.deliveryType = '';
   338→          next.meetingPlace = '';
   339→          next.courierName = '';
   340→          next.deliverySpeed = '';
   341→          next.deliveryChargePaidBy = 'seller';
   342→          next.deliveryCharge = '';
   343→          next.ticketDocument = '';
   344→          setUploadFileName('');
   345→        }
   346→        if (key === 'transportType') {
   347→          next.seatClass = '';
   348→          next.deckType = '';
   349→        }
   350→        if (key === 'deliveryType') {
   351→          next.meetingPlace = '';
   352→          next.courierName = '';
   353→          next.deliverySpeed = '';
   354→          next.deliveryCharge = '';
   355→          next.deliveryChargePaidBy = 'seller';
   356→        }
   357→        return next;
   358→      });
   359→    },
   360→    [],
   361→  );
   362→
   363→  /* ---- file upload ---- */
   364→  const handleFileUpload = async (file: File) => {
   365→    if (form.ticketType === 'online_copy') {
   366→      if (file.type !== 'application/pdf') {
   367→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   368→        return;
   369→      }
   370→    } else {
   371→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   372→      if (!validImageTypes.includes(file.type)) {
   373→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   374→        return;
   375→      }
   376→    }
   377→
   378→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   379→    if (file.size > maxSize) {
   380→      toast.error(
   381→        isBn
   382→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   383→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   384→      );
   385→      return;
   386→    }
   387→
   388→    if (!token) {
   389→      navigate('login');
   390→      return;
   391→    }
   392→
   393→    setUploading(true);
   394→    try {
   395→      const formData = new FormData();
   396→      formData.append('file', file);
   397→
   398→      const res = await fetch('/api/upload', {
   399→        method: 'POST',
   400→        headers: { Authorization: `Bearer ${token}` },
   401→        body: formData,
   402→      });
   403→
   404→      if (!res.ok) {
   405→        const data = await res.json();
   406→        throw new Error(data.error || 'Upload failed');
   407→      }
   408→
   409→      const data = await res.json();
   410→      set('ticketDocument', data.url);
   411→      setUploadFileName(file.name);
   412→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   413→    } catch (err: unknown) {
   414→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   415→    } finally {
   416→      setUploading(false);
   417→    }
   418→  };
   419→
   420→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   421→    const file = e.target.files?.[0];
   422→    if (file) handleFileUpload(file);
   423→  };
   424→
   425→  const handleDrop = (e: React.DragEvent) => {
   426→    e.preventDefault();
   427→    setDragOver(false);
   428→    const file = e.dataTransfer.files?.[0];
   429→    if (file) handleFileUpload(file);
   430→  };
   431→
   432→  const handleDragOver = (e: React.DragEvent) => {
   433→    e.preventDefault();
   434→    setDragOver(true);
   435→  };
   436→
   437→  const handleDragLeave = () => {
   438→    setDragOver(false);
   439→  };
   440→
   441→  /* ---- validation ---- */
   442→  const validate = (): string | null => {
   443→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   444→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   445→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   446→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   447→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   448→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   449→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   450→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   451→
   452→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   453→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   454→
   455→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   456→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   457→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   458→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   459→    if (parseFloat(form.sellingPrice) > parseFloat(form.originalPrice))
   460→      return isBn ? 'বিক্রয় মূল্য মূল টিকেট মূল্যের বেশি হতে পারে না' : 'Selling price cannot exceed original ticket price';
   461→
   462→    if (isCounterCopy) {
   463→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   464→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   465→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   466→      if (form.deliveryType === 'courier') {
   467→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   468→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   469→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   470→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   471→      }
   472→    }
   473→
   474→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   475→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   476→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   477→
   478→    return null;
   479→  };
   480→
   481→  /* ---- submit ---- */
   482→  const handleSubmit = async (e: React.FormEvent) => {
   483→    e.preventDefault();
   484→    setError('');
   485→
   486→    if (!isAuthenticated || !token) {
   487→      navigate('login');
   488→      return;
   489→    }
   490→
   491→    const validationError = validate();
   492→    if (validationError) {
   493→      setError(validationError);
   494→      toast.error(validationError);
   495→      return;
   496→    }
   497→
   498→    setSubmitting(true);
   499→    try {
   500→      const body = {
   501→        transportType: form.transportType,
   502→        transportCompany: form.transportCompany.trim(),
   503→        ticketType: form.ticketType,
   504→        pnrNumber: form.pnrNumber.trim(),
   505→        ticketDocument: form.ticketDocument,
   506→        routeFrom: form.from,
   507→        routeTo: form.to,
   508→        departureDate: form.departureDate,
   509→        departureTime: form.departureTime,
   510→        boardingPoint: form.boardingPoint.trim() || undefined,
   511→        droppingPoint: form.droppingPoint.trim() || undefined,
   512→        seatClass: form.seatClass || undefined,
   513→        deckType: form.deckType || undefined,
   514→        seatNumber: form.seatNumber.trim() || undefined,
   515→        coachNumber: form.coachNumber.trim() || undefined,
   516→        originalPrice: parseFloat(form.originalPrice),
   517→        price: parseFloat(form.sellingPrice),
   518→        description: form.description.trim() || undefined,
   519→        sellerNotes: form.sellerNotes.trim() || undefined,
   520→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   521→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
