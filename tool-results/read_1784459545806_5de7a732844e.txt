   204→  const { language } = useLanguageStore();
   205→  const isBn = language === 'bn';
   206→  const fontClass = isBn ? 'font-bangla' : '';
   207→
   208→  const [form, setForm] = useState<FormState>(initialForm);
   209→  const [uploading, setUploading] = useState(false);
   210→  const [uploadFileName, setUploadFileName] = useState('');
   211→  const [submitting, setSubmitting] = useState(false);
   212→  const [error, setError] = useState('');
   213→  const [success, setSuccess] = useState(false);
   214→  const [showPreview, setShowPreview] = useState(false);
   215→  const [checkingAuth, setCheckingAuth] = useState(true);
   216→  const [kycStatus, setKycStatus] = useState<'unknown' | 'none' | 'pending' | 'approved' | 'rejected'>('unknown');
   217→  const [dragOver, setDragOver] = useState(false);
   218→  const fileInputRef = useRef<HTMLInputElement>(null);
   219→
   220→  // Fetch user profile & KYC status on mount
   221→  useEffect(() => {
   222→    if (!isAuthenticated || !token) {
   223→      setCheckingAuth(false);
   224→      return;
   225→    }
   226→    (async () => {
   227→      try {
   228→        const res = await fetch('/api/auth/me', {
   229→          headers: { Authorization: `Bearer ${token}` },
   230→        });
   231→        if (res.ok) {
   232→          const data = await res.json();
   233→          if (data.user) updateUser(data.user);
   234→          if (data.user?.isKycVerified) {
   235→            setKycStatus('approved');
   236→          } else {
   237→            const kycRes = await fetch('/api/kyc', {
   238→              headers: { Authorization: `Bearer ${token}` },
   239→            });
   240→            if (kycRes.ok) {
   241→              const kycData = await kycRes.json();
   242→              if (kycData.application) {
   243→                setKycStatus(kycData.application.status === 'pending' ? 'pending' : kycData.application.status === 'approved' ? 'approved' : kycData.application.status === 'rejected' ? 'rejected' : 'none');
   244→              } else {
   245→                setKycStatus('none');
   246→              }
   247→            } else {
   248→              setKycStatus('none');
   249→            }
   250→          }
   251→        }
   252→      } catch {
   253→        setKycStatus('none');
   254→      } finally {
   255→        setCheckingAuth(false);
   256→      }
   257→    })();
   258→  }, [isAuthenticated, token, updateUser]);
   259→
   260→  /* ---- derived ---- */
   261→  const platformFee = useMemo(() => {
   262→    const sp = parseFloat(form.sellingPrice);
   263→    if (!sp || sp <= 0) return 0;
   264→    return Math.max(PLATFORM_FEE_MINIMUM, Math.round(sp * (PLATFORM_FEE_PERCENTAGE / 100)));
   265→  }, [form.sellingPrice]);
   266→
   267→  const sellerReceives = useMemo(() => {
   268→    const sp = parseFloat(form.sellingPrice);
   269→    if (!sp || sp <= 0) return 0;
   270→    return sp - platformFee;
   271→  }, [form.sellingPrice, platformFee]);
   272→
   273→  const showSeatClass = form.transportType === 'bus';
   274→  const showDeckType = showSeatClass && DECK_REQUIRED_CLASSES.includes(form.seatClass);
   275→  const isCounterCopy = form.ticketType === 'counter_copy';
   276→
   277→  /* ---- setters ---- */
   278→  const set = useCallback(
   279→    <K extends keyof FormState>(key: K, value: FormState[K]) => {
   280→      setForm((prev) => {
   281→        const next = { ...prev, [key]: value };
   282→        if (key === 'seatClass' && !DECK_REQUIRED_CLASSES.includes(value as string)) {
   283→          next.deckType = '';
   284→        }
   285→        if (key === 'ticketType') {
   286→          next.deliveryType = '';
   287→          next.meetingPlace = '';
   288→          next.courierName = '';
   289→          next.deliverySpeed = '';
   290→          next.deliveryChargePaidBy = 'seller';
   291→          next.deliveryCharge = '';
   292→          next.ticketDocument = '';
   293→          setUploadFileName('');
   294→        }
   295→        if (key === 'transportType') {
   296→          next.seatClass = '';
   297→          next.deckType = '';
   298→        }
   299→        if (key === 'deliveryType') {
   300→          next.meetingPlace = '';
   301→          next.courierName = '';
   302→          next.deliverySpeed = '';
   303→          next.deliveryCharge = '';
   304→          next.deliveryChargePaidBy = 'seller';
   305→        }
   306→        return next;
   307→      });
   308→    },
   309→    [],
   310→  );
   311→
   312→  /* ---- file upload ---- */
   313→  const handleFileUpload = async (file: File) => {
   314→    if (form.ticketType === 'online_copy') {
   315→      if (file.type !== 'application/pdf') {
   316→        toast.error(isBn ? 'শুধুমাত্র PDF ফাইল আপলোড করুন' : 'Only PDF files can be uploaded for Online Copy');
   317→        return;
   318→      }
   319→    } else {
   320→      const validImageTypes = ['image/png', 'image/jpeg', 'image/jpg'];
   321→      if (!validImageTypes.includes(file.type)) {
   322→        toast.error(isBn ? 'শুধুমাত্র PNG/JPG/JPEG ফাইল আপলোড করুন' : 'Only PNG/JPG/JPEG files can be uploaded for Counter Copy');
   323→        return;
   324→      }
   325→    }
   326→
   327→    const maxSize = form.ticketType === 'online_copy' ? 10 * 1024 * 1024 : 5 * 1024 * 1024;
   328→    if (file.size > maxSize) {
   329→      toast.error(
   330→        isBn
   331→          ? `ফাইলের আকার ${form.ticketType === 'online_copy' ? '10MB' : '5MB'} এর কম হতে হবে`
   332→          : `File size must be under ${form.ticketType === 'online_copy' ? '10MB' : '5MB'}`,
   333→      );
   334→      return;
   335→    }
   336→
   337→    if (!token) {
   338→      navigate('login');
   339→      return;
   340→    }
   341→
   342→    setUploading(true);
   343→    try {
   344→      const formData = new FormData();
   345→      formData.append('file', file);
   346→
   347→      const res = await fetch('/api/upload', {
   348→        method: 'POST',
   349→        headers: { Authorization: `Bearer ${token}` },
   350→        body: formData,
   351→      });
   352→
   353→      if (!res.ok) {
   354→        const data = await res.json();
   355→        throw new Error(data.error || 'Upload failed');
   356→      }
   357→
   358→      const data = await res.json();
   359→      set('ticketDocument', data.url);
   360→      setUploadFileName(file.name);
   361→      toast.success(isBn ? 'ফাইল আপলোড সফল!' : 'File uploaded successfully!');
   362→    } catch (err: unknown) {
   363→      toast.error(err instanceof Error ? err.message : isBn ? 'আপলোড ব্যর্থ' : 'Upload failed');
   364→    } finally {
   365→      setUploading(false);
   366→    }
   367→  };
   368→
   369→  const handleFileInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   370→    const file = e.target.files?.[0];
   371→    if (file) handleFileUpload(file);
   372→  };
   373→
   374→  const handleDrop = (e: React.DragEvent) => {
   375→    e.preventDefault();
   376→    setDragOver(false);
   377→    const file = e.dataTransfer.files?.[0];
   378→    if (file) handleFileUpload(file);
   379→  };
   380→
   381→  const handleDragOver = (e: React.DragEvent) => {
   382→    e.preventDefault();
   383→    setDragOver(true);
   384→  };
   385→
   386→  const handleDragLeave = () => {
   387→    setDragOver(false);
   388→  };
   389→
   390→  /* ---- validation ---- */
   391→  const validate = (): string | null => {
   392→    if (!form.transportCompany.trim()) return isBn ? 'পরিবহন কোম্পানির নাম দিন' : 'Transport company name is required';
   393→    if (!form.pnrNumber.trim()) return isBn ? 'টিকেট/PNR নম্বর দিন' : 'Ticket/PNR number is required';
   394→    if (!form.ticketDocument) return isBn ? 'টিকেট নথি আপলোড করুন' : 'Ticket document upload is required';
   395→    if (!form.from) return isBn ? 'যাত্রা শুরুর স্থান নির্বাচন করুন' : 'From location is required';
   396→    if (!form.to) return isBn ? 'গন্তব্য নির্বাচন করুন' : 'To location is required';
   397→    if (form.from === form.to) return isBn ? 'শুরু ও গন্তব্য এক হতে পারে না' : 'From and To cannot be the same';
   398→    if (!form.departureDate) return isBn ? 'যাত্রার তারিখ দিন' : 'Travel date is required';
   399→    if (!form.departureTime) return isBn ? 'যাত্রার সময় দিন' : 'Departure time is required';
   400→
   401→    if (showSeatClass && !form.seatClass) return isBn ? 'ক্লাস নির্বাচন করুন' : 'Seat class is required for bus tickets';
   402→    if (showDeckType && !form.deckType) return isBn ? 'ডেক টাইপ নির্বাচন করুন' : 'Deck type is required';
   403→
   404→    if (!form.originalPrice || parseFloat(form.originalPrice) <= 0)
   405→      return isBn ? 'মূল টিকেটের মূল্য দিন' : 'Original ticket price is required';
   406→    if (!form.sellingPrice || parseFloat(form.sellingPrice) <= 0)
   407→      return isBn ? 'বিক্রয় মূল্য দিন' : 'Selling price is required';
   408→
   409→    if (isCounterCopy) {
   410→      if (!form.deliveryType) return isBn ? 'ডেলিভারি টাইপ নির্বাচন করুন' : 'Delivery type is required for counter copy';
   411→      if (form.deliveryType === 'in_person' && !form.meetingPlace.trim())
   412→        return isBn ? 'মিটিং প্লেস দিন' : 'Meeting place is required';
   413→      if (form.deliveryType === 'courier') {
   414→        if (!form.courierName) return isBn ? 'কুরিয়ার নির্বাচন করুন' : 'Courier name is required';
   415→        if (!form.deliverySpeed) return isBn ? 'ডেলিভারি ধরন নির্বাচন করুন' : 'Delivery speed is required';
   416→        if (form.deliveryChargePaidBy === 'buyer' && (!form.deliveryCharge || parseFloat(form.deliveryCharge) <= 0))
   417→          return isBn ? 'ডেলিভারি চার্জ দিন' : 'Delivery charge is required';
   418→      }
   419→    }
   420→
   421→    if (!form.confirmCorrect) return isBn ? 'টিকেট তথ্য নিশ্চিত করুন' : 'Please confirm ticket information is correct';
   422→    if (!form.confirmLegalRight) return isBn ? 'আইনি অধিকার নিশ্চিত করুন' : 'Please confirm you have the legal right to transfer';
   423→    if (!form.confirmFakeWarning) return isBn ? 'সতর্কতা স্বীকার করুন' : 'Please acknowledge the fake ticket warning';
   424→
   425→    return null;
   426→  };
   427→
   428→  /* ---- submit ---- */
   429→  const handleSubmit = async (e: React.FormEvent) => {
   430→    e.preventDefault();
   431→    setError('');
   432→
   433→    if (!isAuthenticated || !token) {
   434→      navigate('login');
   435→      return;
   436→    }
   437→
   438→    const validationError = validate();
   439→    if (validationError) {
   440→      setError(validationError);
   441→      toast.error(validationError);
   442→      return;
   443→    }
   444→
   445→    setSubmitting(true);
   446→    try {
   447→      const body = {
   448→        transportType: form.transportType,
   449→        transportCompany: form.transportCompany.trim(),
   450→        ticketType: form.ticketType,
   451→        pnrNumber: form.pnrNumber.trim(),
   452→        ticketDocument: form.ticketDocument,
   453→        routeFrom: form.from,
   454→        routeTo: form.to,
   455→        departureDate: form.departureDate,
   456→        departureTime: form.departureTime,
   457→        boardingPoint: form.boardingPoint.trim() || undefined,
   458→        droppingPoint: form.droppingPoint.trim() || undefined,
   459→        seatClass: form.seatClass || undefined,
   460→        deckType: form.deckType || undefined,
   461→        seatNumber: form.seatNumber.trim() || undefined,
   462→        coachNumber: form.coachNumber.trim() || undefined,
   463→        originalPrice: parseFloat(form.originalPrice),
   464→        price: parseFloat(form.sellingPrice),
   465→        description: form.description.trim() || undefined,
   466→        sellerNotes: form.sellerNotes.trim() || undefined,
   467→        deliveryType: isCounterCopy ? form.deliveryType : undefined,
   468→        meetingPlace: isCounterCopy && form.deliveryType === 'in_person' ? form.meetingPlace.trim() : undefined,
   469→        courierName: isCounterCopy && form.deliveryType === 'courier' ? form.courierName : undefined,
   470→        deliverySpeed: isCounterCopy && form.deliveryType === 'courier' ? form.deliverySpeed : undefined,
   471→        deliveryChargePaidBy: isCounterCopy && form.deliveryType === 'courier' ? form.deliveryChargePaidBy : undefined,
   472→        deliveryCharge:
   473→          isCounterCopy && form.deliveryType === 'courier' && form.deliveryChargePaidBy === 'buyer'
   474→            ? parseFloat(form.deliveryCharge)
   475→            : 0,
   476→        isConfirmed: true,
   477→      };
   478→
   479→      const res = await fetch('/api/tickets', {
   480→        method: 'POST',
   481→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   482→        body: JSON.stringify(body),
   483→      });
   484→
   485→      if (!res.ok) {
   486→        const data = await res.json();
   487→        throw new Error(data.error || (isBn ? 'টিকেট তৈরি ব্যর্থ' : 'Failed to create ticket'));
   488→      }
   489→
   490→      setSuccess(true);
   491→      toast.success(isBn ? 'টিকেট সফলভাবে তালিকাভুক্ত হয়েছে!' : 'Ticket listed successfully!');
   492→    } catch (err: unknown) {
   493→      const msg = err instanceof Error ? err.message : isBn ? 'একটি ত্রুটি ঘটেছে' : 'An error occurred';
   494→      setError(msg);
   495→      toast.error(msg);
   496→    } finally {
   497→      setSubmitting(false);
   498→    }
   499→  };
   500→
   501→  /* ---- helpers ---- */
   502→  const getDistrictLabel = (dist: (typeof ALL_BD_DISTRICTS)[number]) =>
   503→    isBn ? dist.labelBn : dist.label;
   504→
   505→  const getClassLabel = (id: string) => {
   506→    const cls = BUS_CLASSES.find((c) => c.id === id);
   507→    return cls ? (isBn ? cls.labelBn : cls.label) : id;
   508→  };
   509→
   510→  const getDeckLabel = (id: string) => {
   511→    const dk = DECK_TYPES.find((d) => d.id === id);
   512→    return dk ? (isBn ? dk.labelBn : dk.label) : id;
   513→  };
   514→
   515→  const getCourierLabel = (id: string) => {
   516→    const cr = COURIER_COMPANIES.find((c) => c.id === id);
