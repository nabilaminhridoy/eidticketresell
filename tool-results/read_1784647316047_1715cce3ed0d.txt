   522→// Page Router
   523→function PageRouter() {
   524→  const { currentPage } = useAppStore();
   525→
   526→  // Home page renders inline (no dynamic import)
   527→  if (currentPage === 'home') {
   528→    return (
   529→      <AppShell>
   530→        <motion.div key="home" initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   531→          <HomePage />
   532→        </motion.div>
   533→      </AppShell>
   534→    );
   535→  }
   536→
   537→  // Auth pages (no shell)
   538→  if (authPages.has(currentPage)) {
   539→    return (
   540→      <Suspense fallback={<PageLoader />}>
   541→        <motion.div key={currentPage} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   542→          <LazyPage pageName={currentPage} />
   543→        </motion.div>
   544→      </Suspense>
   545→    );
   546→  }
   547→
   548→  // All other pages (with shell)
   549→  if (pageComponents[currentPage]) {
   550→    return (
   551→      <AppShell>
   552→        <Suspense fallback={<PageLoader />}>
   553→          <motion.div key={currentPage} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   554→            <LazyPage pageName={currentPage} />
   555→          </motion.div>
   556→        </Suspense>
   557→      </AppShell>
   558→    );
   559→  }
   560→
   561→  // Fallback to home
   562→  return (
   563→    <AppShell>
   564→      <HomePage />
   565→    </AppShell>
   566→  );
   567→}
   568→
   569→export default function Home() {
   570→  return <PageRouter />;
   571→}
