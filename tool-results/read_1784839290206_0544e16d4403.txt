    12→async function generatePrefixedId(prefix: string): Promise<string> {
    13→  const counterName = `${prefix.toLowerCase()}_seq`;
    14→  const counter = await db.counter.upsert({
    15→    where: { name: counterName },
    16→    update: { value: { increment: 1 } },
    17→    create: { name: counterName, value: 1 },
    18→  });
    19→  return `${prefix}-${counter.value}`;
    20→}
